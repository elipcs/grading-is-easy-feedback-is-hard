% Two-Stage Prompting Strategy Figure - Simplified
\scalebox{0.8}{
\begin{tikzpicture}[
    node distance=0.6cm and 0.8cm,
    inputbox/.style={rectangle, draw, thick, fill=blue!10, minimum width=2.5cm, minimum height=0.6cm, align=center, font=\tiny},
    stagebox/.style={rectangle, draw, thick, fill=green!15, minimum width=3cm, minimum height=1.8cm, align=left, font=\tiny},
    outputbox/.style={rectangle, draw, thick, fill=orange!10, minimum width=2.5cm, minimum height=0.6cm, align=center, font=\tiny},
    arrow/.style={->, >=stealth, thick},
    label/.style={font=\tiny\bfseries}
]

% ============================================================================
% INPUTS (Top)
% ============================================================================

\node[inputbox] (code) at (0,0) {Student Code};
\node[inputbox, right=0.3cm of code] (rubric) {Rubric};
\node[inputbox, right=0.3cm of rubric] (spec) {Spec};

% ============================================================================
% STAGE 1: SCORING PASS
% ============================================================================

\node[stagebox, below=0.8cm of rubric] (stage1) {
    \textbf{Stage 1: Scoring}\\[0.1cm]
    • Analyze code vs. criteria\\
    • Assign scores (0-10)\\
    • Identify problems\\
    • Output: JSON
};

\node[outputbox, below=of stage1] (json-output) {JSON: Scores\\+ Problems};

% ============================================================================
% STAGE 2: FEEDBACK PASS
% ============================================================================

\node[stagebox, below=0.8cm of json-output] (stage2) {
    \textbf{Stage 2: Feedback}\\[0.1cm]
    • Transform to pedagogy\\
    • Explain violations\\
    • Suggest improvements\\
    • Output: Natural language
};

\node[outputbox, below=of stage2] (feedback-output) {Student-Facing\\Feedback};

% ============================================================================
% ARROWS
% ============================================================================

% Inputs to Stage 1
\draw[arrow] (code) -- (stage1);
\draw[arrow] (rubric) -- (stage1);
\draw[arrow] (spec) -- (stage1);

% Stage 1 to JSON
\draw[arrow] (stage1) -- (json-output);

% JSON to Stage 2
\draw[arrow] (json-output) -- (stage2);

% Stage 2 to Feedback
\draw[arrow] (stage2) -- (feedback-output);

% ============================================================================
% CONFIGURATION BOX (Bottom)
% ============================================================================

\node[below=0.3cm of feedback-output, draw, thick, fill=yellow!10, 
      minimum width=6cm, minimum height=0.5cm, align=center, font=\tiny] (config) {
    \textbf{Config:} Gemini 3.1 Pro | Temp=0 | MaxTokens=4096
};

\end{tikzpicture}
}
