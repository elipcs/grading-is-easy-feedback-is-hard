package filmnow;

import java.util.Objects;

/**
 * Abstração dos filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class Filme {
	
	/**
	* Atributo que guarda o nome do filme.
	*/
	String nome;
	/**
	* Atributo que guarda o ano de lançamento.
	*/
	int ano;
	/**
	* Atributo que guarda o local onde o filme pode ser assistido.
	*/
	String local;
	
	/**
	* Construtor de filme.
	* 
	* @param nome nome do filme
	* @param ano ano em que o filme foi lançado
	* @param local local aonde o filme pode ser assistido
	*/
	public Filme(String nome, int ano, String local) {
		super();
		this.nome = nome;
		this.ano = ano;
		this.local = local;
	}
	
	public String getNome() {
		return nome;
	}

	public void setNome(String nome) {
		this.nome = nome;
	}

	public int getAno() {
		return ano;
	}

	public void setAno(int ano) {
		this.ano = ano;
	}

	public String getLocal() {
		return local;
	}

	public void setLocal(String local) {
		this.local = local;
	}
	
	public String nomeAno() {
		return nome + ", " + ano;
	}

	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}

	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return ano == other.ano && Objects.equals(nome, other.nome);
	}

	@Override
	public String toString() {
		return nome + ", " + ano + "\n"
			   + local;
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	private static final int TAMANHO_HOT = 10;
	
	private Filme[] filmes; //uma representacao simploria da lista de filmes
	/**
	* Lista de filmes com maior preferência.
	*/
	private Filme[] hotFilmes;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotFilmes = new Filme[TAMANHO_HOT];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes;
	}
	
	/**
	 * Acessa a lista de hot filmes mantida.
	 * @return O array de hot filmes.
	 */
	public Filme[] getHotFilmes() {
		return this.hotFilmes;
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		return filmes[posicao];
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, int ano, String local) {
		if (posicao > 100 || posicao < 1) {
			return "POSIÇÃO INVÁLIDA";
		} else if (nome == "" || local == "") {
			return "FILME INVALIDO";
		} else {
			Filme novoFilme = new Filme(nome, ano, local);
			if (existeFilme(novoFilme, filmes)) {
				return "FILME JA ADICIONADO";
			} else {
				this.filmes[posicao] = novoFilme;
				return "FILME ADICIONADO";
			}
		}
	}
	
	/**
	 * Testa se o filme já exite no array filmes. 
	 * 
	 * @param novoFilme filme que queremos saber se já existe no Array.
	 * 
	 * @return boolean que representa se o filme existe ou não na lista de filmes.
	 */
	public boolean existeFilme(Filme novoFilme, Filme [] filmes) {
		for (Filme filme: filmes) {
			if (novoFilme.equals(filme)) {
				return true;
			}
		}
		return false;
	}
	
	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param filme posição do filme na lista comum
	 * @param posicao posição que será adicionado na hotList
	 */
	public String atribuirHotFilme(int filme, int posicao) {
		if (filmes[filme] == null) {
			return "POSIÇÃO INVÁLIDA";
		} else if (existeFilme(filmes[filme], hotFilmes)) {
			return "FILME JA ADICIONADO";
		} else {
			this.hotFilmes[posicao] = filmes[filme];
			return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicao + "!";
		}
	}
	
	/**
	 * Remove um filme da hotList na posição passada. 
	 * @param posicao posição do filme a ser removido.
	 */
	public String removerHotFilme(int posicao) {
		if (hotFilmes[posicao] == null) {
			return "POSIÇÃO INVÁLIDA";
		} 
		hotFilmes[posicao] = null;
		return "FILME REMOVIDO";
	}
	
	/**
	 * Imprime os detalhes de um filme. 
	 * 
	 * @param posicao posicao em que o filme está na lista de filmes
	 */
	public String detalharFilme(int posicao) {
		Filme filme = getFilme(posicao);
		String retorno = "";
		if (existeFilme(filme, getHotFilmes())) {
			retorno += "🔥 ";
		}
		return retorno + filme;
	}
	
	/**
	 * Imprime lista de filmes.
	 */
	public String mostrarFilmes() {
		String retorno = "";
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				retorno += formataFilme(i, filmes[i]) + "\n";
			}
		}
		return retorno;
	}
	
	/**
	 * Exibe os Filmes atribuidos como "Hot".
	 */
	public String exibirHotList() {
		String retorno = "";
		for (int i = 0; i < hotFilmes.length; i++) {
			if (hotFilmes[i] != null) {
				retorno += formataFilme(i, hotFilmes[i]) + "\n";
			}
		}
		return retorno;
	}
	
	/**
	 * Formata um filme para impressão. 
	 * 
	 * @param posicao A posição do filme (que é exibida)/
	 * @param filme O filme a ser impresso.
	 * @return A String formatada.
	 */
	private static String formataFilme(int posicao, Filme filme) {
		return posicao + " - " + filme.getNome();
	}

}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}
