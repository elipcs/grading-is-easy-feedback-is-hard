package filmnow;

import java.util.Objects;

/**
 * Classe que compoe os atributos e metodos que representam um filme.
 * @author [REDACTED]
 */

public class Filme {
	
	private String nome; // String que representa o nome do filme
	private int ano; // Inteiro que representa o ano de lançamento do filme 
	private int posicao; // inteiro que representa a POSICAO REAL DO FILME EM UM ARRAY!
	private String local; // Local onde o filme foi assistido pelo usuario
	
	/**
	 * Cria filme.
	 */
	
	public Filme(String nome, String ano, int posicao, String local) {
		this.nome = nome;
		this.ano = Integer.parseInt(ano);
		this.posicao = posicao - 1;
		this.local = local;
		
	}
	
	/**
	 * Acessa o nome do filme
	 * @return String representando nome do filme
	 */
	
	public String getNome() {
		return this.nome;
	}
	
	/**
	 * Acessa o ano de lançamento do filme
	 * @return Inteiro que representa o ano de lançamento do filme
	 */
	
	public int getAno() {
		return this.ano;
	}
	
	/**
	 * Acessa posicao real do filme em um array
	 * @return Inteiro que representa a posicao do filme
	 */
	
	public int getPosicao() {
		return this.posicao;
	}
	
	/**
	 * Acessa o local onde o filme foi assistido 
	 * @return String indicando o local onde foi assistindo
	 */
	
	public String getLocal() {
		return this.local;
	}
	
	/**
	 * Retorna um representaçao textual do objeto, no formato NOME, ANO \nLOCAL
	 */
	
	public String toString() {
		if (this.ano == 0) {
			return this.nome + "\n" + this.local;
		}
		return this.nome + ", " + this.ano + "\n" + this.local;
	}

	/**
	 * Retorna um codigo hash com base no ano e nome do objeto.
	 */
	
	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}

	/**
	 * Compara se dois objetos do sao iguais e do mesmo tipo, com base no nome e ano do filme
	 * @param filme Objeto do tipo Filme a ser comparado 
	 * @return Valor boleano idicando se o os dois Objetos sao iguais
	 */
	
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return ano == other.ano && Objects.equals(nome, other.nome);
	}
	
	

}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	
	private Filme[] filmes; //uma representacao simploria da lista de filmes
	private Filme[] filmesHot; //uma representaçao da lista de filmes hot 

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.filmesHot = new Filme[10];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		return filmes[posicao - 1];
		
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if (ano == "" ) {
			ano = "0";
		}
		if (nome == "") {
			return "FILME INVALIDO";
		}
		
		if (local == "" ) {
			return "FILME INVALIDO"; 
		}
		
		Filme filme = new Filme(nome, ano, posicao, local);
		
		if (posicao < 1 || posicao > 100) { 
			return "POSICAO INVALIDA"; 
		}
		if (inFilmes(filme)) {
			this.filmes[filme.getPosicao()] = filme;
			return "FILME JA ADICIONADO";
		}
		this.filmes[filme.getPosicao()] = filme;
		return "FILME ADICIONADO";
	}
	
	/**
	 * Adiciona um filme em uma posicao de uma lista de preferencia do usuario. Caso ja 
	 * exista um filme na posicao, ele sera sobrescrito.
	 * @param posicaoFilme valor inteiro da posicao do filme no array filmes.
	 * @param posicaoHot valor inteiro da posicao do filme em um array filmesHot.
	 */
	
	public String cadastraHot(int posicaoFilme, int posicaoHot) {
		posicaoHot -= 1;
		posicaoFilme -= 1;
		if (posicaoHot < 0 || posicaoHot > 9) {
			return "POSICAO INVALIDA";
		}
		if (posicaoFilme < 0 || posicaoFilme > 99) {
			return "POSICAO INVALIDA";
		}
		if (filmesHot[posicaoHot] != null) {
			removerHot(posicaoHot);
		}
		filmesHot[posicaoHot] = filmes[posicaoFilme];
		return "FILME ADICIONADO";
	}
	
	/**
	 * Acessa a array de filmes filmesHot.
	 * @return retorna uma copia do array filmesHot.
	 */
	
	public Filme[] getFilmesHot() {
		return this.filmesHot.clone();
	}

	/**
	 * Acessa as informaçoes de um filme (caso exista) a partir da posicao, e verifica
	 * se o filme tambem esta presente na lista filmesHot
	 * @param posicao Inteiro que representa a posicao do filme no array de filmes
	 * @return Uma representaçao textual do filme
	 */
	
	public String detalharFilmeNow(int posicao) {
		if (posicao < 1 || posicao > 100) {
			return "POSICAO INVALIDA";
		}
		
		Filme filme = getFilme(posicao);
		
		if (filme == null) {
			return "";
		}
		if (inHot(filme)) {
			return "🔥 " + filme.toString();
		} else {
			return filme.toString();
		}
	}
	
	/**
	 * Verifica se um filme esta presente no array de filmesHot.
	 * @param filme Filme a ser procurado no array.
	 * @return Retorna um valor boleano referente a presença ou nao do filme.
	 */
	
	public boolean inHot(Filme filme) {
		for (Filme filmeHot : this.filmesHot) {
			if (filmeHot != null) {
				if (filmeHot == filme) {
					return true;
				}
			}
		}
		return false;
	}
	
	/**
	 * Verifica se um filme esta contido no array de filmes
	 * @param filme Filme a ser verificado
	 * @return Valor boleano indicando a presença ou nao do filme
	 */
	
	public boolean inFilmes(Filme filme) {
		for (Filme filme1 : this.filmes) {
			if (filme1 != null) {
				if (filme1.equals(filme)) {
					return true;
				}
			}
		}
		return false;
	}

	/**
	 * Remove um filme do array filmesHot, e reorganiza os demais filmes para
	 * ocupar o espaço removido.
	 * @param posicao Valor inteiro que representa a posicao do filme no array
	 */
	
	public void removerHot(int posicao) {
		posicao = posicao - 1;
		filmesHot[posicao] = null;
		while (true) {
			if (filmesHot[posicao + 1] == null || posicao >= 9) {
				filmesHot[posicao] = null;
				break;
			} else {
				filmesHot[posicao] = filmesHot[posicao + 1];
				posicao++;
			}
		}
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
