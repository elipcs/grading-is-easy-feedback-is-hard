package filmnow;

import java.util.Objects;

/**
 * Representação cada filme armazenado na lista de filmes.
 * @author [REDACTED]
 */
public class Filme {
	private String nome;
	private String anoLancamento;
	private String local;
	private boolean temHot;
	
	/**
	 * Constrói o filme a partir do sou nome, ano de lançamento, local onde ele pode ser encontrado e se ele tem o status hot ou não.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser encontrado.
	 */
	public Filme(String nome, String anoLancamento, String local) {
		this.nome = nome;
		this.anoLancamento = anoLancamento;
		this.local = local;
		this.temHot = false;
	}

	/**
	 * Retorna nome do filme.
	 * @return nome do filme.
	 */
	public String getNome() {
		return nome;
	}
	
	/**
	 * Retorna ano de lançamento do filme.
	 * @return ano de lançamento do filme.
	 */
	public String getAno() {
		return anoLancamento;
	}
	
	/**
	 * Retorna local onde pode ser encontrado o filme.
	 * @return ano de lançamento do filme.
	 */
	public String getLocal() {
		return local;
	}
	
	/**
	 * Atribui ao filme o status hot ou retira esse status. true se o filme tem hot, false caso não tenha.
	 * @param temHot Status hot do filme.
	 */
	public void setTemHot(boolean temHot) {
		this.temHot = temHot;
	}
	
	/**
	 * Retorna o status hot do filme (true se tem hot, false senão tem).
	 * @return o status hot do filme.
	 */
	public boolean getTemHot() {
		return temHot;
	}
	
	/**
	 * Constrói os detalhes de um filme específico, com seu nome e local onde pode ser encontrado.
	 * @return representação dos detalhes de um filme específico, seja ele com ano de lançamento informado ou não.
	 */
	public String exibeFilme() {
		if (anoLancamento.isEmpty()) {
			return String.format("%s\n%s", nome, local);

		} else {
			return String.format("%s, %s\n%s", nome, anoLancamento, local);
		}
	}
	
	/**
	 * Constrói os detalhes de um filme específico com status hot.
	 * @return representação dos detalhes de um filme específico com status hot.
	 */
	public String exibeFilmeHot() {
		return String.format("%s, %s\n", nome, anoLancamento);
	}
	
	/**
	 * Gera um código identificador para cada filme, considerando seu nome e local onde pode ser encontrado.
	 */
	@Override
	public int hashCode() {
		return Objects.hash(local, nome);
	}

	/**
	 * Compara conteúdo de dois filmes a partir do seu ano de lançamento e o seu nome.
	 */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return Objects.equals(anoLancamento, other.anoLancamento) && Objects.equals(nome, other.nome);
	}

	/**
	 * Gera uma representação para cada filme a partir do seu nome.
	 */
	@Override
	public String toString() {
		return nome;
	}


}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	
	private Filme[] filmes; //Uma representacao simploria da lista de filmes.
	private Integer[] hotList; //Representação das posições de cada objeto filme com status hot.
	
	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Integer[10];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes, formatado com sua posição na lista e o seu nome.
	 */
	public String[] getFilmes() {
		String[] todosFilmes = new String[TAMANHO];
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				todosFilmes[i] = i + 1 + "-" + " " + filmes[i].getNome();
			}
		}
		return todosFilmes;
	}
	
	/**
	 * Acessa os dados de um filme específico.
	 * @param pos Posição do filme no sistema.
	 * @return Dados do filme, tendo uma formatação específica para quando ele tem o status hot ou não. "Posição inválida" se não há filme na posição.
	 */
	public String getFilme(int pos) {
		if (pos < 1 || pos > 100) {
			return "POSIÇÃO INVÁLIDA";
		} else if (filmes[pos - 1] != null) {
			if (filmes[pos - 1].getTemHot()) {
				return "🔥" + " " + filmes[pos - 1].exibeFilme();
			} else {
				return filmes[pos - 1].exibeFilme();
			}
		} else {
			return "";
		}
	}
	
	/**
	 * Adiciona a referência do filme em hotList, além de sinalizar aos seus atributos que agora ele tem o status hot. "Posição inválida" se não há filme na posição.
	 * @param posFilme Posição do filme na lista filmes.
	 * @param posHotList Posição em hotList que o usuário deseja armazenar a referência (posição) de um filme específico.
	 * @return A Informação de se o filme foi adicionado na hotList ou se não foi possível, caso ele já exista na lista.
	 */
	public String adicionaHot(int posFilme, int posHotList) {
		if (filmes[posFilme - 1] == null || posHotList < 1 || posHotList > 10) {
			return "POSIÇÃO INVÁLIDA";
		} else {
			for (int i = 0; i < hotList.length; i++ ) {
				if (hotList[i] != null &&
					posFilme == hotList[i]) { //compara por posições, já que hotList armazena as referências para os objetos.
					return "FILME JÁ ADICIONADO";
				}
			}
			hotList[posHotList - 1] = posFilme;
			filmes[posFilme - 1].setTemHot(true);
			return "ADICIONADO À HOTLIST NA POSIÇÃO" + " " + posHotList + "!";
		}
	}
	
	/**
	 * Remove o hot de um filme e a sua referência em hotList, caso por opção direta do usuário ou pelo fato do filme que tinha hot ter sido sobrescrito. 
	 * @param removerPorHotList True caso a remoção seja por opção direta do usuário, False se for por sobrescrita do filme, fazendo assim o segundo bloco ser executado.
	 * @param posRemover Posição para remover pode assumir valor da posição em hotlist ou posição do filme sobrescrito no array filmes.
	 */ 
	public void removerDaHotList(boolean removerPorHotList, int posRemover) {
		if (removerPorHotList) {
			filmes[hotList[posRemover -1] - 1].setTemHot(false);
			hotList[posRemover - 1] = null;
		} else {
			for (int i = 0; i < hotList.length; i++) {
				if (hotList[i] != null &&
					posRemover == hotList[i]) {
					hotList[i] = null; 
				}
			}
		}
	}
	
	/**
	 * Acessa a lista de filmes com hot mantida.
	 * @return o array hotList, formatado com a posição de cada filme com hot e o seu nome.
	 */
	public String[] mostrarHotList() {
		String [] filmesHots = new String[10];
		for (int i = 0; i < hotList.length; i++) {
			if (hotList[i] != null) { 
				filmesHots[i] = i+1 + " " + "-" + " " + filmes[hotList[i] - 1].exibeFilmeHot();
			}
		}
		return filmesHots;
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assistido.
	 */
	public String cadastraFilme(int posicao, String nome, String anoLancamento, String local) {
		Filme filme = new Filme(nome, anoLancamento, local);
		if (posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		} else if (nome.isEmpty() || local.isEmpty()) {
			return "FILME INVÁLIDO";
		} else {
			for (int i = 0; i < filmes.length; i++) {
				if (filmes[i] != null &&
					filmes[i].equals(filme)) {
					return "FILME JÁ ADICIONADO";
				}
			}
			if (filmes[posicao - 1] != null) {
				removerDaHotList(false, posicao); //remover a referência do filme na hotList, caso ele tenha sido sobrescrito.
			}
			filmes[posicao - 1] = filme;
			return "FILME ADICIONADO";
		}
	}
		
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String anoLancamento = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, anoLancamento, local);
	}

}

						"(H)Atribuir Hot\n" +
