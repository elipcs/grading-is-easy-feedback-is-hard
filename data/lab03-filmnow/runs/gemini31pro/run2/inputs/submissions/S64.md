package filmnow;

import java.util.Objects;

/**
 *  Classe que guarda os dados de um filme.
 *  
 *  @author [REDACTED]
 */

public class Filme {
	
	private String nome;
	private String ano;
	private String local;
	private String hot;
	
	/**
	 * Constroi um objeto nome com os parametros passados.
	 * 
	 * @param nome nome do filme.
	 * @param ano ano de lançamento do filme.
	 * @param local local onde pode assistir filme.
	 */
	
	public Filme(String nome,String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.hot = "";
	}
	
	/**
	 * Acessa o filme e retorna o nome do filme.
	 * 
	 * @return nome do filme.
	 */
	
	public String getFilme() {
		return this.nome;
	}
	
	/**
	 * Acessa o filme e retorna o ano de lançamento do filme.
	 * 
	 * @return o ano de lançamento do filme.
	 */
	
	public String getAno() {
		return this.ano;
	}
	
	@Override
	public String toString() {
		if(this.ano.equals("")) {
			return this.hot + this.nome + "\n" + this.local;
		}else {
			return this.hot + this.nome + ", " + this.ano +  "\n" + this.local;
		}
	}
	
	/**
	 * Acessa o filme e define o hot.
	 * 
	 * @param hot string para definir se é hot ou não.
	 */
	
	public void setHot(String hot) {
		this.hot = hot;
	}
	
	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}

	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);
	}



}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos.
 * 
 * @author [REDACTED]
 */
public class FilmNow {
	
	private Filme[] filmes;
	private Filme[] hotList;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[100];
		this.hotList = new Filme[10];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * 
	 * @return O string com todos os filmes.
	 */
	public String getFilmes() {
		String filmesString = "";
		for(int i = filmes.length-1; i > 0; i--) {
			if(filmes[i] != null) {
				if(filmesString.equals("")) {
					filmesString += "\n" + i + " - " + filmes[i].getFilme(); 
				}else if(primeiroFilme() == i) {
					filmesString = i + " - " + filmes[i].getFilme() + filmesString; 
					
				}else {
					filmesString ="\n" + i + " - " + filmes[i].getFilme() + filmesString; 
				}
			}
		}
		
		return filmesString;
	}

	/**
	 * Acessa os dados de um filme específico.
	 * 
	 * @param posicao Posição do filme no sistema.
	 * 
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public String getFilme(int posicao) {
		
		if(posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		}else if(filmes[posicao-1] == null) {
			return null;
		}else {
			return filmes[posicao-1].toString();
		}
	}
	
	public String getAno(int posicao) {
		return filmes[posicao].getAno();
	}
	
	/**
	* Verifica se o filme existe no array filmes.
	 * 
	 * @param nome nome do filme.
	 * @param ano ano de lançamento do filme.
	 * 
	 * @return true se o filme existe e false se não existe.
	 */
	public boolean filmeExiste(String nome, String ano) {
		for(int i = 0; i < this.filmes.length; i++) {
			if(filmes[i] != null) {
				if(filmes[i].getAno().equals(ano) && filmes[i].getFilme().equals(nome)) {
					return true;
				}
			}
		}
		return false;
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 * 
	 * @return se os dados passam por todas as especicações retorna FILME JA ADICIONADO,
	 * se não volta FILME INVALIDO.
	 */
	public String cadastraFilme(String posicao, String nome, String ano, String local) {
		if((Integer.parseInt(posicao) > 100 || Integer.parseInt(posicao) < 1) && ano.getClass() == nome.getClass()) {
			return "POSIÇÃO INVÁLIDA";
		}else{
			
			// Nome não é vazio?
			if(nome.equals("")) {
				return "FILME INVALIDO";
			}else {
				
				// Filme já existe no array?
				if(filmeExiste(nome, ano)) {
					return "FILME JÁ ADICIONADO";
				}else {
					
					// Local é vazio?
					if(!local.equals("")) {
						filmes[Integer.parseInt(posicao)-1] = new Filme(nome,ano,local);
						return "FILME ADICIONADO";
					}else {
						return "FILME INVALIDO";
					}
				}
			}

		}
		
	}
	
	/**
	 * Adiciona um filme na lista hotList. Se já existir filme na posição, retorna Filme já existe.
	 * 
	 * @param posicaoFilme indice onde o filme está na lista filmes.
	 * @param posicao posição onde o filme deve ser adicionado na lista hotList.
	 * 
	 * @return se o filme já existe na lista hotList retorna FILME JÁ EXISTE NA HOTLISTH!, 
	 * se não retorna ADICIONADO À HOTLIST NA POSIÇÃO posicao!.
	 */
	public String adicionaHot(int posicaoFilme, int posicao) {
		
		if(existeHot(filmes[posicaoFilme-1])) {
			return "FILME JÁ EXISTE NA HOTLISTH!";
		}else {
			filmes[posicaoFilme-1].setHot("🔥");
			hotList[posicao-1] = filmes[posicaoFilme-1]; 
			return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicao +"!";
		}
		
	}
	
	/** Acessa a lista hotList e mostrar todos elemento do objeto.
	 * 
	 * @retorn string com todos os filmes da hotList.
	*/
	public String mostrarHot() {
		String hotString = "";
		for(int i = 0; i > hotList.length; i++) {
			System.out.println(i);
			if(hotList[i] != null) {
				if(ultimoHot() != i) {
					hotString += i + " - " + hotList[i].getFilme() + ", " + hotList[i].getAno() + "\n";
				}else {
					hotString += i + " - " + hotList[i].getFilme() + ", " + hotList[i].getAno();
				}
			}
		}
		return hotString;
	}
	
	/** Acessa a lista hotList para achar o ultimo indice não nulo do Array.
	 * 
	 * @return o numero do ultimo indice não nulo do Array, se não achar retorna 0.
	*/
	private int ultimoHot() {
		for(int i = hotList.length-1; i > -1; i--) {
			if(hotList[i] != null) {
				return i;
			}
		}
		return 0;
	}
	
	/** Acessa a lista hotList e remove elemento de uma posição.
	 * 
	 * @param posicao indice do elemento que deve ser removido.
	*/
	public void removeHot(int posicao) {
		
		this.hotList[posicao] = null;
		filmes[posicao].setHot("");
		
	}
	
	/** Acessa e percorre a hotList para verificar se filme existe nela.
	 * 
	 * @param filme objeto Filme
	 * 
	 * @return se o existir na lista retorna true, se não retorna false.
	*/
	private boolean existeHot(Filme filme) {
		for(int i = 0; i < hotList.length; i++) {
			if(hotList[i] != null) {
				if(filme.equals(hotList[i])) {
					return true;
				}
			}
			
		}
		return false;
	}
	
	/**Acessa a lista hotList para achar o primeiro indice não nulo do Array.
	 * 
	 * @return o numero do ultimo indice não nulo do Array, se não achar retorna 0.
	*/
  	private int primeiroFilme() {
		for(int i = 0; i < filmes.length; i++) {
			if(filmes[i] != null) {
				return i;
			}
		}
		return 0;
	}

}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		String posicao = campos[COLUNA_POSICAO];
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
