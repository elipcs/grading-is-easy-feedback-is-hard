package filmnow;

import java.util.Objects;

/**
 * Classe para a criação do objeto filme com suas informações.
 * 
 *  @author [REDACTED]
 */

public class Filme {
	private String nomefilme;
	private String ano;
	private String plataforma;
	private boolean hotlist = false;

	/**
	 * Construtor da classe Filme.
	 * 
	 * @param nomefilme Nome do filme.
	 * @param ano Ano de lançamento.
	 * @param plataforma Plataforma para ter acesso ao filme.
	 */
	public Filme(String nomefilme, String ano, String plataforma) {
		this.nomefilme = nomefilme;
		this.ano = ano;
		this.plataforma = plataforma;
	}

	/**
	 * Método para retornar a string do objeto.
	 * @return string com as informações do filme.
	 */
	@Override
	public String toString() {
		if(this.ano.isBlank()==false) {
			return this.nomefilme + ", " + this.ano + "/n" + this.plataforma;
		}else {
			return this.nomefilme + "/n" + this.plataforma;
		}
	}

	/**
	 * Método para retornar o nome do filme.
	 * @return o nome.
	 */
	public String getNome() {
		return this.nomefilme;
	}

	/**
	 * Mostra se o filme está em alguma hotlist ou não.
	 * @return hotlist.
	 */
	public boolean getHotlist() {
		return this.hotlist;
	}

	/**
	 * Muda o valor da hotlist quando um objeto for adicionado no mesmo.
	 */
	public void adicionadoHotlist() {
		this.hotlist = true;
	}

	/**
	 * Muda o valor da hotlist quando um objeto for removido do mesmo.
	 */
	public void removidoHotlist() {
		this.hotlist = false;
	}
	@Override
	public int hashCode() {
		return Objects.hash(ano, nomefilme);
	}
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return ano == other.ano && Objects.equals(nomefilme, other.nomefilme);
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	private static final int TAMANHOHOTLIST = 10;
	
	private Filme[] filmes; //uma representacao simploria da lista de filmes
	private Filme[] hotlist;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotlist = new Filme[TAMANHOHOTLIST];
	}

	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		return filmes[posicao-1];
	}


	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		String retorno;
		Filme filme = new Filme(nome, ano, local);
		if (posicao > 100 || posicao < 1) {
			retorno = "POSIÇÃO INVÁLIDA";
		}
		else {
			if(nome.isBlank()) {
				retorno = "FILME INVALIDO";
			}
			else {
				if(local.isBlank()) {
					retorno = "FILME INVALIDO";
				}
				else {
					if(checaIgual(nome, ano, local)==false) {
						this.filmes[posicao-1] = filme;
						retorno = "FILME ADICIONADO";
					}
					else {
						retorno = "FILME JÁ ADICIONADO";
					}
				}
			}
		}
		return retorno;
	}

	/**
	 * Analisa se tem algum filme armazenado em determinada posição ou não.
	 * 
	 * @param posicao Posição de análise.
	 * @return retorna true se tiver um filme e false se não tiver.
	 */
	public boolean existeFilme(int posicao) {
		if(this.filmes[posicao-1] == null) {
			return false;
		}
		return true;
	}
	
	/**

	 * Analisa se tem algum filme igual a algum que já está armazenado, através do nome e do ano.

	 * 

	 * @param nome Nome do filme.

	 * @param ano Ano de lançamento.

	 * @param local Plataforma para assistir.

	 */
	public boolean checaIgual(String nome, String ano, String local) {
		Filme filme = new Filme(nome, ano, local);
		for(Filme f: this.filmes) {
			if(f!=null) {
				if(filme.equals(f)) {
					return true;
				}
			}
		}
		return false;
	}
	
	/**
	 * Adiciona um filme na hotlist.
	 * 
	 * @param posicaoFilme qual posição o filme está na lista.
	 * @param posicaoHot qual posição o filme deve ser inserido.
	 * @return resultado do processamento.
	 */
	public String adicionaHot(int posicaoFilme, int posicaoHot) {
		String retorno;
		if(this.filmes[posicaoFilme-1].getHotlist()==false) {
			if(this.hotlist[posicaoHot-1]== null) {
				this.hotlist[posicaoHot-1] = this.filmes[posicaoFilme-1];
				this.filmes[posicaoFilme-1].adicionadoHotlist();
				retorno = String.format("ADICIONADO À HOTLIST NA POSIÇÃO %d!", posicaoHot);
			}
			else {
				this.hotlist[posicaoHot-1].removidoHotlist();
				this.hotlist[posicaoHot-1] = this.filmes[posicaoFilme-1];
				this.hotlist[posicaoHot-1].adicionadoHotlist();
				retorno = String.format("ADICIONADO À HOTLIST NA POSIÇÃO %d!", posicaoHot);
			}
		}
		else {
			retorno = "FILME JÁ ADICIONADO!";
		}
	return retorno;
	}
	
	/**
	 * Remove um filme da Hotlist
	 * @param posicao Posição do filme a ser removido.
	 * @return Mostra que foi removido.
	 */
	public String removeHot(int posicao) {
		String retorno = "FILME REMOVIDO!";
		this.hotlist[posicao-1].removidoHotlist();
		this.hotlist[posicao -1] = null;
		return retorno;
	}

	/**
	 * Retorna a Hotlist.
	 * @return Uma array dos filmes.
	 */
	public Filme[] getHot() {
		return this.hotlist.clone();
	}
	
	/**
	 * Busca as informações do filme a partir de sua posição.
	 * @param posicao Posição do filme.
	 * @return Retorna as informações do filme.
	 */
	public String detalhaFilme(int posicao) {
		String retorno;
		if(posicao < 1 || posicao > 100) {
			retorno = "POSIÇÃO INVÁLIDA!";
		}
		else {
			Filme filme = getFilme(posicao);
			if(existeFilme(posicao)== false) {
				return "";
			}
			else {
				if(filme.getHotlist()) {
					retorno = "🔥 " + filme.toString();
				}
				else {
					retorno = filme.toString();
				}
			}
		}
		return retorno;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

package filmnow;

		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */

		System.out.println(
				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" + 
						"(E)Exibir HotList\n"+
						"(H)Atribuir Hot\n"+
						"(R)Remover Hot\n"+
						"(S)air\n"+
						"\n" + 
						"Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.

			mostrarFilmes(fn);
			break;
		case "D":

			exibirHot(fn);
			break;

			adicionaHot(fn, scanner);
			break;
		case "R":
			removeHot(fn, scanner);
			break;

			sai();
			break;
