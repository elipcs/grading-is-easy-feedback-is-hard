package filmnow;



/**
 * Representa um filme com informações básicas como nome, ano e local de exibição.
 * 
 * @author [REDACTED]
 */
public class Filme {
    
    private String nome; // Nome do filme
    private String ano;  // Ano de lançamento do filme
    private String local; // Local onde o filme pode ser assistido
    
    /**
     * Cria uma instância de Filme com os detalhes fornecidos.
     * 
     * @param nome Nome do filme.
     * @param ano Ano de lançamento do filme.
     * @param local Local onde o filme pode ser assistido.
     */
    public Filme(String nome, String ano, String local) {
        this.nome = nome;
        this.ano = ano;
        this.local = local;
    }

    /**
     * Retorna o nome do filme.
     * 
     * @return O nome do filme.
     */
    public String getNome() {
        return nome;
    }

    /**
     * Define o nome do filme.
     * 
     * @param nome O nome a ser definido para o filme.
     */
    public void setNome(String nome) {
        this.nome = nome;
    }

    /**
     * Retorna o ano de lançamento do filme.
     * 
     * @return O ano de lançamento do filme.
     */
    public String getAno() {
        return ano;
    }

    /**
     * Define o ano de lançamento do filme.
     * 
     * @param ano O ano a ser definido para o filme.
     */
    public void setAno(String ano) {
        this.ano = ano;
    }

    /**
     * Retorna o local onde o filme pode ser assistido.
     * 
     * @return O local de exibição do filme.
     */
    public String getLocal() {
        return local;
    }

    /**
     * Define o local onde o filme pode ser assistido.
     * 
     * @param local O local a ser definido para o filme.
     */
    public void setLocal(String local) {
        this.local = local;
    }

    /**
     * Retorna uma representação em String do filme.
     * 
     * @return Uma String contendo o nome e o ano do filme, seguido pelo local de exibição.
     */
    @Override
    public String toString() {
        return nome + "," + ano + "\n" + local;
    }

    /**
     * Compara este filme com outro objeto para verificar se são iguais.
     * 
     * @param obj O objeto com o qual comparar este filme.
     * @return {@code true} se o objeto for um filme com o mesmo nome e ano, {@code false} caso contrário.
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (obj == null || getClass() != obj.getClass()) {
            return false;
        }
        Filme filme = (Filme) obj;
        return this.nome.equals(filme.nome) &&
               this.ano.equals(filme.ano);
    }

}

package filmnow;

/**
 * Sistema que mantém os filmes prediletos do usuário. Pode armazenar até 100 filmes
 * e uma lista de 10 filmes favoritos (HOT).
 * 
 * @author [REDACTED]
 */
public class FilmNow {
    
    private static final int TAMANHO = 100;
    private static final int TAMANHOHOT = 10;
    
    private Filme[] filmes; // Representação simplória da lista de filmes
    private Filme[] filmesHot; // Lista dos filmes favoritos (HOT)
    
    /**
     * Cria uma instância do sistema FilmNow.
     */
    public FilmNow() {
        this.filmes = new Filme[TAMANHO];
        this.filmesHot = new Filme[TAMANHOHOT];
    }
    
    /**
     * Acessa a lista de filmes mantida pelo sistema.
     * 
     * @return O array de filmes, uma cópia do array interno.
     */
    public Filme[] getFilmes() {
        return this.filmes.clone();
    }

    /**
     * Acessa os dados de um filme específico.
     * 
     * @param posicao Posição do filme no sistema.
     * @return Dados do filme como uma String. Retorna {@code null} se não há filme na posição.
     */
    public String getFilme(int posicao) {
        return filmes[posicao] != null ? filmes[posicao].toString() : null;
    }

    /**
     * Acessa o objeto de um filme específico.
     * 
     * @param posicao Posição do filme no sistema.
     * @return O objeto do filme na posição especificada. Retorna {@code null} se não há filme na posição.
     */
    public Filme getFilmeOBJ(int posicao) {
        return filmes[posicao];
    }

    /**
     * Acessa a lista de filmes HOT.
     * 
     * @return O array de filmes HOT, uma cópia do array interno.
     */
    public Filme[] getHot() {
        return this.filmesHot.clone();
    }

    /**
     * Adiciona um filme em uma posição específica. Se já existir um filme na posição,
     * o filme anterior é sobrescrito.
     * 
     * @param posicao Posição onde o filme será adicionado.
     * @param nome Nome do filme.
     * @param ano Ano de lançamento do filme.
     * @param local Local onde o filme pode ser assistido.
     */
    public void cadastraFilme(int posicao, String nome, String ano, String local) {
        filmes[posicao] = new Filme(nome, ano, local);
    }

    /**
     * Adiciona um filme à lista HOT em uma posição específica.
     * 
     * @param posicao Posição do filme na lista de filmes.
     * @param posicaoHot Posição onde o filme será adicionado na lista HOT.
     */
    public void cadastraHot(int posicao, int posicaoHot) {
        filmesHot[posicaoHot] = filmes[posicao];
    }

    /**
     * Verifica se um filme pode ser removido da lista HOT com base no índice.
     * 
     * @param indice Índice do filme na lista HOT.
     * @return {@code true} se o índice for válido e houver um filme na posição, {@code false} caso contrário.
     */
    public boolean verificaRemover(int indice) {
        return indice >= 0 && indice < TAMANHOHOT && filmesHot[indice] != null;
    }

    /**
     * Remove um filme da lista HOT na posição especificada.
     * 
     * @param indice Índice do filme na lista HOT a ser removido.
     */
    public void removerHot(int indice) {
        if (verificaRemover(indice)) {
            filmesHot[indice] = null;
        }
    }

    /**
     * Verifica se um filme já está na lista HOT.
     * 
     * @param filmeASerAdicionado O filme a ser verificado.
     * @return {@code true} se o filme já está na lista HOT, {@code false} caso contrário.
     */
    public boolean adicionadoHot(Filme filmeASerAdicionado) {
        for (Filme element : filmesHot) {
            if (element != null && element.equals(filmeASerAdicionado)) {
                return true;
            }
        }
        return false;
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

/**
 * Interface com menus texto para manipular o sistema FilmNow.
 * 
 * @author [REDACTED]
 */
public class MainFilmNow {

    public static void main(String[] args) {
        FilmNow fn = new FilmNow();

        System.out.println("Carregando filmes ...");
        try {
            /*
             * Essa é a maneira de lidar com possíveis erros por falta do arquivo. 
             */
            carregaFilmes("filmes_inicial.csv", fn);
        } catch (FileNotFoundException e) {
            System.err.println("Arquivo não encontrado: " + e.getMessage());
        } catch (IOException e) {
            System.err.println("Erro lendo arquivo: " + e.getMessage());
        }

        Scanner scanner = new Scanner(System.in);
        String escolha = "";
        while (true) {
            escolha = menu(scanner);
            comando(escolha, fn, scanner);
        }
    }

    /**
     * Exibe o menu e captura a escolha do usuário.
     * 
     * @param scanner Para captura da opção do usuário.
     * @return O comando escolhido.
     */
    private static String menu(Scanner scanner) {
        System.out.println(
                "\n---\nMENU\n" + 
                "(A) Adicionar filme\n" + 
                "(M) Mostrar todos\n" + 
                "(D) Detalhar filme\n" + 
                "(E) Exibir HotList\n" +
                "(H) Atribuir Hot\n" +
                "(R) Remover Hot\n" +
                "(S) Sair\n" + 
                "\n" + 
                "Opção> ");
        return scanner.next().toUpperCase();
    }

    /**
     * Interpreta a opção escolhida pelo usuário.
     * 
     * @param opcao   Opção digitada.
     * @param fn  O sistema FilmNow que estamos manipulando.
     * @param scanner Objeto scanner para o caso do comando precisar de mais input.
     */
    private static void comando(String opcao, FilmNow fn, Scanner scanner) {
        switch (opcao) {
        case "A":
            adicionaFilme(fn, scanner);
            break;
        case "M":
            mostrarFilmes(fn);
            break;
        case "D":
            detalharFilme(fn, scanner);
            break;
        case "E":
            exibirHot(fn);
            break;
        case "H":
            adicionarHot(fn, scanner);
            break;
        case "S":
            sai();
            break;
        case "R":
            removeHot(fn, scanner);
            break;
        default:
            System.out.println("OPÇÃO INVÁLIDA!");
        }
    }

    /**
     * Imprime lista de filmes.
     * 
     * @param fn O sistema FilmNow a ser manipulado.
     */
    private static void mostrarFilmes(FilmNow fn) {
        Filme[] filmes = fn.getFilmes();
        for (int i = 0; i < filmes.length; i++) {
            if (filmes[i] != null) {
                System.out.println(formataFilme(i, filmes[i]));
            }
        }
    }

    /**
     * Formata um filme para impressão.
     * 
     * @param posicao A posição do filme (que é exibida).
     * @param filme O filme a ser impresso.
     * @return A String formatada.
     */
    private static String formataFilme(int posicao, Filme filme) {
        return posicao + " - " + filme.getNome();
    }

    /**
     * Formata um filme da lista HOT para impressão.
     * 
     * @param posicao A posição do filme (que é exibida).
     * @param filme O filme a ser impresso.
     * @return A String formatada.
     */
    private static String formataHot(int posicao, Filme filme) {
        return posicao + " - " + filme.getNome() + ", " + filme.getAno();
    }

    /**
     * Cadastra um filme no sistema.
     * 
     * @param fn O sistema FilmNow a ser manipulado.
     * @param scanner Scanner para pedir informações do filme.
     */
    private static void adicionaFilme(FilmNow fn, Scanner scanner) {
        System.out.print("\nPosição no sistema> ");
        int posicao = scanner.nextInt();
        scanner.nextLine();
        System.out.print("\nNome> ");
        String nome = scanner.nextLine();
        System.out.print("\nAno> ");
        String ano = scanner.nextLine();
        System.out.print("\nLocal> ");
        String local = scanner.nextLine();
        fn.cadastraFilme(posicao, nome, ano, local);
    }

    /**
     * Adiciona um filme à lista HOT.
     * 
     * @param fn O sistema FilmNow a ser manipulado.
     * @param scanner Scanner para capturar informações do filme e da posição HOT.
     */
    private static void adicionarHot(FilmNow fn, Scanner scanner) {
        System.out.println("\nFilme>");
        int posicao = scanner.nextInt();
        System.out.println("\nPosicao>");
        int posicaoHot = scanner.nextInt();

        Filme filmeASerAdicionado = fn.getFilmeOBJ(posicao);

        if (fn.adicionadoHot(filmeASerAdicionado)) {
            System.out.println("Filme já adicionado");
            return;
        }
        fn.cadastraHot(posicao, posicaoHot);
    }

    /**
     * Exibe a lista de filmes HOT.
     * 
     * @param fn O sistema FilmNow a ser manipulado.
     */
    private static void exibirHot(FilmNow fn) {
        Filme[] filmes = fn.getHot();
        for (int i = 0; i < filmes.length; i++) {
            if (filmes[i] != null) {
                System.out.println(formataHot(i, filmes[i]));
            }
        }
    }

    /**
     * Sai da aplicação.
     */
    private static void sai() {
        System.out.println("\nVlw flw o/");
        System.exit(0);
    }

    /**
     * Remove um filme da lista HOT.
     * 
     * @param fn O sistema FilmNow a ser manipulado.
     * @param scanner Scanner para capturar o índice do filme a ser removido.
     */
    public static void removeHot(FilmNow fn, Scanner scanner) {
        System.out.println("Posicao>");
        int indice = scanner.nextInt();
        if (!fn.verificaRemover(indice)) {
            System.out.println("Índice inválido ou não há filme para remover.");
            return;
        }
        fn.removerHot(indice);
        System.out.println("Filme removido do HOT com sucesso.");
    }

    /**
     * Lê carga de filmes de um arquivo CSV.
     * 
     * @param arquivoFilmes O caminho para o arquivo.
     * @param fn O sistema FilmNow a ser populado com os dados.
     * @throws IOException Caso o arquivo não exista ou não possa ser lido.
     */
    private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
        LeitorFilmNow leitor = new LeitorFilmNow();
        int carregados = leitor.carregaContatos(arquivoFilmes, fn);
        System.out.println("Carregamos " + carregados + " registros.");
    }

    /**
     * Detalha um filme, adicionando um emoji de fogo se o filme estiver na lista HOT.
     * 
     * @param fn O sistema FilmNow a ser manipulado.
     * @param scanner Scanner para capturar a posição do filme.
     */
    private static void detalharFilme(FilmNow fn, Scanner scanner) {
        System.out.print("\nQual filme> ");
        int posicao = scanner.nextInt();
        Filme filme = fn.getFilmeOBJ(posicao);
        boolean isHot = fn.adicionadoHot(filme);

        String nomeFilme = (isHot ? "🔥 " : "") + filme.toString();

        System.out.println("Dados do filme:\n" + nomeFilme);
    }
}
