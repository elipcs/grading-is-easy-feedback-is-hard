package filmnow;

import java.util.Objects;

/**
 * Representa um filme no sistema FilmNow.
 * 
 * Um filme possui um nome, ano de lançamento e local. Além disso, pode ser marcado como "Hot".
 * 
 * @author [REDACTED]
 */

public class Filme {
	/** 
    * O nome do filme. 
    */
	private String nome;
	/** 
	* O ano de lançamento do filme.
	*/
	private int ano;
	/** 
	* O local de onde o filme é. 
	* */
	private String local;
	/** 
	* Indica se o filme está na HotList. 
	* */
	private boolean isHot;

	/**
     * Construtor para criar um novo Filme.
     * 
     * @param nome O nome do filme.
     * @param ano O ano de lançamento do filme.
     * @param local O local de onde o filme é.
     */
	public Filme(String nome, int ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.isHot = false;
		
	}
	/**
     * Obtém o nome do filme.
     * 
     * @return O nome do filme.
     */
	public String getNome() {
		return nome;
	}

	 /**
     * Obtém o ano de lançamento do filme.
     * 
     * @return O ano de lançamento do filme.
     */
	public int getAno() {
		return ano;
	}

	 /**
     * Obtém o local de onde o filme é.
     * 
     * @return O local de onde o filme é.
     */
	
	public String getLocal() {
		return local;
	}
	
	/**
     * Verifica se o filme está na HotList.
     * 
     * @return true se o filme está na HotList, caso contrário, false.
     */
	
	public boolean getIsHot() {
		return this.isHot;
	}
	
	/**
     * Define se o filme está na HotList.
     * 
     * @param isHot true se o filme está na HotList, caso contrário, false.
     */
	
	public void setIsHot(boolean isHot) {
        this.isHot = isHot;
    }
	
	/**
     * Calcula o hash code do filme com base no nome e ano.
     * 
     * @return O valor hash code do filme.
     */
	
	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}
	/**
     * Verifica se o objeto fornecido é igual a este filme.
     * 
     * @param obj O objeto a ser comparado.
     * @return true se os objetos são iguais, false caso contrário.
     */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return ano == other.ano && Objects.equals(nome, other.nome);
	}

	/**
     * Retorna os detalhes do filme.
     * 
     * Se o filme estiver na HotList, um emoji de fogo será adicionado ao início do nome do filme.
     * 
     * @return Uma string contendo os detalhes do filme.
     */
	public String detalhesFilme() {
		if(isHot) {
			return "🔥 " + this.nome + ", " + this.ano + "\n" + this.local;
		}else { 
			return this.nome + ", " + this.ano + "\n" + this.local;
		}
	}
	
	/**
     * Retorna uma representação de string do filme (nome e ano).
     * 
     * @return Uma string contendo o nome e o ano do filme.
     */
	
	@Override
	public String toString() {
		return this.nome + ", " + this.ano;
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * O sistema FilmNow permite aos usuários manterem uma lista de seus filmes prediletos. 
 * Ele suporta até 100 filmes e fornece funcionalidades como adicionar filmes, visualizar 
 * detalhes de filmes, adicionar filmes à HotList e remover filmes da HotList.
 * 
 * A classe mantém duas listas principais: uma lista de todos os filmes e uma HotList, 
 * que contém os filmes mais populares.
 * 
 * @author [REDACTED]
 * 
 */
 
public class FilmNow {
	/** O tamanho máximo da lista de filmes. */
	private static final int TAMANHO = 100;
	
	/** O tamanho máximo da HotList. */
	private static final int TAMANHO_HOTLIST = 10;
	
	
	 /** A lista de todos os filmes. */
	private Filme[] filmes; //uma representacao simploria da lista de filmes
	
	/** A HotList que contém os filmes mais populares. */
	private Filme[] hotList;
	
	 /**
     * Construtor para criar uma instância de FilmNow.
     * 
     * Inicializa as listas de filmes e HotList.
     */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[TAMANHO_HOTLIST];
	}
	
	/**
    * Obtém a lista de todos os filmes mantida pelo sistema.
    * 
    * @return O array de filmes.
    */
	public Filme[] getFilmes() {
		return this.filmes;
	}

	/**
     * Obtém os dados de um filme específico.
     * 
     * @param posicao A posição do filme no sistema.
     * @return Os dados do filme. Null se não há filme na posição.
     */
	public Filme getFilme(int posicao) {
		return filmes[posicao - 1];
	}
	
	/**
     * Obtém a HotList que contém os filmes mais populares.
     * 
     * @return O array de filmes da HotList.
     */
	public Filme[] getHotList() {
		return hotList;
	}
	
	/**
     * Define a HotList.
     * 
     * @param hotList O array de filmes para ser definido como HotList.
     */
	public void setHotList(Filme[] hotList) {
		this.hotList = hotList;
	}

	/**
     * Adiciona um novo filme na lista de filmes.
     * 
     * @param posicao A posição onde o filme será adicionado.
     * @param nome O nome do filme.
     * @param ano O ano de lançamento do filme.
     * @param local O local onde o filme pode ser assistido.
     * @return Uma mensagem indicando o resultado da operação.
     */
	public String cadastraFilme(int posicao, String nome, int ano, String local) {
		Filme novoFilme = new Filme(nome, ano, local);
		if (posicao < 1 || posicao > TAMANHO){
			return "POSIÇÃO INVÁLIDA";
		} else if (nome.trim().isEmpty() || local.trim().isEmpty()) {
			return "FILME INVÁLIDO";
		} else if (existeFilme(novoFilme)) {
			return "FILME JÁ ADICIONADO";
		}
		
		this.filmes[posicao - 1] = novoFilme;
		return "FILME ADICIONADO";
	}
	
	/**
     * Verifica se um filme já existe na lista de filmes.
     * 
     * @param filme O filme a ser verificado.
     * @return true se o filme já existe na lista, false caso contrário.
     */
	private boolean existeFilme(Filme filme) {
		for (int i = 0; i < this.filmes.length; i++) {
			Filme f = this.filmes[i];
			if (f != null) {
				if(f.equals(filme)) {
					return true;
				}
			}
		}
		return false;
	}
	
	/**
     * Adiciona um filme à HotList.
     * 
     * @param posicaoHotList A posição onde o filme será adicionado na HotList.
     * @param posicao A posição do filme na lista de filmes.
     * @return Uma mensagem indicando o resultado da operação.
     */
	public String adicionaHot(int posicaoHotList, int posicao){
		Filme filme = this.filmes[posicaoHotList - 1];
		if(posicao < 1 || posicao > TAMANHO_HOTLIST) {
			return "POSIÇÃO INVÁLIDA!";
		}
		
		for (Filme f : hotList) {
			if (f != null && f.equals(filme)) {
				return "FILME JÁ NA HOTLIST";
			}
		}
		hotList[posicao - 1] = filme;
		filme.setIsHot(true);
		return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicao + "!";
		
	}
	
	/**
     * Remove um filme da HotList.
     * 
     * @param posicaoHotList A posição do filme na HotList.
     * @return true se o filme foi removido com sucesso, false caso contrário.
     */
	public boolean removeHot(int posicaoHotList) {
	    Filme filme = this.hotList[posicaoHotList - 1];
	    
	    if (filme != null) {
	        this.hotList[posicaoHotList - 1] = null;
	        filme.setIsHot(false);
	        return true; 
	    }

	    return false; 
	}
	
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}

 *

public class MainFilmNow {

		Scanner scanner = new Scanner(System.in);

		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */
	
	private static String menu(Scanner scanner) {
		System.out.println(
				"\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" + 
						"(E)Exibir HotList\n" + 
						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)air\n" + 
						"\n" + 
						"Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opcao   Opção digitada.
	 * @param fn  O sistema FilmNow que estamos manipulando.

			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibeHotList(fn, scanner);
			break;
		case "H":
			adicionaHotList(fn, scanner);
			break;
		case "R":
			removeHotList(fn, scanner);
			break;
		case "S":
			sai();
			break;

			System.out.println("Opção INVÁLIDA!");
		}
	}

	/**
