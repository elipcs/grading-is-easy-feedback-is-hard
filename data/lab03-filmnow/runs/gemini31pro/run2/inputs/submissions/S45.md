/**

import java.util.Objects;

public class Filmes {
	
	private String nome;
	private String ano;
	private String local;
	
	/**
	 * Cria os filmes que contem nome ano e local
	 * @param nome Da Nome ao filme
	 * @param ano Ano em que o filme foi lançado
	 * @param local Local onde o filme esta disponivel
	 */
	public Filmes(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
	}
	
	/**
	 * Pega o nome do filme em questão
	 * @return Retorna o atributo nome do filme
	 */
	public String getNome() {
		return this.nome;
	}
	
	/**
	 * Pega o Ano do filme em questão
	 * @return Retorna o atributo ano do filme
	 */
	public String getAno() {
		return this.ano;
	}
	
	/**
	 * Pega um Array de String com todas as informações do filme
	 * @return  Retorna o nome, ano e local do filme em forma de Array String
	 */
	public String[] detalhesString() {
		String[] detalhes = {this.nome, this.ano, this.local};
		return detalhes;
	}
	
	/**
	 * Override do hashCode baseado nos atributos nome e ano
	 */
	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}
	
	/**
	 * Override do equals comparando os atributos nome e ano
	 */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filmes other = (Filmes) obj;
		return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	
	private Filmes[] filmes; //uma representacao simploria da lista de filmes
	private Filmes[] hotList;

	/**
	 * Cria o FilmNow que contem um array de todos os Filmes e uma HotList (Array de 10 filmes especificos).
	 */
	public FilmNow() {
		this.filmes = new Filmes[TAMANHO];
		this.hotList = new Filmes[10];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public String[] getFilmes() {
		String[] listaFilmes = new String[100];
		for(int i = 0; i < listaFilmes.length; i++) {
			if(this.filmes[i] != null) {
				listaFilmes[i] = this.filmes[i].getNome();
			}
		}
		return listaFilmes;
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dependendo do caso retorna: Dados do filme. POSICAO INVÁLIDA. FILME INEXISTENTE. saida com emoji Hot. saida comum.
	 */
	public String getFilme(String posicao) {
		if(!ehPosicaoValida(posicao)) {
			return "POSICAO INVÁLIDA";
		}
		if(this.filmes[Integer.parseInt(posicao) - 1] == null) {
			return "FILME INEXISTENTE";
		}
		String[] detalhesFilme = this.filmes[Integer.parseInt(posicao) - 1].detalhesString();
		if(ehHot(this.filmes[Integer.parseInt(posicao) - 1])) {
			return "🔥 " + detalhesFilme[0] + ", " + detalhesFilme[1] + "\n" + detalhesFilme[2];
		}
		if(detalhesFilme[1].equals("")) {
			return detalhesFilme[0] + "\n" + detalhesFilme[2];
		}
		return  detalhesFilme[0] + ", " + detalhesFilme[1] + "\n" + detalhesFilme[2];
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. Contem saidas para casos como POSICAO INVALIDA, FILME INVALIDO e FILME JA EXISTENTE. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(String posicao, String nome, String ano, String local) {
		if(!ehPosicaoValida(posicao)) {
			return("POSIÇÃO INVÁLIDA");
		}
		
		if(nome.equals("") || local.equals("") || (!ano.equals("") && !ehAnoValido(ano))) {
			return("FILME INVÁLIDO");
		}
		
		Filmes filme = new Filmes(nome, ano, local);
		
		if(temFilme(filme)) {
			return("FILME JÁ ADICIONADO");
		}
		

		
		this.filmes[Integer.parseInt(posicao) - 1] = filme;
		return("FILME ADICIONADO");
	}
	
	/**
	 * Adiciona um filme na lista de Hot.
	 * @param posicaoFilme Posicao do filme a ser adicionado
	 * @param posicaoHot Posicao do HotList a ser alocado
	 * @return Retorna saida para casos como FILME INVALIDO, POSICAO INVALIDA, FILME INEXISTENTE, FILME JA ADICIONADO e ADICIONAR FILME.
	 */
	public String adicionaHot(String posicaoFilme, String posicaoHot) {
		if(!ehPosicaoValida(posicaoFilme)) {
			return("FILME INVÁLIDO");
		}
		if(!ehPosicaoHotValida(posicaoHot)) {
			return("POSICAO INVÁLIDA");
		}
		if(this.filmes[Integer.parseInt(posicaoFilme) - 1] == null){
			return("FILME INEXISTENTE");
		}
		if(ehHot(this.filmes[Integer.parseInt(posicaoFilme) - 1])) {
			return("FILME JÁ ADICIONADO");
		}
		this.hotList[Integer.parseInt(posicaoHot) - 1] = this.filmes[Integer.parseInt(posicaoFilme) - 1];
		return("ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!");
	}
	
	/**
	 * Transcreve toda a HotList para String.
	 * @return Retorna saida para casos como HOTLIST VAZIA e LISTA HOTLIST.
	 */
		public String hotToString() {
		String saida = "";
		for(int i = 0; i < this.hotList.length; i++) {
			if(this.hotList[i] != null) {
				saida += "\n" + (i+1) + " - " + this.hotList[i].getNome() + ", " + this.hotList[i].getAno();
			}
		}
		if(saida.equals("")) {
			return("HOTLIST VAZIA");
		}
		return saida;
	}
	
	/**
	* Retira um filme de um espaco alocado na HotList.
	* @param posicaoHot Posicao a ser esvaziada.
	* @return Retorna saidas para casos como POSICAO INVALIDA, FILME INEXISTENTE e FILME REMOVIDO
	*/
	public String deletaHot(String posicaoHot) {
		if(!ehPosicaoHotValida(posicaoHot)) {
			return("POSICAO INVÁLIDA");
		}
		if(this.hotList[Integer.parseInt(posicaoHot) - 1] == null) {
			return("FILME INEXISTENTE");
		}
		this.hotList[Integer.parseInt(posicaoHot) - 1] = null; 
		return("FILME REMOVIDO");
	}
	
	/**
	 * Metodo privado para analisar se um filme existe no array.
	 * @param filme Filme a ser buscado.
	 * @return Retorna se existe ou não o filme.
	 */
	private boolean temFilme(Filmes filme) {
		for(int i = 0; i < this.filmes.length; i++) {
			if(filme.equals(filmes[i])) {
				return true;
			}
		}
		return false;
	}
	
	/**
	 * Metodo privado para analisar se uma posicao recebida é valida.
	 * @param posicao Posicao a ser analisada.
	 * @return Retorna se é valida ou não.
	 */
	private boolean ehPosicaoValida(String posicao) {
		try {Integer.parseInt(posicao);
		} catch(NumberFormatException e) {
			return false;
		}
		
		int posicaoNum = Integer.parseInt(posicao);
				
		if( posicaoNum < 1 || posicaoNum > 100) {
			return false;
		}
		return true;
	}
	
	/**
	 * Metodo privado para alanisar se um ano recebido é valido.
	 * @param ano Ano a ser analisado.
	 * @return Retorna se é valido ou não.
	 */
	private boolean ehAnoValido(String ano) {
		try {Integer.parseInt(ano);
		} catch(NumberFormatException e) {
			return false;
		}
		return true;
	}
	
	/**
	 * Metodo privado para analisar se uma posicao recebida é valida na HorList.
	 * @param posicao Posicao a ser analisada.
	 * @return Retorna se é valida ou não.
	 */
	private boolean ehPosicaoHotValida(String posicao) {
		try {Integer.parseInt(posicao);
		} catch(NumberFormatException e) {
			return false;
		}
		
		int posicaoNum = Integer.parseInt(posicao);
				
		if( posicaoNum < 1 || posicaoNum > 10) {
			return false;
		}
		return true;
	}
	
	/**
	 * Metodo privado para ver se um filme esta presente na HotList.
	 * @param filme Filme a ser buscado.
	 * @return Retorna se esta ou não presente.
	 */
	private boolean ehHot(Filmes filme) {
		for(Filmes filmeHot : this.hotList) {
			if(filmeHot != null) {
				if(filmeHot.equals(filme)) {
					return true;
				}
			}
		}
		return false;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados - 1;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		String posicao = campos[COLUNA_POSICAO].trim();
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
