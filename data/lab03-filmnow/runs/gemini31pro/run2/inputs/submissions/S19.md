package filmnow;

import java.util.Objects;

/**
 * Representa um filme com nome, ano de lançamento e local de exibição.
 */
public class Filme {
	private String nome;
	private String ano;
	private String local;

	/**
	 * Construtor da classe.
	 *
	 * @param nome  O nome do filme.
	 * @param ano   O ano de lançamento do filme.
	 * @param local O local onde o filme está disponível.
	 */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
	}

	/**
	 * Retorna o nome do filme.
	 *
	 * @return O nome do filme.
	 */
	public String getNome() {
		return this.nome;
	}

	/**
	 * Retorna o ano de lançamento do filme.
	 *
	 * @return O ano do filme.
	 */
	public String getAno() {
		return this.ano;
	}

	/**
	 * Retorna o local onde o filme está disponível.
	 *
	 * @return O local de exibição do filme.
	 */
	public String getLocal() {
		return this.local;
	}

	/**
	 * Verifica se este filme é igual a outro objeto.
	 * Dois filmes são considerados iguais se possuem o mesmo nome e o mesmo ano.
	 *
	 * @param o O objeto a ser comparado.
	 * @return true se os filmes forem iguais, false caso contrário.
	 */
	@Override
	public boolean equals(Object o) {
		if (o == null) {
			return false;
		}

		if (!(o instanceof Filme)) {
			return false;
		}

		Filme oFilme = (Filme) o;

		return this.nome.equals(oFilme.getNome()) && this.ano.equals(oFilme.getAno());
	}

	/**
	 * Retorna o valor do hash code para o filme.
	 * Baseado no nome e no ano do filme.
	 *
	 * @return O valor do hash code.
	 */
	@Override
	public int hashCode() {
		return Objects.hash(this.nome, this.ano);
	}

	/**
	 * Retorna a representação em string do filme.
	 *
	 * @return Uma string com o nome, ano e local do filme.
	 */
	@Override
	public String toString() {
		String str = "";
		str += nome;

		if (!ano.isEmpty()) {
			str += ", " + ano + "\n";
		} else {
			str += "\n";
		}

		str += local;
		return str;
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes. Suporta até 100 filmes cadastrados
 * e uma lista dos favoritos chamada "Hotlist" com 10 filmes.
 * 
 * Cada filme possui nome, ano de lançamento e local onde pode ser assistido.
 * 
 */
public class FilmNow {

	private static final int TAMANHO = 100;
	private static final int hotTamanho = 10;

	private Filme[] filmes;     // Representação da lista de filmes.
	private int[] hotFilmes;    // Lista de posições de filmes favoritos.

	/**
	 * Cria uma nova instância do sistema FilmNow, com 100 posições para filmes e 10 para a hotlist.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotFilmes = new int[hotTamanho];
	}

	/**
	 * Retorna a lista completa de filmes cadastrados (cópia).
	 * 
	 * @return Um array contendo todos os filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Retorna um filme específico baseado em sua posição.
	 * 
	 * @param posicao A posição do filme (de 1 a 100).
	 * @return O filme na posição desejada, ou {@code null} se estiver vazia.
	 */
	public Filme getFilme(int posicao) {
		return filmes[posicao - 1];
	}

	/**
	 * Cadastra um novo filme na posição informada. Se já houver um filme na posição, ele será sobrescrito.
	 * 
	 * condicionais para evitar posição inválida, filme inválido e já adicionado.
	 * @param posicao A posição desejada (de 1 a 100).
	 * @param nome    O nome do filme.
	 * @param ano     O ano de lançamento do filme.
	 * @param local   O local onde o filme está disponível.
	 * @return Mensagem indicando o resultado do cadastro:
	 * 
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {

		if (posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		}
		if (nome.isEmpty() || local.isEmpty()) {
			return "FILME INVÁLIDO";
		}

		Filme dados = new Filme(nome, ano, local);
		for (int i = 0; i < 100; i++) {
			if (filmes[i] != null && filmes[i].equals(dados)) {
				return "FILME JÁ ADICIONADO";
			}
		}

		Filme fm = new Filme(nome, ano, local);
		this.filmes[posicao - 1] = fm;
		return "FILME ADICIONADO";
	}

	/**
	 * Retorna a lista de filmes em destaque (hotlist). Cada valor representa a posição de um filme na lista principal.
	 * 
	 * @return Um array com os índices (1 a 100) dos filmes destacados.
	 */
	public int[] getHotFilmes() {
		return this.hotFilmes.clone();
	}

	/**
	 * Cadastra um filme na hotlist.
	 * 
	 * Condicionais para evitar posições inválida e filmes já adicionados.
	 * 
	 * @param posicao A posição desejada na hotlist (de 1 a 10).
	 * @param filme   A posição do filme na lista principal (de 1 a 100).
	 * @return Mensagem indicando o resultado da operação:
	 *         
	 */
	public String cadastraHot(int posicao, int filme) {
		if (posicao < 1 || posicao > 10) {
			return "POSIÇÃO INVÁLIDA";
		}

		for (int i = 0; i < 10; i++) {
			if (hotFilmes[i] == filme) {
				return "FILME JÁ ADICIONADO À HOTLIST!";
			}
		}

		this.hotFilmes[posicao - 1] = filme;
		return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicao + " !";
	}

	/**
	 * Remove um filme da hotlist na posição informada.
	 * 
	 * Condicionais para evitar posições inválidas e vazias.
	 * @param posicao A posição na hotlist (de 1 a 10).
	 * @return Mensagem indicando o resultado da operação:
	 *         
	 */
	public String retiraHot(int posicao) {
		if (posicao < 1 || posicao > 10) {
			return "POSIÇÃO INVÁLIDA";
		}

		if (this.hotFilmes[posicao - 1] == 0) {
			return "POSIÇÃO VAZIA";
		}

		this.hotFilmes[posicao - 1] = 0;
		return "FILME REMOVIDO DA HOTLIST NA POSIÇÃO " + posicao;
	}

	/**
	 * Retorna os detalhes de um filme da lista.
	 * 
	 * @param posicaoFilme A posição do filme (de 1 a 100).
	 * @return Uma representação textual do filme. Se estiver na hotlist, é precedido de "🔥 ".
	 *         Se não houver filme na posição, retorna uma string vazia.
	 */
	public String detalhesFilme(int posicaoFilme) {
		if (posicaoFilme < 1 || posicaoFilme > 100) {
			return "POSIÇÃO INVÁLIDA";
		}

		Filme filme = getFilme(posicaoFilme);
		if (filme == null) {
			return ""; //A especifição pediu para retornar nada, porém acho que seria melhor uma mensagem "POSIÇÃO VAZIA"
		}

		String fm = "" + filme;
		for (int i = 0; i < 10; i++) {
			if (hotFilmes[i] == posicaoFilme) {
				fm = "🔥 " + filme;
				return fm;
			}
		}
		return fm;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

/**
 * Classe principal que controla a execução do sistema FilmNow.
 * Permite carregar filmes, interagir via menu para adicionar, mostrar,
 * detalhar filmes, e gerenciar uma lista "Hot" de filmes.
 */
public class MainFilmNow {

    /**
     * Método principal que inicializa o sistema, carrega os filmes do arquivo
     * e inicia o loop do menu interativo para o usuário.
     * 
     * @param args argumentos de linha de comando (não utilizados)
     */
    public static void main(String[] args) {
        FilmNow fn = new FilmNow();

        System.out.println("Carregando filmes ...");
        try {
            /*
             * Essa é a maneira de lidar com possíveis erros por falta do arquivo. 
             */
            carregaFilmes("filmes_inicial.csv", fn);
        } catch (FileNotFoundException e) {
            System.err.println("Arquivo não encontrado: " + e.getMessage());
        } catch (IOException e) {
            System.err.println("Erro lendo arquivo: " + e.getMessage());
        }

        Scanner scanner = new Scanner(System.in);
        String escolha = "";
        while (true) {
            escolha = menu(scanner);
            comando(escolha, fn, scanner);
        }
    }

    /**
     * Exibe o menu principal e lê a escolha do usuário.
     * 
     * @param scanner objeto Scanner para entrada do usuário
     * @return opção escolhida convertida para maiúscula
     */
    private static String menu(Scanner scanner) {
        System.out.println(
                "\n---\nMENU\n" + 
                "(A)Adicionar filme\n" + 
                "(M)Mostrar todos\n" + 
                "(D)Detalhar filme\n" + 
                "(E)Exibir HotList\n" +
                "(H)Atribuir Hot\n" +
                "(R)Remover Hot\n" +
                "(S)air\n" + 
                "\n" + 
                "Opção> ");
        return scanner.next().toUpperCase();
    }

    /**
     * Executa a ação correspondente à opção escolhida pelo usuário no menu.
     * 
     * @param opcao string com a opção escolhida
     * @param fn instância do sistema FilmNow para manipulação dos filmes
     * @param scanner objeto Scanner para entrada do usuário
     */
    private static void comando(String opcao, FilmNow fn, Scanner scanner) {
        switch (opcao) {
            case "A":
                adicionaFilme(fn, scanner);
                break;
            case "M":
                mostrarFilmes(fn);
                break;
            case "D":
                detalharFilme(fn, scanner);
                break;
            case "E":
                exibeHotFilmes(fn);
                break;
            case "H":
                adicionaHotFilme(fn, scanner);
                break;
            case "R":
                removeHotFilme(fn, scanner);
                break;
            case "S":
                sai();
                break;
            default:
                System.out.println("Opção inválida!");
        }
    }

    /**
     * Imprime a lista de todos os filmes cadastrados no sistema.
     * 
     * @param fn instância do sistema FilmNow
     */
    private static void mostrarFilmes(FilmNow fn) {
        Filme[] filmes = fn.getFilmes();
        for (int i = 0; i < filmes.length; i++) {
            if (filmes[i] != null) {
                System.out.println(formataFilme(i, filmes[i]));
            }
        }
    }

    /**
     * Exibe detalhes de um filme baseado na posição informada pelo usuário.
     * 
     * @param fn instância do sistema FilmNow
     * @param scanner objeto Scanner para ler a posição do filme
     */
    private static void detalharFilme(FilmNow fn, Scanner scanner) {
        System.out.print("\nQual filme> ");
        int posicao = scanner.nextInt();
        
        String detalhado = fn.detalhesFilme(posicao);
        
        System.out.println(detalhado);
    }

    /**
     * Formata a exibição básica do filme com sua posição e nome.
     * 
     * @param posicao índice do filme na lista
     * @param filme objeto Filme a ser formatado
     * @return string formatada com posição e nome do filme
     */
    private static String formataFilme(int posicao, Filme filme) {
        int pos = posicao + 1;
        return pos + " - " + filme.getNome();
    }

    /**
     * Solicita ao usuário os dados para adicionar um filme em uma posição específica.
     * 
     * @param fn instância do sistema FilmNow
     * @param scanner objeto Scanner para entrada dos dados
     */
    private static void adicionaFilme(FilmNow fn, Scanner scanner) {
        System.out.print("\nPosição no sistema> ");
        int posicao = scanner.nextInt();
        scanner.nextLine();
        
        System.out.print("\nNome> ");
        String nome = scanner.nextLine();
        
        System.out.print("\nAno> ");
        String ano = scanner.nextLine();
        
        System.out.print("\nLocal> ");
        String local = scanner.nextLine();
        
        System.out.println(fn.cadastraFilme(posicao, nome, ano, local));
    }

    /**
     * Finaliza a execução do programa com mensagem de despedida.
     */
    private static void sai() {
        System.out.println("\nVlw flw o/");
        System.exit(0);
    }

    /**
     * Carrega filmes de um arquivo CSV para o sistema FilmNow.
     * 
     * @param arquivoFilmes nome do arquivo CSV contendo os filmes
     * @param fn instância do sistema FilmNow
     * @throws FileNotFoundException caso o arquivo não seja encontrado
     * @throws IOException caso ocorra erro na leitura do arquivo
     */
    private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
        LeitorFilmNow leitor = new LeitorFilmNow();
        
        int carregados =  leitor.carregaContatos(arquivoFilmes, fn);
        System.out.println("Carregamos " + carregados + " registros.");
    }

    /**
     * Exibe a lista de filmes "Hot" do sistema.
     * 
     * @param fn instância do sistema FilmNow
     */
    private static void exibeHotFilmes(FilmNow fn) {
        Filme[] filmes = fn.getFilmes();
        int[] hotFilmes = fn.getHotFilmes();
        
        for (int i = 0; i < hotFilmes.length; i++) {
            int posicao = i + 1;
            
            if (hotFilmes[i] != 0) {
                Filme filme = filmes[hotFilmes[i] - 1];
                System.out.println(posicao + " - " + filme.getNome() + ", " + filme.getAno());
            }
        }
    }

    /**
     * Solicita ao usuário a posição e o filme para atribuir a lista "Hot".
     * 
     * @param fn instância do sistema FilmNow
     * @param scanner objeto Scanner para entrada dos dados
     */
    private static void adicionaHotFilme(FilmNow fn, Scanner scanner) {
        System.out.println("Filme>");
        int filme = scanner.nextInt();
        
        System.out.println("Posicao>");
        int posicao = scanner.nextInt();
        
        System.out.println(fn.cadastraHot(posicao, filme));
    }

    /**
     * Solicita ao usuário a posição para remover da lista "Hot".
     * 
     * @param fn instância do sistema FilmNow
     * @param scanner objeto Scanner para entrada dos dados
     */
    private static void removeHotFilme(FilmNow fn, Scanner scanner) {
        System.out.println("Posicao>");
        int posicao = scanner.nextInt();
        
        fn.retiraHot(posicao);
    }
