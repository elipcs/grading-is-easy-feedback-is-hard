package filmnow;

public class Filme {
    private String nome;
    private int posicao;
    private String ano;  
    private String local;

    public Filme(int posicao, String nome, String ano, String local){
        this.nome = nome;
        this.ano = ano;
        this.posicao = posicao;
        this.local = local;
    }

    public String getNome(){
        return nome;
    }

    public void setNome(String nome) {
        this.nome = nome;
    }

    public String getAno() {
        return ano;
    }

    public void setAno(String ano) {
        this.ano = ano;
    }

    public int getPosicao() {
        return posicao;
    }

    public void setPosicao(int posicao) {
        this.posicao = posicao;
    }

    public String getLocal() {
        return local;
    }

    public void setLocal(String local) {
        this.local = local;
    }

    public String nomeAno() {
        return nome + ", " + ano;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || getClass() != obj.getClass()) return false;
        Filme filme = (Filme) obj;
        return ano.equals(filme.ano) && nome.equals(filme.nome);
    }

    @Override
    public String toString() {
        return posicao + " - " + nome ;
    }
}

package filmnow;

public class FilmNow {

    private static final int TAMANHO = 100;
    private static final int TAMANHO_HOTLIST = 10;

    private Filme[] filmes; // uma representação simplória da lista de filmes
    private Filme[] hotList; // lista de filmes "hot"

    public FilmNow() {
        this.filmes = new Filme[TAMANHO];
        this.hotList = new Filme[TAMANHO_HOTLIST]; 
    }

    public Filme[] getFilmes() {
        return this.filmes.clone();
    }

    public Filme getFilme(int posicao) {
        if (posicao < 0 || posicao >= TAMANHO) {
            return null;
        }
        return filmes[posicao];
    }

    public String cadastraFilme(int posicao, String nome, String ano, String local) {
        if (posicao < 1 || posicao > TAMANHO) {
            return "POSIÇÃO INVÁLIDA";
        }

        if (nome.equals("") || local.equals("") || (nome.equals("") && local.equals(""))) {
            return "FILME INVÁLIDO";
        }

        Filme novoFilme = new Filme(posicao, nome, ano, local);
        for (Filme filme : filmes) {
            if (filme != null && filme.equals(novoFilme)) {
                return "FILME JÁ ADICIONADO";
            }
        }

        filmes[posicao - 1] = novoFilme;
        return "FILME ADICIONADO";
    }

    public String detalharFilme(int posicao) {
        if (posicao < 1 || posicao > TAMANHO) {
            return "POSIÇÃO INVÁLIDA";
        }

        Filme filme = filmes[posicao - 1];
        if (filme == null) {
            return "";
        }

        boolean isHot = isHot(filme);
        String nome = filme.getNome();
        if (isHot) {
            nome = "🔥 " + nome;
        }

        String ano = filme.getAno();
        if (ano != null && !ano.isEmpty()) {
            return nome + ", " + ano + "\n" + filme.getLocal();
        } else {
            return nome + "\n" + filme.getLocal();
        }
    }
    public String cadastraHot(int posicaoHot, int posicaoFilme) {
        if (posicaoFilme < 1 || posicaoFilme > TAMANHO || posicaoHot < 1 || posicaoHot > TAMANHO_HOTLIST) {
        	return "POSIÇÃO INVÁLIDA";
            
        }

        Filme filme = filmes[posicaoFilme - 1];
        if (filme == null) {
        	return "NÃO HÁ FILME NESSA POSIÇÃO";
            
        }

        for (int i = 0; i < TAMANHO_HOTLIST; i++) {
            if (hotList[i] != null && hotList[i].equals(filme)) {
            	return "FILME JÁ ESTÁ NA HOTLIST";
            }
        }

        hotList[posicaoHot - 1] = filme;
		return "FILME ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!";
    }

    public Filme[] getHotList() {
        return this.hotList.clone();
    }
    
    public boolean isHot(Filme filme) {
        for (Filme hot : hotList) {
            if (hot != null && hot.equals(filme)) {
                return true;
            }
        }
        return false;
    }

    public String removeHot(int posicaoHot) {
        if (posicaoHot < 1 || posicaoHot > TAMANHO_HOTLIST) {
        	return "POSIÇÃO INVÁLIDA";
            
        }

        Filme filme = hotList[posicaoHot - 1];
        if (filme == null) {
        	return "NÃO HÁ FILME NESSA POSIÇÃO";
            
        }

        hotList[posicaoHot - 1] = null;

        String nome = filme.getNome().replace("🔥 ", "");
        filme.setNome(nome);

		return "FILME REMOVIDO DA HOTLIST";
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

public class MainFilmNow {

    public static void main(String[] args) {
        FilmNow fn = new FilmNow();

        System.out.println("Carregando filmes ...");
        try {
            carregaFilmes("filmes_inicial.csv", fn);
        } catch (FileNotFoundException e) {
            System.err.println("Arquivo não encontrado: " + e.getMessage());
        } catch (IOException e) {
            System.err.println("Erro lendo arquivo: " + e.getMessage());
        }

        Scanner scanner = new Scanner(System.in);
        String escolha = "";
        while (true) {
            escolha = menu(scanner);
            comando(escolha, fn, scanner);
        }
    }

    private static String menu(Scanner scanner) {
        System.out.println(
                "\n---\nMENU\n" + 
                        "(A)Adicionar filme\n" + 
                        "(M)Mostrar todos\n" + 
                        "(D)Detalhar filme\n" + 
                        "(E)Exibir HotList\n" +
                        "(H)Atribuir Hot\n" +
                        "(R)Remover Hot\n" +
                        "(S)air\n" + 
                        "\n" + 
                        "Opção> ");
        return scanner.next().toUpperCase();
    }

    private static void comando(String opcao, FilmNow fn, Scanner scanner) {
        switch (opcao) {
        case "A":
            adicionaFilme(fn, scanner);
            break;
        case "M":
            mostrarFilmes(fn);
            break;
        case "D":
            detalharFilme(fn, scanner);
            break;
        case "E":
            exibeHot(fn);
            break;
        case "H":
            atribuiHot(fn, scanner);
            break;
        case "R":
            removeHot(fn, scanner);
            break;
        case "S":
            sai();
            break;
        default:
            System.out.println("OPÇÃO INVÁLIDA!");
        }
    }

    private static void mostrarFilmes(FilmNow fn) {
        Filme[] filmes = fn.getFilmes();
        for (int i = 0; i < filmes.length; i++) {
            if (filmes[i] != null) {
                System.out.println(filmes[i].toString());
            }
        }
    }

    private static void detalharFilme(FilmNow fn, Scanner scanner) {
        System.out.print("\nFilme> ");
        int posicao = scanner.nextInt();
        String detalhes = fn.detalharFilme(posicao);
        System.out.println(detalhes);
    }

    private static void adicionaFilme(FilmNow fn, Scanner scanner) {
        System.out.print("\nPosição no sistema> ");
        int posicao = scanner.nextInt();
        scanner.nextLine(); 
        System.out.print("\nNome> ");
        String nome = scanner.nextLine();
        System.out.print("\nAno> ");
        String ano = scanner.nextLine();
        System.out.print("\nLocal> ");
        String local = scanner.nextLine();
        String resultado = fn.cadastraFilme(posicao, nome, ano, local);
        System.out.println(resultado);
    }

    private static void atribuiHot(FilmNow fn, Scanner scanner) {
        System.out.print("\nFilme> ");
        int filme = scanner.nextInt();
        System.out.print("\nPosição na Hotlist> ");
        int posicao = scanner.nextInt();
        String resultado = fn.cadastraHot(posicao, filme);
        System.out.println(resultado);
    }

    private static void exibeHot(FilmNow fn) {
        Filme[] hotList = fn.getHotList(); 
        for (int i = 0; i < hotList.length; i++) {
            if (hotList[i] != null) {
                System.out.println((i + 1) + " - " + hotList[i].getNome() + ", " + hotList[i].getAno());
            }
        }
    }

    private static void removeHot(FilmNow fn, Scanner scanner) {
        System.out.print("\nPosição da HotList> ");
        int posicao = scanner.nextInt();
        fn.removeHot(posicao);
    }

    private static void sai() {
        System.exit(0);
    }

    private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
        LeitorFilmNow leitor = new LeitorFilmNow();
        
        int carregados =  leitor.carregaContatos(arquivoFilmes, fn);
        System.out.println("Carregamos " + carregados + " registros.");
    }
}
