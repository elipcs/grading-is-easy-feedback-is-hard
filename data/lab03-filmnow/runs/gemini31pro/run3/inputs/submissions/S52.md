package filmnow;

import java.util.Objects;

/**
 * A classe Filme representa um filme com nome, 
 * ano de lançamento, local de exibição e status de destaque (hot).
 */
public class Filme {
    private String nome;
    private String ano;
    private String local;
    private boolean hot;

    /**
     * Construtor para criar um novo objeto Filme.
     *
     * @param nome  o nome do filme
     * @param ano   o ano de lançamento do filme
     * @param local o local de exibição do filme
     */
    public Filme(String nome, String ano, String local) {
        this.nome = nome;
        this.ano = ano;
        this.local = local;
        this.hot = false;
    }

    /**
     * Define o status de destaque (hot) do filme.
     *
     * @param status true se o filme é destaque, false caso contrário
     */
    public void setHot(boolean status) {
        this.hot = status;
    }

    public String getNome() {
        return this.nome;
    }

    public String getAno() {
        return this.ano;
    }

    /**
     * Retorna uma representação em string do filme.
     * Se o filme estiver marcado como destaque (hot), 
     * o símbolo de fogo (🔥) será adicionado antes do nome.
     *
     * @return uma string representando o filme
     */
    public String toString() {
        String out = this.hot ? "🔥 " : "";
        out += this.nome;
        if (!this.ano.isEmpty()) out += ", " + this.ano + "\n";
        else out += "\n";
        return out += this.local;
    }

    @Override
    public int hashCode() {
        return Objects.hash(ano, nome);
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (obj == null)
            return false;
        if (getClass() != obj.getClass())
            return false;
        Filme other = (Filme) obj;
        return ano.equals(other.ano) && Objects.equals(nome, other.nome);
    }
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos.
 * Podem existir até 100 filmes e 10 filmes na HotList.
 */
public class FilmNow {

    private static final int TAMANHO = 100;
    private static final int TAMANHOHOTLIST = 10;

    private Filme[] filmes;
    private Filme[] hotList;

    /**
     * Cria um novo objeto FilmNow com capacidade para até 100 filmes e 10 filmes na HotList.
     */
    public FilmNow() {
        this.filmes = new Filme[TAMANHO];
        this.hotList = new Filme[TAMANHOHOTLIST];
    }

    public Filme[] getFilmes() {
        System.out.println(this.filmes[1].toString());
        return this.filmes.clone();
    }

    public Filme[] getHotList() {
        return this.hotList.clone();
    }

    /**
     * Acessa os dados de um filme específico.
     * 
     * @param posicao Posição do filme no sistema (de 1 a 100).
     * @return Uma string com os dados do filme ou "OPÇÃO INVÁLIDA!" se a posição for inválida ou não houver filme na posição.
     */
    public String getFilme(int posicao) {
        if (!posicaoValida(posicao, TAMANHO) || this.filmes[posicao - 1] == null) {
            return "OPÇÃO INVÁLIDA!";
        }
        return "Dados do filme:\n" + filmes[posicao - 1].toString();
    }

    /**
     * Verifica se uma posição é válida dentro do intervalo permitido.
     * 
     * @param posicao Posição a ser verificada.
     * @param range   Limite superior do intervalo.
     * @return true se a posição for válida, false caso contrário.
     */
    private boolean posicaoValida(int posicao, int range) {
        return posicao > 0 && posicao <= range;
    }

    /**
     * Verifica se um filme já está na lista de filmes.
     * 
     * @param filme Filme a ser verificado.
     * @return true se o filme já está na lista, false caso contrário.
     */
    private boolean contemFilme(Filme filme) {
        for (Filme filmeArr : this.filmes) {
            if (filmeArr != null && filmeArr.equals(filme)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Adiciona um filme em uma posição específica. Se já existir filme na posição, sobrescreve o anterior.
     * 
     * @param posicao Posição do filme (de 1 a 100).
     * @param nome    Nome do filme.
     * @param ano     Ano de lançamento do filme.
     * @param local   Local onde o filme pode ser assistido.
     * @return Uma mensagem indicando o resultado da operação.
     */
    public String cadastraFilme(int posicao, String nome, String ano, String local) {
        if (nome.isBlank() || local.isBlank()) {
            return "FILME INVÁLIDO";
        }

        Filme filme = new Filme(nome, ano, local);
        if (contemFilme(filme)) {
            return "FILME JA ADICIONADO";
        }

        if (!posicaoValida(posicao, TAMANHO)) {
            return "POSIÇÃO INVÁLIDA";
        }
        this.filmes[posicao - 1] = filme;
        return "FILME ADICIONADO";
    }

    /**
     * Adiciona um filme da lista principal à HotList em uma posição específica.
     * 
     * @param posicaoFilme Posição do filme na lista principal (de 1 a 100).
     * @param posicaoHot   Posição na HotList (de 1 a 10).
     * @return Uma mensagem indicando o resultado da operação.
     */
    public String cadastraHot(int posicaoFilme, int posicaoHot) {
        if (!posicaoValida(posicaoHot, TAMANHOHOTLIST)) {
            return "POSIÇÃO INVÁLIDA";
        }
        if (!posicaoValida(posicaoFilme, TAMANHO) || this.filmes[posicaoFilme - 1] == null) {
            return "FILME INVÁLIDO";
        }
        for (Filme filmeHot : hotList) {
            if (filmeHot != null && filmeHot.equals(this.filmes[posicaoFilme - 1])) {
                return "FILME JÁ EXISTENTE EM HOTLIST";
            }
        }

        removeHot(posicaoHot);
        this.filmes[posicaoFilme - 1].setHot(true);
        this.hotList[posicaoHot - 1] = this.filmes[posicaoFilme - 1];

        return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!";
    }

    /**
     * Remove um filme da HotList em uma posição específica.
     * 
     * @param posicao Posição na HotList (de 1 a 10).
     */
    public void removeHot(int posicao) {
        if (posicaoValida(posicao, TAMANHOHOTLIST) && this.hotList[posicao - 1] != null) {
            this.hotList[posicao - 1].setHot(false);
            this.hotList[posicao - 1] = null;
        }
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

		while (true) {
			escolha = menu(scanner);

		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */
	
	private static String menu(Scanner scanner) {
		System.out.println(
				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" + 
						"(E)Exibir HotList\n" + 
						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)Sair\n" + 
						"\n" + 
						"Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opcao   Opção digitada.
	 * @param fn  O sistema FilmNow que estamos manipulando.
	 * @param scanner Objeto scanner para o caso do comando precisar de mais input.

	private static void comando(String opcao, FilmNow fn, Scanner scanner) {
		switch (opcao) {
		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibirHotList(fn);
			break;
		case "H":
			atribuirHot(fn, scanner);
			break;
		case "R":
			removerHot(fn, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("Opção inválida!");
		}
	}

	/**
	 * Imprime lista de filmes.

	 * @param fn O sistema FilmNow a ser manipulado.

	 * Formata um filme para impressão.

	 * @return A String formatada.
