public class Filme {
	private String nome;

	 * Constroi o filme de acordo com as informações recebidas.

	 * @param ano o ano de lançamento do filme.
	 * @param local o local onde pode assisti-lo.
	 */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.hot = false;
	
	}
	/**
	 * Recebe o ano de lançamento do filme do scanner.
	 * @return o ano de lançamento do filme.

		return ano;
	}

	 * Modifica o valor do parametro hot.
	 * @param hot
	 */
	public void setHot(boolean hot) {
		this.hot = hot;
	}
	/**
	 * Cria endereços para comparação de dois filmes para confirmar se ele ja foi adicionado a lista;
	 */
	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}

	 * Compara dois filmes para confirmar se ele ja foi adicionado a lista

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 * @author [REDACTED]
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	private static final int TAMANHO_HOT = 10;
	private Filme[] filmes; //uma representacao simploria da lista de filmes
	private Filme[] hotFilmes; //uma representacao simploria da lista de filmes da hotList

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotFilmes = new Filme[TAMANHO_HOT];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public String[] getFilmes() {
		return formataToString(filmes);
	}
	/**
	 * Acessa a hotList de filmes mantida.
	 * @return o array de filmes hot
	 */
	public String[] getFilmesHot() {	
		return formataToString(hotFilmes);
	}
	/**
	 * Formata o as informações sobre o filme e retorna uma String.
	 * @param array um Array contendo as Strings formatadas dos filmes. 
	 * @return
	 */
	private String[] formataToString(Filme[] array) {
		String[] saida = new String[array.length];
		for(int i = 0; i < array.length; i++) {
			if (array[i] != null) {
				saida[i] = array[i].toString();
			}
		}
		return saida;
	}

	/**
	 * Verifica se o filme adicionado é igual a quaquer outro da lista.
	 * @param filme o filme a ser verificado.
	 * @return true ou false a depender da verificação.
	 */
	public boolean temFilme(Filme filme) {
		for(int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null && filmes[i].equals(filme)) {
				return true;
			}
		} 
		return false;
	}
	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 * @return
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if(posicao >= 1 && posicao <= TAMANHO) {
			Filme filme = new Filme(nome, ano, local);
			if (nome == null || nome.equals("") || local == null || local.equals("")) {
				return "FILME INVÁLIDO";
			}
			if (!temFilme(filme)) {
				filmes[posicao -1] = filme;
				return "FILME ADICIONADO";
			}
			return "FILME JA ADICIONADO";
		}
		return "POSIÇÃO INVÁLIDA";
	}
	/**
	 * Verifica se o filme adicionado é igual a quaquer outro da hotlist.
	 * @param filme o filme a ser verificado.
	 * @return true ou false a depender da verificação.
	 */
	
	public boolean temFilmeHot (Filme filme) {
		for(int i = 0; i < hotFilmes.length; i++) {
			if (hotFilmes[i] != null && hotFilmes[i].equals(filme)) {
				return true;
			}
		}
		return false;
	}
	/**
	 * Adiciona um filme a hotList em uma posição. Se já existir filme na posição, sobrescreve o anterior.
	 * @param posicaoHot Posição do filme na hotList.
	 * @param posicao Posição do filme na lista de filmes.
	 * @return Mensagem de status.
	 */
	public String adicionaHotList(int posicao, int posicaoHotList) {
		if ((posicaoHotList >= 1 && posicaoHotList <= TAMANHO_HOT) || (posicao >= 1 && posicao >= TAMANHO)) {
			if(hotFilmes[posicaoHotList-1] != null) {
				removeHotList(posicaoHotList);
			}
			if (!temFilmeHot(filmes[posicao-1])) {
				filmes[posicao-1].setHot(true);
				hotFilmes[posicaoHotList-1] = filmes[posicao-1];
				return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHotList + "!";
			}
			return "FILME JÁ ADICIONADO";		
		}
		return "POSIÇÃO INVÁLIDA";	
	}
	/**
	 * Remove um filme da hotList.
	 * @param posicao Posição do filme na hotList (baseado em 1).
	 */
	public void removeHotList(int posicao) {
		if (hotFilmes[posicao-1] != null) {
			hotFilmes[posicao-1].setHot(false);
			hotFilmes[posicao-1] = null;
		}
	}
	/**
	 * Detalha os filmes acrescentando seu ano de lançamento e o local onde pode-se assisti-lo.
	 * @param posicao
	 * @return
	 */
	public String detalharFilme(int posicao) {
		if(posicao < 1 || posicao > TAMANHO || filmes[posicao - 1] == null) {
			return "POSIÇÃO INVÁLIDA";
		}
		return filmes[posicao - 1].detalha();
	}
	/**
	 Exibe os todos os filmes da lista no formato "posição - nome, ano
	 *                                             local"
	 * @param fn O sistema FilmNow que estamos manipulando.
	 * @return Scanner para pedir informações do contato.
	 */
	public String exibirFilmes(FilmNow fn) {
		String[] filmes = fn.getFilmes();
		String saida = "";
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				saida += (i + 1) + " - " + filmes[i].toString() + "\n";
			}
		}
		return saida;
	}
	/**
	 * Exibe os todos os filmes da lista no formato "posição - nome, ano
	 *                                               local"
	 * @param fn O sistema FilmNow que estamos manipulando.
	 * @return a saida formatada
	 */
	public String exibirHotFilmes(FilmNow fn) {
		String[] hot = fn.getFilmesHot();
		String saida = "";
		for (int i = 0; i < hot.length; i++) {
			if (hot[i] != null) {
				saida += (i+1) + " - " + hot[i].toString() + "\n";
			}
		}
		return saida;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
