package filmnow;

import java.util.Objects;

/**
 * A classe Filme representa um filme com informações básicas, como nome,
 * local onde pode ser assistido, ano de lançamento e posição em uma lista.
 * Esta classe inclui métodos para acessar esses dados e sobrescreve os 
 * métodos equals() e hashCode() para garantir a correta comparação e uso
 * em coleções baseadas em hash.
 */

public class Filme {
	
	/**
     * O nome do filme.
     */
	private String nomeFilme;
	
	 /**
     * O local (plataforma) onde o filme pode ser assistido.
     */
	private String localFilme;
	
	/**
     * O ano de lançamento do filme.
     */
	private String anoFilme;
	
	/**
     * Construtor da classe Filme.
     *
     * @param nome  O nome do filme.
     * @param ano   O ano de lançamento do filme.
     * @param local O local (plataforma) onde o filme pode ser assistido.
     */
	public Filme(String nome, String ano, String local) {
		this.nomeFilme = nome;
		this.localFilme = local;
		this.anoFilme = ano;
		
	}
	

	public String getNome() {
		return nomeFilme;
		
	}
	
	public String getAno() {
		return anoFilme;
	}
	
	public String getLocal() {
		return localFilme;
	}
	
	public String toString() {
		return nomeFilme;
	}
	
	@Override
	public boolean equals(Object obj) {
		if (this == obj) {
			return true;
		}
		if (obj == null || getClass() != obj.getClass()) {
			return false;
		}
		Filme filme = (Filme) obj;
		return anoFilme == filme.anoFilme && nomeFilme.equals(filme.nomeFilme);
		
	}
	
	@Override
	public int hashCode() {
		return Objects.hash(nomeFilme , anoFilme);
	}

}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100; // Capacidade máxima da lista de filmes
	private static final int tamanhoHOT = 10; // Capacidade máxima da HotList (lista de filmes "quentes")
	
	private Filme[] filmes; // Array que armazena os filmes cadastrados
	private int quantidadeFilmes; // Número de filmes cadastrados
	private Filme[] listaHot; // Array que armazena os filmes na HotList
	private int quantidadeFilmesHot; // Número de filmes na HotList
	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.quantidadeFilmes = 0;
		this.listaHot = new Filme[tamanhoHOT];
		this.quantidadeFilmesHot = 0;
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		if (posicao >= 0 && posicao <= this.filmes.length && this.filmes[posicao] != null) {
			return this.filmes[posicao];
		}
		return null;
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public void cadastraFilme(int posicao, String nome, String ano, String local) {
			if (posicao <= 100 && posicao < this.filmes.length) { 
				this.filmes[posicao] = new Filme(nome , ano , local); 
				if (this.quantidadeFilmes < this.filmes.length) {
					this.quantidadeFilmes += 1;
			}
		}
	}
	
	/**
     * Retorna uma lista de filmes existentes (não nulos).
     * 
     * @param filmes Array de filmes.
     * @return Array de filmes existentes.
     */
	public Filme[] filmesExistentes(Filme[] filmes) {
		
		Filme[] filmesExistentes = new Filme[this.quantidadeFilmes];
		
		int indice  = 0;
		
		for (int i = 0 ; i < filmes.length ; i ++) {
			
			if (filmes[i] != null) {
				
				filmesExistentes[indice] = filmes[i];
				indice++;
			} 
		}
		return filmesExistentes;
		
		
		
	}
	
	/**
     * Verifica se um filme com determinado nome e ano já existe na lista de filmes.
     * 
     * @param nomeFilme Nome do filme.
     * @param anoFilme Ano de lançamento do filme.
     * @return true se o filme existe, false caso contrário.
     */
	public boolean filmeExiste(String nomeFilme , String anoFilme) {
		
		for (Filme filme : filmesExistentes(this.filmes)) {
			
			if (filme.getNome().equals(nomeFilme) && filme.getAno() == anoFilme ) {
				
				return true;
			}
		}
		return false;
	}
	
	
	/**
	 * Adiciona um filme à HotList em uma posição específica, se a posição for válida e o filme não estiver duplicado.
	 * 
	 * @param posicaoFilme Posição do filme na lista geral (baseada em 1).
	 * @param posicaoHotList Posição na HotList onde o filme será adicionado (baseada em 1).
	 */
	public void adicionaFilmeHot(int posicaoFilme, int posicaoHotList) {
	    // Verifica se a posição do filme e a posição na HotList são válidas.
	    if (posicaoFilme < 1 || posicaoFilme > this.quantidadeFilmes || posicaoHotList < 1 || posicaoHotList > tamanhoHOT) {
	        System.out.println("Posição inválida");
	        return;
	    }
	    
	    // Obtém o filme na posição especificada.
	    Filme filmeParaAdicionar = filmesExistentes(this.filmes)[posicaoFilme - 1];
	    
	    // Verifica se o filme já está na HotList.
	    for (Filme filme : this.listaHot) {
	        if (filme != null && filme.equals(filmeParaAdicionar)) {
	            System.out.println("ESSE FILME JÁ ESTÁ NA LISTA");
	            return;
	        }
	    }
	    
	    // Adiciona o filme à HotList na posição especificada.
	    this.listaHot[posicaoHotList - 1] = filmeParaAdicionar;
	    this.quantidadeFilmesHot++;
	    System.out.println("ADICIONADO A HOT LIST NA POSIÇÃO " + posicaoHotList);
	}
	
	/**
     * Exibe o conteúdo da HotList.
     * 
     * @return Uma string com os detalhes dos filmes na HotList.
     */
	public String exibirConteudoHot() {
	    StringBuilder conteudo = new StringBuilder();
	    
	    for (int i = 0; i < this.listaHot.length; i++) {
	        if (this.listaHot[i] != null) {
	            conteudo.append("Posição ").append(i + 1).append(": ").append(this.listaHot[i].toString()).append(", " + this.listaHot[i].getAno()).append("\n");
	        } 
	    }
	    
	    return conteudo.toString();
	}
	
	/**
	 * Remove um filme da HotList na posição especificada.
	 * 
	 * @param posicaoHotList A posição do filme na HotList a ser removido.
	 *                       A posição deve ser entre 1 e o tamanho da HotList.
	 *                       Exibe uma mensagem se a posição for inválida ou se não houver filme na posição.
	 */
	public void removeHot(int posicaoHotList) {
		
		if (posicaoHotList > this.listaHot.length || posicaoHotList < 1) {
			System.out.println("POSIÇÃO INVÁLIDA");
			
		}
		
		if (this.listaHot[posicaoHotList - 1] != null) {
			this.listaHot[posicaoHotList - 1] = null;
			this.quantidadeFilmesHot --;
			System.out.println("FILME REMOVIDO DA POSIÇÃO " + posicaoHotList);
		}else {
			System.out.println("NÃO EXISTE NENHUM FILME NESSA POSIÇÃO");
			
		}
			
	}

	
	/**
	 * Retorna a HotList com os filmes atuais.
	 * 
	 * @return Um array de filmes na HotList, sem elementos nulos.
	 */
	public Filme[] getHotList() {
		
		return filmesExistentes(this.listaHot);
	}
	
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

/**
 * Interface com menus texto para manipular o sistema FilmNow.
 * 
 * @author [REDACTED]
 *
 */
public class MainFilmNow {

	public static void main(String[] args) {
		FilmNow fn = new FilmNow();

		System.out.println("Carregando filmes ...");
		try {
			/*
			 * Essa é a maneira de lidar com possíveis erros por falta do arquivo. 
			 */
			carregaFilmes("filmes_inicial.csv", fn);
		} catch (FileNotFoundException e) {
			System.err.println("Arquivo não encontrado: " + e.getMessage());
		} catch (IOException e) {
			System.err.println("Erro lendo arquivo: " + e.getMessage());
		}

		Scanner scanner = new Scanner(System.in);
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */
	
	private static String menu(Scanner scanner) {
		System.out.println(
				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" +
						"(E)Exibir Hotlist\n" +
						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)Sair\n" + 
						"\n" + 
						"Opção> ");
		return scanner.nextLine().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opcao   Opção digitada.
	 * @param fn  O sistema FilmNow que estamos manipulando.
	 * @param scanner Objeto scanner para o caso do comando precisar de mais input.

	private static void comando(String opcao, FilmNow fn, Scanner scanner) {

		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibirHotList(fn);
			break;
		case "H":
			adicionarHot(fn , scanner);
			break;
		case "R":
			removerHot(fn , scanner);
			break;
		case "S":
			sai();
			break;	
		default:
			System.out.println("Opção inválida!");
		}
	}

	/**
     * Remove um filme da HotList na posição especificada.
     * 
     * @param fn O sistema FilmNow que está sendo manipulado.
     * @param scanner Scanner para capturar a posição a ser removida.

	 * @param fn O sistema FilmNow a ser manipulado.

	private static void mostrarFilmes(FilmNow fn) {

		Filme[] filmes = fn.getFilmes();
