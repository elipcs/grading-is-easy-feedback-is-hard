package filmnow;

import java.util.Objects;

/**
 * Representa um filme com suas informações básicas como nome,
 * ano de lançamento e plataforma onde está disponível.
 * * @author [REDACTED]
 *
 */
public class Filme {

	private String nome;
	private String lancamento;
	private String platafoma;
	private String hot;
	
	
	/**
	 * Constrói um objeto Filme.
	 * * @param nome O nome do filme.
	 * @param lancamento O ano de lançamento do filme.
	 * @param platafoma A plataforma ou local onde o filme pode ser assistido.
	 */
	public Filme(String nome, String lancamento, String platafoma) {
		this.nome = nome;
		this.lancamento = lancamento;
		this.platafoma = platafoma;
	}

	public String getNome() {
		return nome;
	}

	public void setNome(String nome) {
		this.nome = nome;
	}

	public String getLancamento() {
		return lancamento;
	}

	public void setLancamento(String lancamento) {
		this.lancamento = lancamento;
	}

	public String getHot() {
		return hot;
	}

	public void setHot(String hot) {
		this.hot = hot;
	}

	public String getPlatafoma() {
		return platafoma;
	}

	public void setPlatafoma(String platafoma) {
		this.platafoma = platafoma;
	}
	
	/**
	 * Gera um código hash para o filme, baseado no nome e no ano de lançamento.
	 * * @return O código hash.
	 */
	public int hashCode() {
		return Objects.hash(lancamento, nome);
	}

	
	/**
	 * Compara este filme com outro objeto para verificar a igualdade.
	 * Dois filmes são considerados iguais se tiverem o mesmo nome e ano de lançamento.
	 * * @param obj O objeto a ser comparado.
	 * @return true se os objetos são iguais, false caso contrário.
	 */
	public boolean equals(Object obj) {
	    if (this == obj)
	        return true;
	    if (obj == null || getClass() != obj.getClass())
	        return false;
	    Filme other = (Filme) obj;
	    return Objects.equals(nome, other.nome) && Objects.equals(lancamento, other.lancamento);
	}

	/**
	 * Retorna uma representação em String do filme, que é o seu nome.
	 * * @return O nome do filme.
	 */
	public String toString() {
		return nome;
	}

	/**
	 * Retorna uma representação detalhada do filme em String,
	 * incluindo nome, ano e plataforma.
	 * * @return A string detalhada do filme.
	 */
	public String toStringDetalhado() {
		return nome + ", " + lancamento + "\n" + platafoma;
	}
	
	/**
	 * Retorna uma representação em String do filme para ser usada na Hotlist,
	 * que é o seu nome.
	 * * @return O nome do filme.
	 */
	public String toStringHotList() {
		return nome;
	}	
	
}

package filmnow;

import java.util.Scanner;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes
 * e uma lista de 10 filmes "Hot".
 * * @author [REDACTED]
 *
 */
public class FilmNow {

	private static final int TAMANHO = 100;
	private static final int TAMANHOHOT = 10;

	private Filme[] filmes; 
	private Filme[] hots;

	/**
	 * Cria o FilmNow, inicializando as listas de filmes e de "hots".
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hots = new Filme[TAMANHOHOT];
	}

	public Filme[] getFilmes() {
		return this.filmes.clone();
	}
	
	public Filme getFilme(int posicao) {
		return filmes[posicao];
	}
	

	public void setFilmes(Filme[] filmes) {
		this.filmes = filmes;
	}

	/**
	 * Cadastra um filme em uma posição especificada (de 1 a 100) no sistema.
	 * O filme não pode ter nome vazio e não pode ser duplicado (mesmo nome e ano).
	 * * @param posicao Posição no sistema onde o filme será cadastrado (1 a 100).
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param plataforma Plataforma/streaming onde o filme está disponível.
	 * @return Uma string indicando o resultado da operação: "FILME ADICIONADO", "POSIÇÃO INVÁLIDA", 
	 * "FILME INVALIDO" ou "FILME JÁ ADICIONADO".
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String plataforma) {
	    if (posicao < 1 || posicao > TAMANHO) {
	        return "POSIÇÃO INVÁLIDA";
	    }
	    if (nome.trim().isEmpty()) {
	        return "FILME INVALIDO";
	        
	    }
	    if (existeNaLista(nome, ano)) { 
	        return "FILME JÁ ADICIONADO";
	    }
	    this.filmes[posicao - 1] = new Filme(nome, ano, plataforma);
	    return "FILME ADICIONADO";
	}
	
	private boolean existeNaLista(String nome, String ano) {
	    Filme filmeTemp = new Filme(nome, ano, ""); 
	    for (Filme f : this.filmes) {
	        if (f != null && f.equals(filmeTemp)) { 
	            return true;
	        }
	    }
	    return false;
	}
	

	public Filme[] getHots() {
		return hots;
	}

	public void setHots(Filme[] hots) {
		this.hots = hots;
	}

	
	/**
	 * Retorna uma string com os detalhes de um filme em uma posição específica.
	 * Se o filme estiver na hotlist, um emoji de fogo é adicionado.
	 * * @param posicao A posição do filme no sistema (1 a 100).
	 * @return Uma string formatada com os detalhes do filme, uma mensagem de "POSIÇÃO INVÁLIDA!"
	 * ou uma string vazia se não houver filme na posição.
	 */
	public String detalharFilme(int posicao) {		
		if (posicao < 1 || posicao > TAMANHO) {
	        return "POSIÇÃO INVÁLIDA!";
	    }
		
		if(filmes[posicao - 1] == null) {
			return "";
		}
		
	    Filme filme = filmes[posicao - 1]; 
	    if("sim".equals(filmes[posicao - 1].getHot())) {
	    	if (filme.getLancamento() == "") {
		    	return "🔥 " + filme.getNome() + "\n" + filme.getPlatafoma();
		    } else {
		    	return "🔥 " + filme.getNome() + ", " + filme.getLancamento() + "\n" + filme.getPlatafoma();
		    }
	    } else {	    
	    	if (filme.getLancamento() == "") {
	    		return filme.getNome() + "\n" + filme.getPlatafoma();
	    	} else {
	    		return filme.getNome() + ", " + filme.getLancamento() + "\n" + filme.getPlatafoma();
	    	}
	    }	
	}
	
	
	/**
	 * Adiciona um filme da lista principal à lista de "Hot" em uma posição específica.
	 * * @param posicaoFilme A posição do filme na lista principal (1 a 100).
	 * @param posicaoHotlist A posição na Hotlist onde o filme será inserido.
	 */
	public void adicionarParaHot(int posicaoFilme, int posicaoHotlist) {
		int indice = posicaoFilme - 1;

		if (indice < 0 || indice >= filmes.length || filmes[indice] == null) {
			return;
		}
		hots[posicaoHotlist] = filmes[indice];
		filmes[indice].setHot("sim");
	}


	/**
	 * Exibe no console todos os filmes que estão na Hotlist.
	 */
	public void exibirHotList() {
		for (int i = 0; i < hots.length; i++) {
			if (hots[i] != null) {
				System.out.println(i + " - " + hots[i].toStringHotList());
			}
		}
	}

	/**
	 * Remove um filme da Hotlist com base na sua posição na lista.
	 * * @param posicaoHot A posição do filme a ser removido da Hotlist.
	 */
	public void removerHot(int posicaoHot) {
	    if (posicaoHot < 0 || posicaoHot >= hots.length || hots[posicaoHot] == null) {
	        return; 
	    }
	    
	    while (true) { 
	    	int cont = 0;
	    	if(hots[posicaoHot].getNome().equals(filmes[cont].getNome())) {
	    		filmes[cont].setHot("não");
	    		break;
	    	}
	    	
	    }		    

	    hots[posicaoHot] = null;

	}

}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;

	


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

package filmnow;

import java.io.FileNotFoundException;

import java.util.Scanner;

/**
 * Interface com menus de texto para manipular o sistema FilmNow.
 *
 * @author [REDACTED]
 *
 */
public class MainFilmNow {

    /**
     * Método principal que inicializa o sistema FilmNow.
     * Carrega os filmes de um arquivo inicial e entra no loop principal
     * de interação com o usuário através de um menu de texto.
     *
     * @param args Argumentos da linha de comando (não utilizados).
     */
    public static void main(String[] args) {
        FilmNow fn = new FilmNow();

        System.out.println("Carregando filmes ...");
        try {
            /*
             * Essa é a maneira de lidar com possíveis erros por falta do arquivo.
             */
            carregaFilmes("filmes_inicial.csv", fn);
        } catch (FileNotFoundException e) {
            System.err.println("Arquivo não encontrado: " + e.getMessage());
        } catch (IOException e) {
            System.err.println("Erro lendo arquivo: " + e.getMessage());
        }

        Scanner scanner = new Scanner(System.in);
        String escolha = "";
        while (true) {
            escolha = menu(scanner);
            comando(escolha, fn, scanner);

        }

    }



    /**
     * Exibe o menu e captura a escolha do/a usuário/a.
     *
     * @param scanner Para captura da opção do usuário/a.
     * @return O comando escolhido.
     */

    private static String menu(Scanner scanner) {
        System.out.println("\n---\nMENU\n" + "(A)Adicionar filme\n" + "(M)Mostrar todos\n" + "(D)Detalhar filme\n"
                + "(E)Exibir Hotlist\n" + "(H)Atribuir Hot\n" + "(R)Remover Hot\n" + "(S)Sair\n" + "\n" + "Opção> ");
        return scanner.next().toUpperCase();
    }

    /**
     * Interpreta a opção escolhida por quem está usando o sistema.
     *
     * @param opcao   Opção digitada.
     * @param fn      O sistema FilmNow que estamos manipulando.
     * @param scanner Objeto scanner para o caso do comando precisar de mais input.
     */
    private static void comando(String opcao, FilmNow fn, Scanner scanner) {
        switch (opcao) {
        case "A":
            String resultado = adicionaFilme(fn, scanner);
            System.out.println(resultado);
            break;
        case "M":
            mostrarFilmes(fn);
            break;
        case "D":
            detalharFilme(fn, scanner);
            break;
        case "E":
            fn.exibirHotList();
            break;
        case "H":
            atribuirHot(fn, scanner);
            break;
        case "R":
            System.out.print("\nPosicao> ");
            int posicaoHot = scanner.nextInt();
            fn.removerHot(posicaoHot);
            break;
        case "S":
            sai();
            break;
        default:
            System.out.println("Opção inválida!");
        }
    }



    /**
     * Imprime lista de filmes.
     *
     * @param fn O sistema FilmNow a ser manipulado.
     */
    private static void mostrarFilmes(FilmNow fn) {

        Filme[] filmes = fn.getFilmes();
        for (int i = 0; i < filmes.length; i++) {
            if (filmes[i] != null) {
                System.out.println(formataFilme(i+1, filmes[i]));
            }
        }
    }


    private static void detalharFilme(FilmNow fn, Scanner scanner) {
        System.out.print("\nQual filme> ");
        int posicao = scanner.nextInt();
        System.out.println(fn.detalharFilme(posicao));
    }



    /**
     * Formata um filme para impressão.
     *
     * @param posicao A posição do filme (que é exibida)
     * @param filme   O filme a ser impresso.
     * @return A String formatada.
     */
    private static String formataFilme(int posicao, Filme filme) {
        return posicao + " - " + filme;
    }

    /**
     * Cadastra um filme no sistema.
     *
     * @param fn      O sistema FilmNow a ser manipulado.
     * @param scanner Scanner para pedir informações do contato.
     */
    private static String adicionaFilme(FilmNow fn, Scanner scanner) {
        System.out.print("\nPosição> ");
        int posicao = scanner.nextInt();
        scanner.nextLine();

        System.out.print("\nNome> ");
        String nome = scanner.nextLine();

        System.out.print("\nAno> ");
        String ano = scanner.nextLine();
       //canner.nextLine();

        System.out.print("\nLocal> ");
        String local = scanner.nextLine();

        return fn.cadastraFilme(posicao, nome, ano, local);

    }

    
    /**
     * Sai da aplicação.
     */
    private static void sai() {
        System.out.println("\nVlw flw o/");
        System.exit(0);
    }


    /**
     * Lê carga de filmes de um arquivo csv.
     *
     * @param arquivoFilmes O caminho para o arquivo.
     * @param fn            O sistema FilmNow a ser populado com os dados.
     * @throws IOException Caso o arquivo não exista ou não possa ser lido.
     */
    private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
        LeitorFilmNow leitor = new LeitorFilmNow();

        int carregados = leitor.carregaContatos(arquivoFilmes, fn);
        System.out.println("Carregamos " + carregados + " registros.");

        System.out.print("\nPosicao> ");

        fn.adicionarParaHot(posicaoFilme, posicaoHot);
        System.out.println("ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot);
