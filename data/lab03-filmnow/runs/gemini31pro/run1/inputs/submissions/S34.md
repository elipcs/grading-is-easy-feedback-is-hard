package filmnow;

/**
 * Representa um filme no sistema FilmNow.
 * Cada filme possui um nome, ano de lançamento (como String) e local onde está disponível.
 */
public class Filme {
    private String nome;
    private String ano;
    private String local;

    /**
     * Constrói um filme com nome, ano e local.
     * 
     * @param nome O nome do filme (não pode ser vazio ou nulo)
     * @param ano O ano de lançamento do filme como String (não pode ser nulo)
     * @param local O local onde o filme está disponível (não pode ser vazio ou nulo)
     */
    public Filme(String nome, String ano, String local) {
        this.nome = nome;
        this.ano = ano;
        this.local = local;
    }

    /**
     * Retorna o nome do filme.
     * 
     * @return O nome do filme
     */
    public String getNome() {
        return nome;
    }

    /**
     * Retorna o ano de lançamento do filme como String.
     * 
     * @return O ano de lançamento
     */
    public String getAno() {
        return ano;
    }

    /**
     * Retorna o local onde o filme está disponível.
     * 
     * @return O local de disponibilidade
     */
    public String getLocal() {
        return local;
    }

    /**
     * Retorna uma string formatada com o nome e ano do filme.
     * 
     * @return String no formato "Nome, Ano"
     */
    public String nomeAno() {
        return this.nome + ", " + this.ano;
    }

    /**
     * Retorna uma representação em string do filme.
     * 
     * @return String no formato "Nome, Ano\nLocal"
     */
    public String toString() {
        return this.nome + ", " + this.ano + "\n" + this.local;
    }

    /**
     * Verifica se este filme é igual a outro objeto.
     * Dois filmes são considerados iguais se tiverem o mesmo nome e ano.
     * 
     * @param obj O objeto a ser comparado
     * @return true se os filmes forem iguais, false caso contrário
     */
    @Override
    public boolean equals(Object obj) {
        if (obj == null) {
            return false;
        }
        if (obj.getClass() != this.getClass()) {
            return false;
        }
        Filme outro = (Filme) obj;
        return this.nome.equals(outro.nome) && this.ano.equals(outro.ano);
    }

    /**
     * Retorna um código hash para o filme, baseado no nome e ano.
     * 
     * @return O código hash do filme
     */
    @Override
    public int hashCode() {
        return (this.nome + this.ano).hashCode();
    }
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos.
 * Permite cadastrar até 100 filmes e gerenciar uma HotList com os 10 filmes mais importantes.
 */
public class FilmNow {
    /** Tamanho máximo da lista de filmes */
    private static final int TAMANHO = 100;
    
    /** Tamanho máximo da HotList */
    private static final int TAMANHO_HOTLIST = 10;

    /** Array de filmes cadastrados */
    private Filme[] filmes;
    
    /** Array da HotList de filmes */
    private Filme[] hotList;

    /**
     * Constrói um FilmNow vazio, com capacidade para 100 filmes e 10 na HotList.
     */
    public FilmNow() {
        this.filmes = new Filme[TAMANHO];
        this.hotList = new Filme[TAMANHO_HOTLIST];
    }

    /**
     * Retorna o array de filmes cadastrados.
     * 
     * @return Array de filmes
     */
    public Filme[] getFilmes() {
        return this.filmes;
    }

    /**
     * Retorna o array da HotList.
     * 
     * @return Array da HotList
     */
    public Filme[] getHotList() {
        return this.hotList;
    }

    /**
     * Adiciona um filme na posição especificada.
     * 
     * @param posicao Posição onde o filme será adicionado (1 a 100)
     * @param nome Nome do filme (não pode ser vazio)
     * @param ano Ano de lançamento do filme como String (não pode ser nulo)
     * @param local Local onde o filme está disponível (não pode ser vazio)
     * @return Mensagem indicando o resultado da operação
     */
    public String adicionaFilme(int posicao, String nome, String ano, String local) {
        if (posicao < 1 || posicao > TAMANHO) {
            return "POSIÇÃO INVÁLIDA";
        }

        if (nome == null || nome.isEmpty() || 
            ano == null || 
            local == null || local.isEmpty()) {
            return "FILME INVÁLIDO";
        }

        Filme novo = new Filme(nome, ano, local);

        for (Filme f : filmes) {
            if (f != null && f.equals(novo)) {
                return "FILME JÁ ADICIONADO";
            }
        }

        filmes[posicao - 1] = novo;
        return "FILME ADICIONADO";
    }

    /**
     * Exibe a lista de filmes marcados como Hot.
     * 
     * @return String formatada com a HotList, ou string vazia se não houver filmes
     */
    public String exibirHotList() {
        String lista = "";

        for (int i = 0; i < hotList.length; i++) {
            if (hotList[i] != null) {
                lista += (i + 1) + " - " + hotList[i].nomeAno() + "\n";
            }
        }

        return lista;
    }

    /**
     * Retorna os detalhes de um filme em uma posição específica.
     * 
     * @param posicao Posição do filme a ser detalhado (1 a 100)
     * @return String com os detalhes do filme ou mensagem de erro se posição inválida
     */
    public String detalharFilme(int posicao) {
        if (posicao < 1 || posicao > TAMANHO || filmes[posicao - 1] == null) {
            return "POSIÇÃO INVÁLIDA";
        }

        Filme f = filmes[posicao - 1];
        String resultado = "";

        if (estaNaHotList(f)) {
            resultado += "🔥 ";
        }

        resultado += f.toString();
        return resultado;
    }

    /**
     * Verifica se um filme está na HotList.
     * 
     * @param filme Filme a ser verificado
     * @return true se o filme está na HotList, false caso contrário
     */
    private boolean estaNaHotList(Filme filme) {
        for (Filme f : hotList) {
            if (f != null && f.equals(filme)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Adiciona um filme à HotList na posição especificada.
     * 
     * @param posicaoFilme Posição do filme no sistema (1 a 100)
     * @param posicaoHot Posição na HotList (1 a 10)
     * @return Mensagem indicando o resultado da operação
     */
    public String atribuirHot(int posicaoFilme, int posicaoHot) {
        if (!verificaFilme(posicaoFilme) || !verificaHot(posicaoHot)) {
            return "POSIÇÃO INVÁLIDA";
        }

        Filme novoFilme = filmes[posicaoFilme - 1];

        if (verificarDualidade(novoFilme)) {
            return "FILME JÁ ESTÁ NA HOTLIST";
        }

        hotList[posicaoHot - 1] = novoFilme;
        return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!";
    }

    /**
     * Verifica se um filme já está na HotList.
     * 
     * @param filme Filme a ser verificado
     * @return true se o filme já está na HotList, false caso contrário
     */
    private boolean verificarDualidade(Filme filme) {
        for (Filme f : hotList) {
            if (f != null && f.equals(filme)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Verifica se a posição do filme é válida e se existe filme nessa posição.
     * 
     * @param posicaoFilme Posição a ser verificada
     * @return true se a posição é válida e contém um filme, false caso contrário
     */
    private boolean verificaFilme(int posicaoFilme) {
        return posicaoFilme >= 1 && posicaoFilme <= TAMANHO && filmes[posicaoFilme - 1] != null;
    }

    /**
     * Verifica se a posição na HotList é válida.
     * 
     * @param posicaoHot Posição a ser verificada
     * @return true se a posição é válida, false caso contrário
     */
    private boolean verificaHot(int posicaoHot) {
        return posicaoHot >= 1 && posicaoHot <= TAMANHO_HOTLIST;
    }

    /**
     * Remove um filme da HotList na posição especificada.
     * 
     * @param posicaoHot Posição na HotList a ser removida (1 a 10)
     */
    public void removerHot(int posicaoHot) {
        if (posicaoHot >= 1 && posicaoHot <= TAMANHO_HOTLIST) {
            hotList[posicaoHot - 1] = null;
        }
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes.
 */
public class LeitorFilmNow {
    /** Índice da coluna da posição no arquivo CSV */
    private static final int COLUNA_POSICAO = 0;
    /** Índice da coluna do nome no arquivo CSV */
    private static final int COLUNA_NOME = 1;
    /** Índice da coluna do ano no arquivo CSV */
    private static final int COLUNA_ANO = 2;
    /** Índice da coluna do local no arquivo CSV */
    private static final int COLUNA_LOCAL = 3;

    /**
     * Lê filmes de um arquivo CSV e os coloca no sistema.
     * 
     * @param arquivoFilmes Caminho para arquivo contendo os filmes
     * @param fn O sistema FilmNow a manipular
     * @return O número de filmes adicionados ao sistema
     * @throws IOException Caso não tenhamos permissão de ler o arquivo
     * @throws FileNotFoundException Caso o arquivo não exista
     */
    public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
        int carregados = 0;

        try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
            String linha;
            while ((linha = br.readLine()) != null) {
                carregados += 1;
                if (carregados == 1) {
                    // pulamos a primeira linha, o cabeçalho
                    continue;
                }
                String[] campos = linha.split(",");
                processaLinhaCsvFilmes(campos, fn);
            }
        }
        
        return carregados;
    }

    /**
     * Processa uma linha do arquivo CSV e adiciona o filme ao sistema.
     * 
     * @param campos As informações lidas do csv
     * @param fn O sistema FilmNow a manipular
     */
    private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
        int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
        String nome = campos[COLUNA_NOME];
        String ano = campos[COLUNA_ANO];
        String local = campos[COLUNA_LOCAL];

        fn.adicionaFilme(posicao, nome, ano, local);
    }
}

        String escolha = "";
        while (true) {
            escolha = menu(scanner);
            comando(escolha, fn, scanner);
        }
    }

    /**
     * Exibe o menu e captura a escolha do usuário.
     * 
     * @param scanner Para captura da opção do usuário
     * @return O comando escolhido em maiúsculas
     */
    private static String menu(Scanner scanner) {
        System.out.println(
            "\n---\nMENU\n" + 
            "(A)Adicionar filme\n" + 
            "(M)Mostrar todos\n" + 
            "(D)Detalhar filme\n" +
            "(E)Exibir HotList\n" +
            "(H)Atribuir Hot\n" +
            "(R)Remover Hot\n" +
            "(S)Sair\n" + 
            "\n" + 
            "Opção> ");
        return scanner.next().toUpperCase();
    }

    /**
     * Interpreta a opção escolhida e executa o comando correspondente.
     * 
     * @param opcao   Opção digitada
     * @param fn      O sistema FilmNow a ser manipulado
     * @param scanner Scanner para entrada adicional quando necessário

    private static void comando(String opcao, FilmNow fn, Scanner scanner) {
        switch (opcao) {
            case "A":
                adicionaFilme(fn, scanner);
                break;
            case "M":
                mostrarFilmes(fn);
                break;
            case "D":
                detalharFilme(fn, scanner);
                break;
            case "E":
                exibirHot(fn);
                break;
            case "H":
                atribuirHot(fn, scanner);
                break;
            case "R":
                removerHot(fn, scanner);
                break;
            case "S":
                sai();
                break;
            default:
                System.out.println("Opção inválida!");
        }
    }

    /**
     * Imprime lista de filmes cadastrados.

     * @param fn O sistema FilmNow a ser manipulado
