/**

public class Filme {
	private String nome;
	private String ano;
	private String local;

	/**
	 * Construtor do Filme 
	 * 
	 * @param nome Nome do filme
	 * @param ano Ano que o filme saiu
	 * @param local Local em que pode ser assistido
	 */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
	}
	
	/**
	 * Sequencia de gets para serem usados no Filmnow.
	 */

		return nome;
	}

	public String getAno() {
		return ano;
	}

	public String getLocal() {
		return local;
	}

	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}

	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);
	}
	
	
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	private static final int TAMANHOHOT = 10;

	private Filme[] filmes;
	private Filme[] hotList;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[TAMANHOHOT];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}
	
	/**
	 * Acessa a lista de hots mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getHots() {
		return this.hotList.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * 
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		return filmes[posicao];
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 * @return retorno em String do resultado do método.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if(posicao > 100 || posicao < 1) {
			return "POSIÇÃO INVÁLIDA";
		}
		else {
			if (nome == null || nome.trim().isEmpty()) {
				return "FILME INVÁLIDO";
			}
			else {
				if (local == null || local.trim().isEmpty()) {
					return "FILME INVÁLIDO";
				}else {
					if(checaFilme(nome, ano)) {
						return "FILME JÁ ADICIONADO";
					}else {
						filmes[posicao-1] = new Filme(nome, ano, local);
						return "FILME ADICIONADO";
					}
				}
			}
		}
	}
	
	/**
	 * Filme detalhado se o filme existir na posição especificada 
	 * 
	 * @param posicao especificada pelo usuário para buscar no array		
	 * @return string formatada
	 */
	public String filmeDetalhado(int posicao) {
		if (posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		}else if(filmes[posicao-1] == null){
			return "";
		}else {
			Filme filme = filmes[posicao-1];
			for (int i = 0; i < hotList.length; i++) {
		        Filme hot = hotList[i];
			    if (hot	 != null) {
			        if (hot.equals(filme)) {
			        	if(filme.getAno() != "") {
			        		return "\n🔥" + filme.getNome() + ", " + filme.getAno() + "\n" + filme.getLocal();
			        	}else {
							return "\n🔥" + filme.getNome() + "\n" + filme.getLocal();
			        	}
			        }
			    }
			}
			if(filme.getAno() != "") {
				return "\n" + filme.getNome() + ", " + filme.getAno() + "\n" + filme.getLocal();
			}else {
				return "\n" + filme.getNome() + "\n" + filme.getLocal();
			}
		}
	}
	
	/**
	 * Checa se um filme ja existe no array filmes antes de cadastrá-lo.
	 * 
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @return boorleano
	 */
	public boolean checaFilme(String nome, String ano) {
		for (int i = 0; i < filmes.length; i++){
			if(filmes[i] == null){
				;
			}
			else if(filmes[i].getNome().equals(nome) && filmes[i].getAno().equals(ano)) {
				return true;
			}
		}
		return false; 
	}
	
	/**
	 * Passa pelo array de filmes para saber se existe filme na posição -1 indicada, também trata de posições fora do array
	 * 
	 * @param posicao do array que depois é subtraida
	 * @return boorleano
	 */
	public boolean existeFilme(int posicao) {
		if(posicao > 100 || posicao < 1) {
			return false;
		}else if(filmes[posicao-1] == null){
			return false;
		}else {
			return true;
		}
	}
	
	/**
	 * Coloca "hot" em um filme em uma posição e o adiciona no array "hots". 
	 * Se já existir filme na posição, sobrescreve o anterior. 
	 * 
	 * @param posicao Posição do filme.
	 * @param Filme filme que vai receber para adicionar na posição
	 * @return retorno em string do resultado do método
	 */
	public String atribuirHot(int posicao, Filme filme) {
		if(posicao > 10 || posicao < 1) {
			return "\nPOSIÇÃO INVÁLIDA";
		}else {
			hotList[posicao-1] = new Filme(filme.getNome(), filme.getAno(), filme.getLocal());
			return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicao + "!\n";
		}
	}
	/**
	 * Método que remove um hot do array hotlist
	 * 
	 * @param posicao Posição do filme
	 * @return retorno em string do resultado do método
	 */
	public String removerHot(int posicao) {
		if(posicao > 10 || posicao < 1) {
			return "\nPOSIÇÃO INVÁLIDA";
		}else if(hotList[posicao-1] == null){
			return "\nSEM HOT NESSA POSIÇÃO";
		}else {
			hotList[posicao-1] = null;
			return "";
		}
	}
	
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();
		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

	/**

				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" +

						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)Sair\n" +
						"\n" + 
						"Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 *

	 * @param fn  O sistema FilmNow que estamos manipulando.

			adicionaFilme(fn, scanner);

		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;

			mostrarHot(fn);
			break;
		case "H":
			atribuirHot(fn,scanner);
			break;
		case "R":
			removerHot(fn,scanner);
			break;
		case "S":
			sai();
			break;

			System.out.println("OPÇÃO INVÁLIDA!");
			
		}
