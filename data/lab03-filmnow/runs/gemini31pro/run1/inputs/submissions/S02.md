package filmnow;

import java.util.Objects;

/**
 * Representa um filme com nome, ano e local de exibição. Dois filmes são
 * considerados iguais se tiverem o mesmo nome e ano.
 * 
 * @author [REDACTED]
 */
public class Filme {
	private String nome;
	private String ano;
	private String local;

	/**
	 * Constrói um novo filme com nome, ano e local.
	 * 
	 * @param nome  o nome do filme
	 * @param ano   o ano de lançamento do filme (pode ser vazio)
	 * @param local o local ou plataforma onde o filme está disponível
	 */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
	}

	/**
	 * Retorna o nome e o ano do filme concatenados. Ex: "Avatar, 2009"
	 * 
	 * @return uma string com nome e ano do filme
	 */
	public String nomeAno() {
		return nome + ", " + ano;
	}

	/**
	 * Retorna o nome do filme.
	 * 
	 * @return o nome
	 */
	public String getNome() {
		return nome;
	}

	/**
	 * Retorna o ano do filme.
	 * 
	 * @return o ano
	 */
	public String getAno() {
		return ano;
	}

	/**
	 * Retorna o local onde o filme está disponível.
	 * 
	 * @return o local
	 */
	public String getLocal() {
		return local;
	}

	/**
	 * Retorna o nome e o ano do filme, omitindo o ano se for nulo ou vazio.
	 * 
	 * @return nome ou "nome, ano" se o ano existir
	 */
	public String getNomeAno() {
		if (ano == null || ano.isEmpty()) {
			return nome;
		}
		return nome + ", " + ano;
	}

	/**
	 * Gera um hash baseado em nome e ano do filme.
	 */
	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}

	/**
	 * Verifica se dois filmes são iguais comparando nome e ano.
	 */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);
	}

	/**
	 * Verifica se o nome fornecido é válido (não nulo nem vazio).
	 * 
	 * @param nome o nome a validar
	 * @return true se o nome for válido
	 */
	public static boolean entradasValidasNome(String nome) {
		return nome != null && !nome.isEmpty();
	}

	/**
	 * Verifica se o local fornecido é válido (não nulo nem vazio).
	 * 
	 * @param local o local a validar
	 * @return true se o local for válido
	 */
	public static boolean entradasValidasLocal(String local) {
		return local != null && !local.isEmpty();
	}

	/**
	 * Retorna a representação textual do filme: "Nome, ano" em uma linha e local na
	 * linha seguinte.
	 * 
	 * @return a string formatada do filme
	 */
	@Override
	public String toString() {
		return getNomeAno() + "\n" + local;
	}

	/**
	 * Verifica se um filme com o mesmo nome e ano já existe no sistema.
	 * 
	 * @param fn   a instância do sistema FilmNow
	 * @param nome o nome do filme a verificar
	 * @param ano  o ano do filme a verificar
	 * @return true se o filme já existir no sistema
	 */
	public static boolean confereExistencia(FilmNow fn, String nome, String ano) {
		String nomeAno = nome + ", " + ano;

		for (String f : fn.getFilmes()) {
			if (f == null || f.isEmpty())
				continue;

			String[] partes = f.split("\n");
			if (partes.length < 1)
				continue;

			String nomeAnoExistente = partes[0];

			if (nomeAno.equals(nomeAnoExistente)) {
				return true;
			}
		}
		return false;
	}

}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir até 100 filmes e
 * até 10 na hotlist.
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {

	private static final int TAMANHO = 100;

	private String[] filmes; // uma representação simploria da lista de filmes

	private String[] hotlist = new String[10];

	/**
	 * Cria o FilmNow com capacidade para armazenar até 100 filmes e 10 filmes na
	 * hotlist.
	 */
	public FilmNow() {
		this.filmes = new String[TAMANHO];
		this.hotlist = new String[10];
	}

	/**
	 * Retorna uma cópia da lista de filmes cadastrados.
	 * 
	 * @return O array de filmes.
	 */
	public String[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Retorna uma cópia da hotlist (filmes destacados).
	 * 
	 * @return O array de filmes na hotlist.
	 */
	public String[] getHotList() {
		return this.hotlist.clone();
	}

	/**
	 * Retorna os dados de um filme em uma determinada posição.
	 * 
	 * @param posicao Posição do filme no sistema (1 a 100).
	 * @return Dados do filme ou null se a posição for inválida ou vazia.
	 */
	public String getFilme(int posicao) {
		if (posicao < 1 || posicao > 100)
			return null;
		return filmes[posicao - 1];
	}

	/**
	 * Cadastra um novo filme na posição especificada. Substitui o anterior, se
	 * houver.
	 * 
	 * @param posicao Posição onde o filme será adicionado (1 a 100).
	 * @param nome    Nome do filme.
	 * @param ano     Ano de lançamento do filme.
	 * @param local   Plataforma ou local onde pode ser assistido.
	 * @return Mensagem de sucesso ou erro.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if (posicao < 1 || posicao > TAMANHO)
			return "POSICAO INVALIDA";
		if (nome == null || nome.isEmpty() || local == null || local.isEmpty())
			return "FILME INVALIDO";

		if (Filme.confereExistencia(this, nome, String.valueOf(ano))) {
			return "FILME JA ADICIONADO";
		}

		Filme filme = new Filme(nome, String.valueOf(ano), local);
		filmes[posicao - 1] = filme.toString();

		return "FILME ADICIONADO";
	}

	/**
	 * Retorna os dados detalhados de um filme, incluindo ícone 🔥 se estiver na
	 * hotlist.
	 * 
	 * @param posicao Posição do filme (1 a 100).
	 * @return Dados formatados do filme ou mensagens de erro.
	 */
	public String detalhaFilme(int posicao) {
		if (posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA!";
		}

		String filme = getFilme(posicao);
		if (filme == null) {
			return null;
		}

		if (estaNaHotList(filme)) {
			String[] partes = filme.split("\n", 2);
			return "🔥 " + partes[0] + "\n" + partes[1];
		}

		return filme;
	}

	/**
	 * Atribui um filme da lista principal a uma posição na hotlist.
	 * 
	 * @param posicaoFilme Posição do filme na lista principal (1 a 100).
	 * @param posicaoHot   Posição desejada na hotlist (1 a 10).
	 * @return Mensagem de sucesso ou erro.
	 */
	public String atribuirHot(int posicaoFilme, int posicaoHot) {
		if (posicaoHot < 1 || posicaoHot > 10 || posicaoFilme < 1 || posicaoFilme > 100) {
			return "POSIÇÃO INVÁLIDA";
		}

		String filme = this.filmes[posicaoFilme - 1];
		if (filme == null) {
			return "POSIÇÃO INVÁLIDA";
		}

		for (String f : hotlist) {
			if (filme.equals(f)) {
				return "FILME JÁ ADICIONADO";
			}
		}

		hotlist[posicaoHot - 1] = filme;
		return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!";
	}

	/**
	 * Remove um filme da hotlist.
	 * 
	 * @param posicaoHot Posição na hotlist (1 a 10).
	 * @return Mensagem de sucesso ou erro.
	 */
	public String removerHot(int posicaoHot) {
		if (posicaoHot < 1 || posicaoHot > 10 || hotlist[posicaoHot - 1] == null) {
			return "POSIÇÃO INVÁLIDA";
		}

		hotlist[posicaoHot - 1] = null;
		return "REMOVIDO DA HOTLIST NA POSIÇÃO " + posicaoHot + "!";
	}

	/**
	 * Verifica se um determinado filme está presente na hotlist.
	 * 
	 * @param filme String representando o filme.
	 * @return true se estiver na hotlist, false caso contrário.
	 */
	public boolean estaNaHotList(String filme) {
		for (String hot : hotlist) {
			if (filme != null && filme.equals(hot)) {
				return true;
			}
		}
		return false;
	}

	/**
	 * Exibe todos os filmes da hotlist, com numeração e nome + ano.
	 * 
	 * @return String formatada com todos os filmes da hotlist.
	 */
	public String exibirHot() {
		String resultado = "";
		for (int i = 0; i < hotlist.length; i++) {
			if (hotlist[i] != null) {
				String[] partes = hotlist[i].split(",");
				String nome = partes[0].trim();
				String ano = partes.length > 1 ? partes[1].split("\n")[0].trim() : "";

				resultado += (i + 1) + " - " + nome;
				if (!ano.isBlank()) {
					resultado += ", " + ano;
				}
				resultado += "\n";
			}
		}

		if (!resultado.isEmpty()) {
			resultado = resultado.substring(0, resultado.length() - 1);
		}

		return resultado;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de
 * filmes.
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;

	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * 
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn            O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException           Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}

		return carregados;
	}

	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial.
	 * 
	 * @param campos As informações lidas do csv.
	 * @param fn     O sistema FilmNow a manipular.
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

/**
 * Interface com menus texto para manipular o sistema FilmNow. Permite
 * cadastrar, exibir e detalhar filmes, bem como gerenciar uma lista especial
 * chamada Hotlist.
 * 
 * @author [REDACTED]
 */
public class MainFilmNow {

	public static void main(String[] args) {
		FilmNow fn = new FilmNow();

		System.out.println("Carregando filmes ...");
		try {
			/*
			 * Carrega os filmes a partir de um arquivo CSV. Lida com possíveis exceções de
			 * leitura.
			 */
			carregaFilmes("filmes_inicial.csv", fn);
		} catch (FileNotFoundException e) {
			System.err.println("Arquivo não encontrado: " + e.getMessage());
		} catch (IOException e) {
			System.err.println("Erro lendo arquivo: " + e.getMessage());
		}

		Scanner scanner = new Scanner(System.in);
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}
	}

	/**
	 * Exibe o menu e captura a escolha do usuário.
	 * 
	 * @param scanner Scanner para entrada de dados.
	 * @return A opção escolhida pelo usuário (em maiúsculas).
	 */
	private static String menu(Scanner scanner) {
		System.out.println("\n---\nMENU\n" + "(A)Adicionar filme\n" + "(M)Mostrar todos\n" + "(D)Detalhar filme\n"
				+ "(E)Exibir Hot\n" + "(H)Atribuir Hot\n" + "(R)Remover Hot\n" + "(S)Sair\n" + "\n" + "Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Executa a operação correspondente à opção escolhida pelo usuário.
	 * 
	 * @param opcao   A opção digitada.
	 * @param fn      O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para ler entradas adicionais do usuário.
	 */
	private static void comando(String opcao, FilmNow fn, Scanner scanner) {
		switch (opcao) {
		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibirHot(fn);
			break;
		case "H":
			atribuirHot(fn, scanner);
			break;
		case "R":
			removerHot(fn, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("Opção inválida!");
		}
	}

	/**
	 * Exibe a lista de filmes cadastrados.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 */
	private static void mostrarFilmes(FilmNow fn) {
		String[] filmes = fn.getFilmes();
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				String nome = filmes[i].split(",")[0];
				System.out.println((i + 1) + " - " + nome);
			}
		}
	}

	/**
	 * Exibe os detalhes de um filme a partir de sua posição.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 * @param sc Scanner para capturar a posição do filme.
	 */
	private static void detalharFilme(FilmNow fn, Scanner sc) {
		System.out.print("\nPosição> ");
		int posicao = sc.nextInt();
		sc.nextLine();

		System.out.println(fn.detalhaFilme(posicao));
	}

	/**
	 * Cadastra um novo filme no sistema.
	 * 
	 * @param fn      O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar as informações do filme.
	 */
	public static void adicionaFilme(FilmNow fn, Scanner scanner) {
		System.out.print("\nPosição no sistema> ");
		int posicao = scanner.nextInt();
		scanner.nextLine();

		System.out.print("\nNome> ");
		String nome = scanner.nextLine();

		System.out.print("\nAno> ");
		String ano = scanner.nextLine();

		System.out.print("\nLocal> ");
		String local = scanner.nextLine();

		String resultado = fn.cadastraFilme(posicao, nome, ano, local);
		System.out.println(resultado);
	}

	/**
	 * Atribui um filme a uma posição da hotlist.
	 * 
	 * @param fn      O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar posições do filme e da hotlist.
	 */
	private static void atribuirHot(FilmNow fn, Scanner scanner) {
		System.out.print("Filme> ");
		int posicaoFilme = scanner.nextInt();
		System.out.print("Posição> ");
		int posicaoHot = scanner.nextInt();
		scanner.nextLine();
		System.out.println(fn.atribuirHot(posicaoFilme, posicaoHot));
	}

	/**
	 * Remove um filme da hotlist a partir da posição fornecida.
	 * 
	 * @param fn      O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar a posição na hotlist.
	 */
	private static void removerHot(FilmNow fn, Scanner scanner) {
		System.out.print("Posição> ");
		int posicaoHot = scanner.nextInt();
		scanner.nextLine();
		System.out.println(fn.removerHot(posicaoHot));
	}

	/**
	 * Exibe os filmes presentes na hotlist.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 */
	private static void exibirHot(FilmNow fn) {
		System.out.println(fn.exibirHot());
	}

	/**
	 * Finaliza a execução do programa.
	 */
	private static void sai() {
		System.out.println("\nVlw flw o/");
		System.exit(0);
	}

	/**
	 * Lê os filmes a partir de um arquivo CSV e carrega no sistema.
	 * 
	 * @param arquivoFilmes Caminho para o arquivo.
	 * @param fn            O sistema FilmNow a ser preenchido.
	 * @throws IOException Caso ocorra erro de leitura ou o arquivo não exista.

	private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {

		int carregados = leitor.carregaContatos(arquivoFilmes, fn);
		System.out.println("Carregamos " + carregados + " registros.");
	}
