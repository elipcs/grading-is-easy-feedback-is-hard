package filmnow;

import java.util.ArrayList;
import java.util.List;

/**
 * Representa o sistema FilmNow, que permite aos usuários adicionar, listar e visualizar detalhes de filmes.
 */
public class FilmNow {
    private static final int MAX_FILMES = 100; // Número máximo de filmes na lista
    private Filme[] listaFilmes; // Lista de filmes

    /**
     * Construtor da classe FilmNow.
     * Inicializa a lista de filmes.
     */
    public FilmNow() {
        this.listaFilmes = new Filme[MAX_FILMES];
    }

    /**
     * Adiciona um novo filme à lista.
     *
     * @param posicao       A posição na lista onde o filme deve ser inserido.
     * @param nome          O nome do filme.
     * @param anoLancamento O ano de lançamento do filme.
     * @param local         O local onde o filme pode ser assistido (Netflix, Prime Vídeo, Cinema, etc).
     * @return Uma mensagem de sucesso ou erro.
     */
    public String adicionarFilme(int posicao, String nome, String anoLancamento, String local) {
        if (nome == null || nome.isEmpty()) {
            return "FILME INVALIDO";
        }

        if (local == null || local.isEmpty()) {
            return "FILME INVALIDO";
        }

        /*
        if (listaFilmes.size() >= MAX_FILMES) {
            throw new IllegalArgumentException("POSIÇÃO INVÁLIDA");
        }
         */
        if (posicao < 1 || posicao > MAX_FILMES) {
            return "POSIÇÃO INVÁLIDA";
        }

        Filme novoFilme = new Filme(nome, anoLancamento, local);

        for (Filme filme : listaFilmes) {
            if (filme != null && filme.equals(novoFilme)) {
                return "FILME JÁ CADASTRADO";
            }
        }

        listaFilmes[posicao - 1] = novoFilme;

        return "FILME ADICIONADO";
    }

    /**
     * Lista todos os filmes na lista.
     * Se a lista estiver vazia, exibe uma mensagem indicando isso.
     */
    public List<String> listarFilmes() {
        List<String> resultado = new ArrayList<>();

        if (listaFilmes.length == 0) {
            resultado.add("A lista de filmes está vazia.");
            return resultado;
        }

        for (int i = 0; i < listaFilmes.length; i++) {
            resultado.add(String.format("%d - %s", i + 1, listaFilmes[i].getNome()));
        }

        return resultado;
    }

    /**
     * Exibe os detalhes de um filme específico na lista.
     *
     * @param posicao A posição do filme na lista.
     */
    public String detalhesFilme(int posicao) throws IllegalArgumentException {
        if (posicao < 1 || posicao > MAX_FILMES){
            return "POSIÇÃO INVÁLIDA";
        }
        Filme filme = listaFilmes[posicao - 1];

        if (filme == null) {
            return "";
        }

        if (filme.getAnoLancamento() == null || filme.getAnoLancamento().isEmpty()) {
            return filme.getNome() + "\n" + filme.getLocal();
        }

        return filme.getNome() + ", " + filme.getAnoLancamento() + "\n" + filme.getLocal();
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

/**
 * Classe responsável por ler dados de um arquivo .csv que contém informações sobre filmes e carregá-los no sistema FilmNow.
 */
public class LeitorFilmNow {
    private String nomeArquivo; // Nome do arquivo .csv que contém os dados dos filmes

    /**
     * Construtor da classe LeitorFilmNow.
     *
     * @param nomeArquivo Nome do arquivo .csv que contém os dados dos filmes.
     */
    public LeitorFilmNow(String nomeArquivo) {
        this.nomeArquivo = nomeArquivo;
    }

    /**
     * Carrega os filmes do arquivo .csv e adiciona-os ao sistema FilmNow.
     *
     * @param filmNow O sistema FilmNow onde os filmes serão adicionados.
     */
    public void carregarFilmes(FilmNow filmNow) {
        int id = 1;
        try (BufferedReader br = new BufferedReader(new FileReader(nomeArquivo))) {
            String linha;
            while ((linha = br.readLine()) != null) {
                String[] dados = linha.split(",");
                if (dados.length >= 3) {
                    String nome = dados[0];
                    String anoLancamento = dados[1];
                    String local = dados[2];
                    filmNow.adicionarFilme(id++, nome, anoLancamento, local);
                } else {
                    System.out.println("Erro ao ler linha: " + linha);
                }
            }
        } catch (IOException e) {
            System.out.println("Erro ao ler o arquivo: " + e.getMessage());
        }
    }

    /**
     * Método principal para carregar filmes do arquivo e adicionar ao sistema FilmNow.
     *
     * @param args Argumentos da linha de comando (não utilizados neste caso).
     */
    public static void main(String[] args) {
        LeitorFilmNow leitor = new LeitorFilmNow("filmes.csv");
        FilmNow filmNow = new FilmNow();
        leitor.carregarFilmes(filmNow);
        System.out.println("Filmes carregados com sucesso!");
    }
}

package filmnow;

import java.util.Scanner;

/**
 * Classe principal que implementa uma interface simples de usuário para interagir com a classe FilmNow.
 */

public class MainFilmNow {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        FilmNow filmNow = new FilmNow();

        int opcao;
        do {
            System.out.println("\n### FilmNow - Sua lista de filmes e séries ###");
            System.out.println("1. Adicionar Filme");
            System.out.println("2. Listar Filmes");
            System.out.println("3. Detalhes de um Filme");
            System.out.println("4. Sair");
            System.out.print("Escolha uma opção: ");
            opcao = scanner.nextInt();

            switch (opcao) {
                case 1:
                    scanner.nextLine(); // Limpa o buffer
                    System.out.print("Nome do filme: ");
                    String nome = scanner.nextLine();
                    System.out.print("Ano de lançamento: ");
                    String anoLancamento = scanner.nextLine();
                    System.out.print("Local de visualização: ");
                    String local = scanner.nextLine();
                    System.out.print("Posição na lista: ");
                    int posicao = scanner.nextInt();

                    System.out.println(filmNow.adicionarFilme(posicao, nome, anoLancamento, local));

                    break;
                case 2:
                    for (String filme : filmNow.listarFilmes()) {
                        System.out.println(filme);
                    }

                    break;
                case 3:
                    System.out.print("Posição do filme na lista: ");
                    try {
                        int posicaoDetalhes = scanner.nextInt();
                        System.out.println(filmNow.detalhesFilme(posicaoDetalhes));
                    } catch (Exception e) {
                        System.out.println(e.getMessage());
                    }
                    break;
                case 4:
                    System.out.println("Saindo do FilmNow. Até mais!");
                    break;
                default:
                    System.out.println("Opção inválida. Por favor, escolha uma opção válida.");
            }
        } while (opcao != 4);

        scanner.close();
    }
}
