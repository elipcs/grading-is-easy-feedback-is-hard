package filmnow;

import java.util.Objects;
/**
 * Representa um filme com nome, ano de lançamento, local onde pode ser assistido
 * e um indicador se está na hotlist.
 * @author [REDACTED]
 */

public class Filme {
	
	private String nome;
	private String local;
	private String ano; 
	private boolean hot;
	
	/**
	 * Constrói um novo filme com nome, ano e local especificados.
	 * O filme é inicializado como não pertencente à hotlist.
	 * 
	 * @param nome  Nome do filme (não pode ser nulo ou vazio).
	 * @param ano   Ano de lançamento do filme (pode ser nulo ou vazio).
	 * @param local Local onde o filme foi assistido (não pode ser nulo ou vazio).
	 */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.hot = false;
	}
	/**
	 * Retorna o nome do filme.
	 * @return Nome do filme(não pode ser nulo ou vazio).
	 */
	public String getNome() {
		return nome;
	}
	/**
	 * Retorna o local onde o filme foi assistido.
	 * @return Local do filme(não pode ser nulo ou vazio).
	 */
	public String getLocal() {
		return local;
	}
    /**
     * Retorna o ano de lançamento do filme.
     * @return Ano do filme (pode ser nulo ou vazio).
     */
	public String getAno() {
		return ano;
	}
	/**
	 * Retorna uma representação em String do filme no formato:
	 * "nome, ano\nlocal".
	 * 
	 * @return String formatada representando o filme.
	 */
	public String toString() {
		return nome + ", "+ ano + "\n"+ local;
	}
	/**
	 * Retorna representação booleana do hot.
	 * @return Hot atributo booleano.
	 */
	public boolean getHot() {
		return this.hot;
	}
	/**
	 * Modifica o atributo que caracteriza se o filme está na hotlist ou não.
	 * @param hot true se o filme está na hotlist, false caso contrário.
	 */
	public void setHot(boolean hot) {
		this.hot = hot;
	}
	/**
	 * Gera um código hash para o filme baseado no nome e ano.
	 * 
	 * @return Código hash do filme.
	 */
	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}
	/**
	 * Verifica se este filme é igual a outro objeto.
	 * Dois filmes são considerados iguais se tiverem o mesmo nome e ano.
	 * 
	 * @param obj Objeto a ser comparado.
	 * @return true se os filmes forem iguais, false caso contrário.
	 */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return ano.equals(other.ano) && Objects.equals(nome, other.nome);
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * Sistema que cria uma lista dos filmes mais desejados. Podem existir 10 filmes.
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;// Uma representação simplória do tamanho, tendo como máximo 100.
	
	private Filme[] filmes; //uma representacao simploria da lista de filmes.
	
	private Filme[] hotlist; //uma representacao simploria da hotlist.
		
	/**
	 * Cria o FilmNow.
	 * Cria os arrays de Filme e Hotlist.
	 */
	public FilmNow() {
		this.filmes= new Filme[TAMANHO];
		this.hotlist = new Filme[10];
	}
	
	/**
	 * Acessa a lista de filmes mantida e faz uma cópia.
	 * @return A copia da lista de filmes em um array de String.
	 */

	public String[] getFilmes() {
	
		String[] copia = new String[TAMANHO];
		
		for(int i = 0; i < filmes.length; i++) {
			if(filmes[i]!= null) {
				copia[i] = filmes[i].getNome();
			}
		}
		return copia;
	}

	/**
	 * Acessa os dados de um filme específico. Checa se o filme está na hotlist.
	 * @param posicao Posição do filme no sistema -1 já que o array vai de 0 a 99, e as posições vão de 1 a 100.
	 * @return String indicando o resultado da operação:
     *         - "POSIÇÃO INVÁLIDA" se a posição estiver fora do intervalo 1-100 ou não tiver filme nela.
     *         -  "🔥 nomeFilme, ano \nlocal" se o filme estiver presente na hotlist.
     *         -  "nomeFilme, ano \nlocal" se o filme estiver presente na lista de filmes.
	 */
	public String getFilme(int posicao) {
		if(posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		}
		if(filmes[posicao -1] == null){
			return "";
		}
		
		for (int i = 0;i<hotlist.length;i++) {
			if(filmes[posicao-1].equals(hotlist[i])) {
				return "🔥 " + filmes[posicao-1].toString();
			}
		}
		return  filmes[posicao-1].toString();
	}
	/**
	 * Adiciona um filme em uma posição específica da lista de filmes.
	 * @param posicao Posição na lista principal (1 a 100).
	 * @param nome Nome do filme (não pode ser nulo ou vazio).
	 * @param ano Ano de lançamento do filme (pode ser vazio).
	 * @param local Onde o filme foi assistido(não pode ser nulo ou vazio).
	 * @return String indicando o resultado da operação:
	 *         - "FILME INVALIDO" se nome ou local forem inválidos
	 *         - "POSIÇÃO INVÁLIDA" se a posição estiver fora do intervalo 1-100
	 *         - "FILME JÁ ADICIONADO" se o filme já existe na lista
	 *         - "FILME ADICIONADO" em caso de sucesso
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if (nome == null || nome.trim().isEmpty() || local == null || local.trim().isEmpty()) {
		    return "FILME INVALIDO";
		}
		if(posicao <= 0 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		}
		
		Filme filme= new Filme(nome,ano,local);
		if(!existeFilme(filme)) { 
			if(ano.equals(null)|| ano.length()==0) {
				this.filmes[posicao-1] = filme;
				return "FILME ADICIONADO";
			}
			this.filmes[posicao-1] = filme;
			return "FILME ADICIONADO";
			
		}
		
		return "FILME JÁ ADICIONADO";
	}
	/**
	 * Checa se o filme já foi adicionado no array de filmes. Comparação feita usando o método equals() da classe Filme. 
	 * @param filme Filme a ser verificado(não pode ser null).
	 * @return True se o filme já estiver na lista, False caso o contrário.
	 */
	private boolean existeFilme(Filme filme) {
		for(int i = 0; i< filmes.length; i++) {
			if(filmes[i] != null && filmes[i].equals(filme)) {
				return true;
			}
		}
		return false;
	}	
	/**
	 * Adiciona um filme em uma posição da hotlist. Se já existir filme na posição, sobrescreve o anterior.
	 * Checa se a posição é válida. Checa se o filme já está na hotlist. 
	 * @param posicao Posição no array de filmes.
	 * @param posicaoHotlist Posição que vai ser colocado na hotlist(1 a 10).
	 * @return String indicando o resultado da operação:
	 * 			- "POSIÇÃO INVÁLIDA" se a posição estiver fora do intervalo de 1-10.
	 * 			- "FILME JÁ ADICIONADO A HOTLIST" se o filme já existe na hotlist.
	 * 			- "ADICIONADO A HOTLIST NA POSICAO posicaoHotlist!" em caso de sucesso.
	 */
	public String cadastraHotlist(int posicao, int posicaoHotlist) {
		if(posicaoHotlist < 1 || posicaoHotlist > 10 || posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		}
		if(!existeHotlist(filmes[posicao-1])) {
			hotlist[posicaoHotlist -1] = filmes[posicao-1]; 
			return "ADICIONADO A HOTLIST NA POSICAO " + posicaoHotlist+ "!";
		}
		return "FILME JÁ ADICIONADO";
	}
	/**
	 * Checa se o filme já foi adicionado no array de hotlist. Comparação feita usando o método equals() da classe Filme. 
	 * @param filme Filme a ser verificado(não pode ser null).
	 * @return True se o filme já estiver na hotlist, False caso contrário.
	 */
	private boolean existeHotlist(Filme filmes) {
		for(int i = 0; i< hotlist.length; i++) {
			if(hotlist[i] != null && hotlist[i].equals(filmes)) {
				return true;
			}
		}
		return false;
	}
	/**
	 * Acessa a lista de filmes mantida e faz uma cópia.
	 * @return A copia da lista de filmes em um array de String.
	 */

	public String[] getHotlist() {
		String[] copiaHotlist = new String[10];
		for(int i = 0; i < hotlist.length; i++) {
			if(hotlist[i]!= null) {
				copiaHotlist[i] = hotlist[i].getNome();
			}
		}
		return copiaHotlist;
	}
	/**
	 * Remove filmes da hotlist. Checa se a posição é nula.
	 * @param posicao Posição do filme na hotlist (1 a 10).
	 * @return True se removeu o filme da hotlist. False caso não exista filme na posição.
	 */
	public boolean removeHotlist(int posicao) {
		if(hotlist[posicao -1]!= null) {
			hotlist[posicao-1].setHot(false);
			hotlist[posicao-1] = null;
			return true;
		}
		return false;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
