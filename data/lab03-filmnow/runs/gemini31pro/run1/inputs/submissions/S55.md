package filmnow;

import java.util.Objects;

/**
 * Representa um filme com nome, ano de lançamento e local de exibição.
 * @author [REDACTED]
 */
public class Filme {
	private String nome;
	private String ano;
	private String local;
	
	 /**
     * Cria um novo filme com as informações fornecidas.
     * 
     * @param nome Nome do filme.
     * @param ano Ano de lançamento do filme.
     * @param local Local onde o filme pode ser assistido.
     */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
	}
	
	public String getNome() {
		return this.nome;
	}
	
	public String getAno() {
		return this.ano;
	}
	
	public String getLocal() {
		return this.local;
	}
	
	/**
     * Retorna uma representação textual do filme.
   	 * Se o ano do filme não for fornecido (ou seja, é uma string vazia), apenas o nome e o local são exibidos. 
   	 * Caso contrário, o nome, o ano e o local são exibidos.
     * @return Uma string representando o filme.
     */
	@Override
	public String toString() {
		if (this.ano.isBlank()) {
			return nome + "\n" + local;
		}
		return nome + ", " + ano + "\n" + local;
	}

	@Override
	public int hashCode() {
		return Objects.hash(ano, local, nome);
	}
	
	/**
     * Compara este filme com o objeto fornecido para verificar se são iguais.
     * Dois filmes são considerados iguais se tiverem o mesmo nome e ano de lançamento. O local não é considerado na comparação de igualdade.
     * @param obj O objeto a ser comparado com este filme.
     * @return {@code true} se o objeto fornecido for um filme com o mesmo nome e ano; {@code false} caso contrário.
     */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		
		Filme outroFilme = (Filme) obj;
		if (outroFilme.ano == this.ano && outroFilme.nome.equals(this.nome))
			return true;
		
		return false;
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes.
 * O usuário pode também fixar 10 filmes em uma lista de destaque. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 101;
	private static final int TAMANHOHOT = 11;
	
	private Filme[] filmes;
	private Filme[] hotList = new Filme[TAMANHOHOT];

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
	}
	
	/**
	 * Retorna a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes;
	}
	
	/**
	 * Retorna a lista de filmes em destaque mantida.
	 * @return O array de hotList.
	 */
	public Filme[] getHotList() {
		return this.hotList;
	}

	/**
	 * Acessa os dados de um filme específico baseado na posição fornecida.
	 * @param posicao Posição do filme no array de filmes.
	 * @return O objeto Filme na posição fornecida, ou null se não houver filme na posição.
	 */
	public Filme getFilme(int posicao) {
		return this.filmes[posicao];
	}

	/**
	 * Adiciona ou atualiza um filme em uma posição específica.
	 * Se já existir um filme na posição, o filme anterior será substituído.
	 * @param posicao Posição onde o filme será adicionado.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assistido.
	 * @return Mensagem indicando o resultado da operação. 
	 *         Retorna "POSICAO INVALIDA" se a posição estiver fora do intervalo permitido.
	 *         Retorna "FILME INVALIDO" se o nome ou local estiverem vazios.
	 *         Retorna "FILME JA ADICIONADO" se o filme já estiver na lista.
	 *         Retorna "FILME ADICIONADO" se o filme for adicionado com sucesso.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if (posicao < 1 || posicao >= this.getTamanhoFilmes()) {
            return "POSICAO INVALIDA";
		}
		
		if (nome.isEmpty() || local.isEmpty()) {
            return "FILME INVALIDO";
        }
		
		Filme filme = new Filme(nome, ano, local);
        if (this.constaFilme(filme)) {
              return "FILME JA ADICIONADO";
        }
        
		this.filmes[posicao] = new Filme(nome, ano, local);
		return "FILME ADICIONADO";
	}
	
	/**
	 * Detalha um filme baseado na posição fornecida.
	 * A posição deve começar em 0 e ser menor que o tamanho total da lista de filmes do sistema.
	 * @param posicao Posição do filme na lista.
	 * @return Detalhes do filme, incluindo um ícone de destaque (🔥) se o filme estiver na hotList.
	 *         Retorna "POSICAO INVALIDA" se a posição estiver fora do intervalo permitido.
	 *         Retorna null se não houver filme na posição.
	 */
	public String detalhaFilme(int posicao) {
		if (posicao < 1 || posicao >= this.getTamanhoFilmes()) {
            return "POSICAO INVALIDA";
		}
		
        Filme filme = this.filmes[posicao];
        if (filme == null) {
        	return null;
        }
        
        if (this.constaHot(filme)) {
        	return "🔥" + filme.toString();
        } else {
        	return filme.toString();
        }
	}
	
	/**
	 * Adiciona um filme em uma posição específica da lista de destaque.
	 * @param posicaoFilme Posição do filme na lista do sistema.
	 * @param posicaoHot Posição na lista de destaque onde o filme será adicionado.
	 * @return Mensagem indicando o resultado da operação.
	 *         Retorna "POSICAO DO FILME INVALIDA" se a posição do filme estiver fora do intervalo permitido.
	 *         Retorna "POSICAO DA HOTLIST INVALIDA" se a posição da hotList estiver fora do intervalo permitido.
	 *         Retorna "FILME INVALIDO" se o filme na posição fornecida for null.
	 *         Retorna "FILME JA ADICIONADO NA HOTLIST" se o filme já estiver na hotList.
	 *         Retorna "FILME ADICIONADO NA POSICAO X" se o filme for adicionado com sucesso na posição X.
	 */
	public String atribuirHot(int posicaoFilme, int posicaoHot) {
		if (posicaoFilme < 1 || posicaoFilme >= this.getTamanhoFilmes()) {
            return "POSICAO DO FILME INVALIDA";
		}
		
		if (posicaoHot < 1 || posicaoHot >= this.getTamanhoHotList()) {
            return "POSICAO DA HOTLIST INVALIDA";
		}
	
		Filme filme = this.getFilme(posicaoFilme);
		if (filme == null) {
			return "FILME INVALIDO";
		}
		
		if (this.constaHot(filme)) {
            return "FILME JA ADICIONADO NA HOTLIST";
        }
		
		hotList[posicaoHot] = filme;
		return "FILME ADICIONADO NA POSICAO " + posicaoHot + "!";
	}
	
	/**
	 * Remove um filme da lista de destaque na posição especificada.
	 * @param posHot Posição do filme na lista de destaque.
	 * @return Mensagem indicando o resultado da operação.
	 *         Retorna "POSICAO DA HOTLIST INVALIDA" se a posição da hotList estiver fora do intervalo permitido.
	 *         Retorna "POSICAO JA VAZIA" se a posição da hotList já estiver vazia.
	 *         Retorna "FILME REMOVIDO" se o filme for removido com sucesso.
	 */
	public String removerHot(int posHot) {
		if (posHot < 1 || posHot >= this.getTamanhoHotList()) {
            return "POSICAO DA HOTLIST INVALIDA";
		}
		
		if (hotList[posHot] == null) {
			return "POSICAO JA VAZIA";
		}
		
		hotList[posHot] = null;
		return "FILME REMOVIDO";
	}
	
	/**
     * Verifica se um filme está na lista de filmes.
     * @param filme Filme a ser verificado.
     * @return {@code true} se o filme está na lista, {@code false} caso contrário.
     */
	public boolean constaFilme(Filme filme) {
		for (Filme jaAdicionado: filmes) {
			if (jaAdicionado != null) {
				if (jaAdicionado.equals(filme)) {
					return true;
				}
			}
		}
		return false;
	}
	
	/**
     * Verifica se um filme está na lista de filmes em destaque.
     * @param filme Filme a ser verificado.
     * @return True se o filme está na lista, False caso contrário.
     */
	public boolean constaHot(Filme filme) {
		for (Filme hot: hotList) {
			if (hot != null) {
				if (hot.equals(filme)) {
					return true;
				}
			}
		}
		return false;
	}
	
	/**
     * Retorna o número máximo de filmes que o sistema pode armazenar.
     * @return O tamanho máximo do array de filmes.
     */
	public int getTamanhoFilmes() {
		return TAMANHO;
	}
	
	/**
     * Retorna o número máximo de filmes na lista de destaque que o sistema pode armazenar.
     * @return O tamanho máximo do array de hotList.
     */
	public int getTamanhoHotList() {
		return TAMANHOHOT;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados - 1;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
