package filmnow;
/**
* Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes.
*
* @author [REDACTED]
*
*/
public class FilmNow {
	
	private static final int TAMANHO = 100;
	
	private Filme[] filmes; //uma Array de filmes do tipo Filme
	private Filme[] hotList;
	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[10];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public String[] getFilmes() {
		String[] out = new String[100];
		
		for (int i=0; i < 100; i++) {
			if (this.filmes[i] != null) {
				out[i] = this.filmes[i].toString();
			}		}
		
		return out;
	}
	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		if (posicao > 0 && posicao <= TAMANHO) {
			return filmes[posicao-1];
		}else {
			return null;
		}
	}
	
	
	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior.
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 * @return
	 */
	public String cadastraFilme(int posicao, String nome, int ano, String local) {
		
		if (posicao <= 0 || posicao > TAMANHO) {
			return "POSIÇÃO INVÁLIDA";
		}
		if (nome == null || nome.trim().isEmpty() || local == null || local.trim().isEmpty() ) {
			return "FILME INVÁLIDO";
		}
		if (verificaFilme(nome, ano)) {
			return "FILME JÁ ADICIONADO";
		}
		this.filmes[posicao-1] = new Filme(nome, ano, local);
		return "FILME ADICIONADO";
	}
	
	/**
	 * Verifica se um filme com o mesmo nome e ano já foi adicionado.
	 * @param nome
	 * @param ano
	 * @return true se o filme já foi adicionado, false caso contrário.
	 */
	private boolean verificaFilme(String nome, int ano) {
		for (Filme f : filmes) {
           if (f != null && f.getName().equals(nome) && f.getAno() == ano) {
               return true;
           }
       }
       return false;
		
	}
	/**
	 * Adiciona um filme da lista principal à hotlist em uma posição específica.
	 * @param posicaoListaOriginal
	 * @param posicaoHot
	 * @return Mensagem indicando o sucesso da operação.
	 */
	
	
	public String adicionarHot(int posicaoOriginal, int posicaoHot) {
	    if (posicaoOriginal <= 0 || posicaoOriginal > TAMANHO || posicaoHot <= 0 || posicaoHot > TAMANHO) {
	        return "POSIÇÃO INVÁLIDA";
	    }
	    Filme filme = filmes[posicaoOriginal - 1];
	    if (filme == null) {
	        return "FILME NÃO ENCONTRADO";
	    }
	    hotList[posicaoHot - 1] = filme;
	    return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!";
	}

	
	/**
	 * Acessa os dados de um filme na hotlist.
	 * @param posicao
	 * @return Dados do filme na hotlist na posição especificada, ou null se não há filme na posição.
	 */
	
	public Filme getFilmeHot(int posicao) {
		if (posicao > 0 && posicao <= 10) {
			return hotList[posicao-1];
		}else {
			return null;
		}
	}
	
	/**
	 * Acessa a lista de filmes na hotlist.
	 * 
	 * @return
	 */
	public String[] getFilmesHot() {
		String[] out = new String[10];
		
		for (int i=0; i < 10; i++) {
			if (this.hotList[i] != null) {
				out[i] = this.hotList[i].toString();
			}		}
		
		return out;
	}
	
	/**
	 * Remove um filme da hotlist em uma posição específica.
	 * @param posicaoHot
	 */
	public void removerHot(int posicaoHot) {
		this.hotList[posicaoHot-1] = null;
		
	}
	/**
	 * Verifica se um filme da lista principal está na hotlist.
	 * @param posicao
	 * @return
	 */
	public boolean ehHot(int posicao) {
	    	for (int i = 0; i < 10; i++) {
	    		if (this.filmes[posicao-1] ==  this.hotList[i]) {
	    			return true;
	    		}
	    	}
	    	return false;
	    }
	
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}
