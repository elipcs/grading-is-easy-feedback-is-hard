package filmnow;

import java.util.Objects;

/**
 * Representa um filme com nome, ano de lançamento e local de lançamento.
 * Dois filmes são considerados iguais se possuem o mesmo nome e ano de lançamento.
 * 
 * @author [REDACTED]
 */
public class Filme {
	
	private String nome;
	private String ano;
	private String localLancamento;

	/**
	 * Constrói um novo objeto Filme.
	 *
	 * @param nome o nome do filme
	 * @param ano o ano de lançamento do filme
	 * @param plataforma o local ou plataforma onde o filme foi lançado
	 */
	public Filme(String nome, String ano, String plataforma) {
		this.nome = nome;
		this.ano = ano;
		this.localLancamento = plataforma;
	}

	public String getNome() {
		return nome;
	}

	public void setNome(String nome) {
		this.nome = nome;
	}

	public String getAno() {
		return ano;
	}

	public void setAno(String ano) {
		this.ano = ano;
	}

	public String getPlataforma() {
		return localLancamento;
	}

	public void setPlataforma(String plataforma) {
		this.localLancamento = plataforma;
	}
	
	/**
	 * Verifica se dois objetos Filme são iguais.
	 * Dois filmes são considerados iguais se têm o mesmo nome e ano de lançamento.
	 *
	 * @param obj o objeto a ser comparado
	 * @return true se os filmes são iguais, false caso contrário
	 */
	@Override
	public boolean equals(Object obj) {
		if (this == obj) {
			return true;
		}
		if (obj == null || getClass() != obj.getClass()) {
			return false;
		}
		Filme filme = (Filme) obj;
		return nome.equals(filme.nome) && ano.equals(filme.ano); 
	}
	
	/**
	 * Retorna o código hash do filme, baseado no nome e no ano de lançamento.
	 *
	 * @return o código hash do filme
	 */
	@Override
	public int hashCode() {
		return Objects.hash(nome, ano);
	}
	
	/**
	 * Retorna a representação em string do filme, que é o nome do filme.
	 *
	 * @return o nome do filme
	 */
	@Override
	public String toString() {
		return this.nome;
	}
}

package filmnow;

/**
 * Sistema que mantém uma lista de filmes prediletos e uma HotList com os filmes selecionados.
 * A lista de filmes pode conter até 100 filmes, e a HotList pode conter até 10 filmes.
 * 
 * @author [REDACTED]
 * @modified por Emerson
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	private Filme[] filmes; // uma representação simplória da lista de filmes
	private static final int TAMANHO_HT = 10;
	private Filme[] hotList; // uma representação simplória da HotList

	/**
	 * Cria um novo sistema FilmNow com capacidade para 100 filmes e uma HotList com capacidade para 10 filmes.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[TAMANHO_HT];
	}
	
	/**
	 * Acessa a lista de filmes mantida pelo sistema.
	 * 
	 * @return Um array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}
	
	/**
	 * Acessa a HotList mantida pelo sistema.
	 * 
	 * @return Um array com os filmes da HotList.
	 */
	public Filme[] getHotList() {
		return this.hotList.clone();
	}

	/**
	 * Acessa os dados de um filme específico na lista de filmes.
	 * 
	 * @param posicao A posição do filme na lista de filmes.
	 * @return O filme na posição especificada, ou null se não houver filme na posição.
	 */
	public Filme getFilme(int posicao) {
		if (!(posicao < 1 || posicao > 100)) {
			return filmes[posicao - 1];
		}
		return null;
	}

	/**
	 * Adiciona um filme em uma posição específica da lista de filmes.
	 * Se já existir um filme na posição, ele será sobrescrito.
	 * 
	 * @param posicao A posição onde o filme será adicionado.
	 * @param nome O nome do filme.
	 * @param ano O ano de lançamento do filme.
	 * @param local O local onde o filme pode ser assistido.
	 * @return Uma mensagem indicando o resultado da operação.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		Filme novoFilme = new Filme(nome, ano, local);
		String output = "";
		if (posicao < 1 || posicao > 100) {
			output += "POSIÇÃO INVÁLIDA";
		}
		else if (nome.isEmpty() || local.isEmpty()) {
			 output += "FILME INVALIDO";
		}
		else if (temFilme(novoFilme)) {
			output += "FILME JÁ ADICIONADO";
		}
		else {
			this.filmes[posicao -1] = novoFilme;
			output += "FILME ADICIONADO";
		}
		return output;
	}
	
	/**
	 * Adiciona um filme à HotList em uma posição específica.
	 * 
	 * @param posFilme A posição do filme na lista de filmes.
	 * @param posHotList A posição na HotList onde o filme será adicionado.
	 * @return Uma mensagem indicando o resultado da operação.
	 */
	public String cadastraHotList(int posFilme, int posHotList) {
		String saida = "";
		Filme filme = getFilme(posFilme);
		
		if(posFilme < 1 || posFilme > 100 || posHotList < 1 || posHotList > 10) {
			saida = "POSIÇÃO INVÁLIDA";
		}
		else if (filme == null) {
			return "FILME NAO EXISTE";
		}
		else if (temFilmeHt(filme)) {
			saida = "FILME JÁ CADASTRADO NA HOTLIST";
		}
		else {
			this.hotList[posHotList -1] = filme;
			saida = "ADICIONADO À HOTLIST NA POSIÇÃO " + posHotList + "!";
		}
		return saida;
	}
	
	/**
	 * Detalha as informações de um filme em uma posição específica na lista de filmes.
	 * Se o filme estiver na HotList, ele será marcado com um emoji de fogo.
	 * 
	 * @param pos A posição do filme na lista de filmes.
	 * @return Uma string contendo os detalhes do filme, ou uma mensagem de erro.
	 */
	public String detalhaFilme(int pos) {
		String strFilme = "";
		Filme filme = getFilme(pos);
		if (pos > 0 && pos < 101 && filme == null) {
			return "";
		}
		if (pos < 1 || pos > 100) {
			return "POSIÇÃO INVÁLIDA";
		}
		if (temFilmeHt(filme)) {
			strFilme += "🔥 ";
		}
		if (!filme.getAno().isEmpty()) {
			strFilme += filme.getNome() + ", " + filme.getAno() + "\n" + filme.getPlataforma();
		} else {
			strFilme += filme.getNome() + "\n" + filme.getPlataforma();
		}
		return strFilme;
	}
	
	/**
	 * Exibe as informações de um filme em uma posição específica na HotList.
	 * 
	 * @param pos A posição do filme na HotList.
	 * @return Uma string contendo o nome e o ano do filme, ou uma string vazia se não houver filme na posição.
	 */
	public String exibeFilmeHot(int pos) {
		if (this.hotList[pos - 1] == null) {
			return "";
		}
		return this.hotList[pos - 1].getNome() + ", " + this.hotList[pos - 1].getAno();
	}
	
	/**
	 * Remove um filme da HotList em uma posição específica.
	 * 
	 * @param pos A posição do filme na HotList.
	 * @return Uma mensagem indicando o resultado da operação.
	 */
	public String removeHotList(int pos) {
		if (pos < 1 || pos > 10) {
			return "POSIÇÃO INVÁLIDA";
		}
		else if(this.hotList[pos - 1] == null) {
			return "FILME NÃO ESTÁ NA HOTLIST";
		}
		else {
			this.hotList[pos -1] = null;
			return "REMOVIDO DA HOTLIST";
		}
	}
	
	/**
	 * Verifica se um filme já está presente na HotList.
	 * 
	 * @param filme O filme a ser verificado.
	 * @return true se o filme já estiver na HotList, false caso contrário.
	 */
	private boolean temFilmeHt(Filme filme) {
		for (Filme film : hotList) {
			if (filme.equals(film)) {
				return true;
			}
		}
		return false;
	}
	
	/**
	 * Verifica se um filme já está presente na lista de filmes.
	 * 
	 * @param filme O filme a ser verificado.
	 * @return true se o filme já estiver na lista, false caso contrário.
	 */
	private boolean temFilme(Filme filme) {
		for (Filme film : filmes) {
			if (filme.equals(film)) {
				return true;
			}
		}
		return false;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

						"(D)Detalhar filme\n" + 
						"(E)Exibir filmes hot\n" +
						"(H)Adicionar hotlist\n" +
						"(R)Remover hot\n" + 
						"(S)air\n" + 
						"\n" +
