package filmnow;

// Definição da classe FilmNow
public class FilmNow {
    // Tamanho máximo da lista de filmes
    private static final int TAMANHO = 100;

    // Array para armazenar os filmes
    private Filme[] filmes;

    // Array para armazenar a lista de filmes em alta (HotList)
    private Filme[] hotList = new Filme[10];

    // Construtor da classe FilmNow
    public FilmNow() {
        this.filmes = new Filme[TAMANHO]; // Inicialização da lista de filmes com tamanho padrão
    }

    // Método para obter um filme na posição especificada
    public Filme getFilme(int posicao) {
        if (posicao >= 1 && posicao <= TAMANHO) {
            return filmes[posicao - 1]; // Retorna o filme na posição especificada
        }
        return null; // Retorna null se a posição estiver fora do intervalo válido
    }

    // Método para cadastrar um filme em uma posição específica
    public boolean cadastraFilme(int posicao, Filme filme) {
        if (posicao < 1 || posicao > TAMANHO) {
            return false; // Retorna false se a posição estiver fora do intervalo válido
        }
        filmes[posicao - 1] = filme; // Adiciona o filme na posição especificada
        return true; // Retorna true após o cadastro bem-sucedido
    }

    // Método para obter uma cópia da lista de filmes
    public Filme[] getFilmes() {
        return filmes.clone(); // Retorna uma cópia da lista de filmes
    }

    // Método para adicionar um filme à lista de filmes em alta (HotList)
    public void adicionarHot(int posicao, Filme filme) {
        if (posicao >= 1 && posicao <= 10) {
            // Verifica se o filme já está na HotList
            boolean filmeJaNaHotList = false;
            for (Filme hot : hotList) {
                if (hot != null && hot.equals(filme)) {
                    filmeJaNaHotList = true;
                    break;
                }
            }

            if (!filmeJaNaHotList) {
                // Remove o filme de sua posição anterior na HotList, se estiver presente
                for (int i = 0; i < hotList.length; i++) {
                    if (hotList[i] != null && hotList[i].equals(filme)) {
                        hotList[i] = null;
                        break;
                    }
                }

                // Adiciona o filme na posição especificada na HotList
                hotList[posicao - 1] = filme;
                filme.setHot(true);
                System.out.println("ADICIONADO À HOTLIST NA POSIÇÃO " + posicao + "!");
            } else {
                System.out.println("O filme já está na HotList em outra posição!");
            }
        }
    }

    // Método para remover um filme da lista de filmes em alta (HotList)
    public void removerHot(int posicao) {
        if (posicao >= 1 && posicao <= 10 && hotList[posicao - 1] != null) {
            hotList[posicao - 1].setHot(false); // Define que o filme não está mais em alta
            hotList[posicao - 1] = null; // Remove o filme da posição especificada na HotList
        }
    }

    // Método para obter uma cópia da lista de filmes em alta (HotList)
    public Filme[] getHotList() {
        return hotList.clone(); // Retorna uma cópia da lista de filmes em alta (HotList)
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

// Classe responsável por carregar os dados de filmes a partir de um arquivo CSV
public class LeitorFilmNow {

    // Constantes para as colunas no arquivo CSV
    private static final int COLUNA_POSICAO = 0;
    private static final int COLUNA_NOME = 1;
    private static final int COLUNA_ANO = 2;
    private static final int COLUNA_LOCAL = 3;

    // Método para carregar os filmes a partir de um arquivo CSV
    public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
        int carregados = 0;

        try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
            String linha;
            // Percorre cada linha do arquivo CSV
            while ((linha = br.readLine()) != null) {
                carregados += 1;
                if (carregados == 1) {
                    continue; // Ignora a primeira linha (cabeçalho)
                }
                String[] campos = linha.split(",");
                processaLinhaCsvFilmes(campos, fn); // Processa os campos de cada linha
            }
        }
        
        return carregados - 1; // Desconta a linha de cabeçalho
    }

    // Método para processar os campos de uma linha do arquivo CSV e cadastrar o filme
    private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
        int posicao = Integer.parseInt(campos[COLUNA_POSICAO]); // Obtém a posição do filme
        String nome = campos[COLUNA_NOME].trim(); // Obtém o nome do filme
        String ano = campos[COLUNA_ANO].trim(); // Obtém o ano de lançamento do filme
        String local = campos[COLUNA_LOCAL].trim(); // Obtém o local de filmagem do filme

        Filme filme = new Filme(nome, ano, local); // Cria um novo objeto Filme
        fn.cadastraFilme(posicao, filme); // Cadastra o filme na instância de FilmNow
    }
}

    }

     * Método para processar a escolha do usuário

     * @param opcao   Opção escolhida pelo usuário

     * @param scanner Objeto Scanner para entrada de dados

                mostrarFilmes(fn); // Mostra todos os filmes cadastrados

                break;
            case "E":
                exibirHotList(fn); // Exibe a lista de filmes em alta (HotList)
                break;
            case "H":
                adicionarHot(fn, scanner); // Adiciona um filme à HotList
                break;
            case "R":
                removerHot(fn, scanner); // Remove um filme da HotList
                break;
            default:

     * @param fn      Instância da classe FilmNow
