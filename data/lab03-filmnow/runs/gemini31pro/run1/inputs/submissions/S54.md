package filmnow;

import java.util.Arrays;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	
	private String[] filmes, locais; //uma representacao simploria da lista de filmes
	private int[] anos;
	private boolean[] hot;
	
	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new String[TAMANHO];
		this.anos = new int[TAMANHO];
		this.locais = new String[TAMANHO];
		this.hot = new boolean[TAMANHO];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public String[] getFilmes() {
		return this.filmes.clone();
	}
	
	/**
	 * Acessa a lista de anos mantida.
	 * @return O array de anos.
	 */
	public int[] getAnos() {
		return this.anos.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public String getFilme(int posicao) {
		if (filmes[posicao] != null) {
			if (hot[posicao]) {
				return "🔥 " + filmes[posicao] + ", " + anos[posicao] + "\n" + locais[posicao];
			} else {
				return filmes[posicao] + ", " + anos[posicao] + "\n" + locais[posicao];
			}
		}
		return null;
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public void cadastraFilme(int posicao, String nome, int ano, String local) {
		this.filmes[posicao] = nome;
		this.anos[posicao] = ano;
		this.locais[posicao] = local;
	}
	/**
	 * Atribui hot. 
	 * @param posicao.
	 */
	public void atribuirHot(int posicao) {
		this.hot[posicao] = true;
	}
	/**
	 * Remove hot. 
	 * @param posicao.
	 */
	public void removeHot(int posicao) {
		this.hot[posicao] = false;
	}
}

package filmnow;

/**
 * Sistema que mantém os filmes hot. Podem existir 10 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class Hotlist {
	private static final int TAMANHOHOT = 10;
	
	private int[] filmeshot;
	
	/**
	 * Cria a Hotlist.
	 */
	public Hotlist() {
		this.filmeshot = new int[TAMANHOHOT];
		for (int i = 0; i < this.filmeshot.length; i++) {
			this.filmeshot[i] = -1; //Valor para indicar que não há um filme na posição.
		}
	}
	
	/**
	 * Acessa a hotlist.
	 * @return O array da hotlist.
	 */
	public int[] getHotlist() {
		return this.filmeshot.clone();
	}
	/**
	 * Adiciona filme a hotlist.
	 */
	public void cadastraHot(FilmNow fn, int posicaohotlist, int posicaofilmes) {
		if (this.filmeshot[posicaohotlist] != -1) {
			removeHotlist(fn, posicaohotlist);
		}
		this.filmeshot[posicaohotlist] = posicaofilmes;
	}
	/**
	 * Remove filme da hotlist.
	 */
	public void removeHotlist(FilmNow fn, int posicaohotlist) {
		fn.removeHot(this.filmeshot[posicaohotlist]);
		this.filmeshot[posicaohotlist] = -1;
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Arrays;
import java.util.Scanner;

/**
 * Interface com menus texto para manipular o sistema FilmNow.
 * 
 * @author [REDACTED]
 *
 */
public class MainFilmNow {

	public static void main(String[] args) {
		FilmNow fn = new FilmNow();
		Hotlist hl = new Hotlist();

		System.out.println("Carregando filmes ...");
		try {
			/*
			 * Essa é a maneira de lidar com possíveis erros por falta do arquivo. 
			 */
			carregaFilmes("filmes_inicial.csv", fn);
		} catch (FileNotFoundException e) {
			System.err.println("Arquivo não encontrado: " + e.getMessage());
		} catch (IOException e) {
			System.err.println("Erro lendo arquivo: " + e.getMessage());
		}

		Scanner scanner = new Scanner(System.in).useDelimiter("\n");
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner, hl);
		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */
	
	private static String menu(Scanner scanner) {
		System.out.println(
				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" + 
						"(E)Exibir HotList\n" +
						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)air\n" + 
						"\n" + 
						"Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opcao   Opção digitada.
	 * @param fn O sistema FilmNow que estamos manipulando.
	 * @param scanner Objeto scanner para o caso do comando precisar de mais input.
	 * @param hl 
	 */
	private static void comando(String opcao, FilmNow fn, Scanner scanner, Hotlist hl) {
		switch (opcao) {
		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			mostrarHotlist(hl, fn);
			break;
		case "H":
			atribuirHot(fn, hl, scanner);
			break;
		case "R":
			removerHot(fn, hl, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("Opção inválida!");
		}
	}

	/**
	 * Imprime lista de filmes.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 */
	private static void mostrarFilmes(FilmNow fn) {

		String[] filmes = fn.getFilmes();
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				System.out.println(formataFilme(i, filmes[i]));
			}
		}
	}

	/**
	 * Imprime os detalhes de um dos filmes. 
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar qual contato.
	 */
	private static void detalharFilme(FilmNow fn, Scanner scanner) {
		System.out.print("\nPosição> ");
		int posicao = scanner.nextInt();
		String filme = fn.getFilme(posicao);
		if (filme != null) {
			System.out.println(filme);
		} else {
			System.out.println("POSIÇÃO INVÁLIDA");
		}
	}
	/**
	 * Imprime a hotlist.
	 * 
	 * @param hl para a hotlist e fn para a lista geral de filmes.
	 */
	private static void mostrarHotlist(Hotlist hl, FilmNow fn) {

		int[] filmeshot = hl.getHotlist();
		for (int i = 0; i < filmeshot.length; i++) {
			if (filmeshot[i] != -1) {
				String[] filmes = fn.getFilmes();
				System.out.println(formataFilme(i, filmes[filmeshot[i]]));
			}
		}
	}
	/**
	 * Atribui "hot" a um filme
	 * 
	 * @param Filme a atribuir e posição na hotlist.
	 */
	private static void atribuirHot(FilmNow fn, Hotlist hl,Scanner scanner) {
		System.out.print("\nFilme> ");
		String filme = scanner.next();
		int[] filmeshot = hl.getHotlist();
		boolean continuar = true;
		for (int i = 0; i < filmeshot.length; i++) {
			if (filmeshot[i] == Integer.parseInt(filme)) {
				System.out.println("FILME JÁ ADICIONADO");
				continuar = false;
				break;
			}
		}
		if (continuar) {
			System.out.print("\nPosicao> ");
			String pos = scanner.next();
			fn.atribuirHot(Integer.parseInt(filme));
			hl.cadastraHot(fn, Integer.parseInt(pos), Integer.parseInt(filme));
			System.out.println("ADICIONADO À HOTLIST NA POSIÇÃO " + pos + "!");
		}
	}
	
	/**
	 * Remove "hot" de um filme
	 * 
	 * @param Filme a remover hot.
	 */
	private static void removerHot(FilmNow fn, Hotlist hl, Scanner scanner) {
		System.out.print("\nPosicao> ");
		String pos = scanner.next();
		fn.removeHot(Integer.parseInt(pos));
		hl.removeHotlist(fn, Integer.parseInt(pos));
	}
	

	/**
	 * Formata um filme para impressão. 
	 * 
	 * @param posicao A posição do filme (que é exibida)/
	 * @param filme O filme a ser impresso.
	 * @return A String formatada.
	 */
	private static String formataFilme(int posicao, String filme) {
		return posicao + " - " + filme;
	}

	/**
	 * Cadastra um filme no sistema. 
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para pedir informações do contato.
	 */
	private static void adicionaFilme(FilmNow fn, Scanner scanner) {
		System.out.print("\nPosição no sistema> ");
		String posicao = scanner.next();
		if ((Integer.parseInt(posicao) >= 1) && (Integer.parseInt(posicao) <= 100)) {
			System.out.print("\nNome> ");
			String nome = scanner.next();
			if (nome.equals("") == false) {
				System.out.print("\nAno> ");
				String ano = scanner.next();
				boolean filmeadicionado = false;
				String[] filmesarr = fn.getFilmes();
				int[] anosarr = fn.getAnos();
				for (int i = 0; i < 100; i++) {
					if (filmesarr[i] != null) {
						if ((filmesarr[i].equals(nome)) && (anosarr[i] == Integer.parseInt(ano))){
							filmeadicionado = true;
							break;
						}
					}
				}
				if (filmeadicionado == false) {
					System.out.print("\nLocal> ");
					String local = scanner.next();
					if (local.equals("") == false) {
						System.out.println(nome);
						fn.cadastraFilme(Integer.parseInt(posicao), nome, Integer.parseInt(ano), local);
						System.out.println("FILME ADICIONADO");
					} else {
						System.out.println("FILME INVALIDO");
					}
				} else {
					System.out.println("FILME JA ADICIONADO");
				}
			} else {
				System.out.println("FILME INVALIDO");
			}
		} else {
			System.out.println("POSIÇÃO INVÁLIDA!");
		}
	}

	/**
	 * Sai da aplicação.
	 */
	private static void sai() {
		System.out.println("\nVlw flw o/");
		System.exit(0);
	}

	/**
	 * Lê carga de filmes de um arquivo csv. 
	 * 
	 * @param arquivoFilmes O caminho para o arquivo.
	 * @param fn O sistema FilmNow a ser populado com os dados.
	 * @throws IOException Caso o arquivo não exista ou não possa ser lido.
	 */
	private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		LeitorFilmNow leitor = new LeitorFilmNow();
		
		int carregados =  leitor.carregaContatos(arquivoFilmes, fn);
		System.out.println("Carregamos " + carregados + " registros.");
	}
}
