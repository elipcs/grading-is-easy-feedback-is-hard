package filmnow;

import java.util.Objects;

/**
 * Representa um filme com nome, ano de lançamento e local de origem.
 * Dois filmes são considerados iguais se possuírem o mesmo nome e ano.
 */
public class Filme {
    /** Nome do filme */
    String nome;

    /** Ano de lançamento do filme */
    Integer ano;

    /** Local de origem do filme (por exemplo, país ou cidade) */
    String local;

    /**
     * Constrói um objeto Filme com nome, ano e local.
     * 
     * @param nome  o nome do filme
     * @param ano   o ano de lançamento do filme
     * @param local o local de origem do filme
     */
    public Filme(String nome, Integer ano, String local) {
        this.nome = nome;
        this.ano = ano;
        this.local = local;
    }

    /**
     * Retorna uma representação textual simples do filme.
     * 
     * @return String com nome, ano e local concatenados
     */
    @Override
    public String toString() {
        return nome + ano + local;
    }

    /**
     * Retorna o nome do filme.
     * 
     * @return o nome do filme
     */
    public String getNome() {
        return nome;
    }

    /**
     * Retorna o ano de lançamento do filme.
     * 
     * @return o ano do filme
     */
    public int getAno() {
        return ano;
    }

    /**
     * Retorna o local de origem do filme.
     * 
     * @return o local do filme
     */
    public String getLocal() {
        return local;
    }

    /**
     * Retorna o código hash do filme, baseado no nome e ano.
     * 
     * @return o hash code calculado
     */
    @Override
    public int hashCode() {
        return Objects.hash(ano, nome);
    }

    /**
     * Compara dois filmes para verificar igualdade com base em nome e ano.
     * 
     * @param obj o objeto a ser comparado
     * @return true se forem filmes com mesmo nome e ano, false caso contrário
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (obj == null)
            return false;
        if (getClass() != obj.getClass())
            return false;
        Filme other = (Filme) obj;
        return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);
    }

    /**
     * Retorna uma descrição detalhada do filme no formato:
     * Nome, Ano<br>Local
     * 
     * @return descrição detalhada do filme
     */
    public String stringDetalha() {
        return nome + ", " + ano + "\n" + local;
    }
}

package filmnow;

public class FilmNow {
	/**
	 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
	 * 
	 * @author [REDACTED]
	 *
	 */
		private static final int TAMANHO = 100;
		
		private static final int TAMANHO_HOTLIST = 10;
		private Filme[] hotList = new Filme[TAMANHO_HOTLIST];
		
		private Filme[] filmes;//uma representacao simploria da lista de filmes
		/**
		 * Cria o FilmNow.
		 */
		public FilmNow() {
			this.filmes = new Filme[TAMANHO];
		}
		
		/**
		 * 
		 * Acessa a lista de filmes mantida.
		 * @return O array de filmes.
		 */
		public String getFilmes() {
			String saida = "";
			for (int i = 0; i < filmes.length; i++) {
				if (filmes[i] != null) {
					saida += (i+1) + " - " + filmes[i].getNome() + "\n";
				}
			}
			return saida;
			
		}

		/**
		 * Acessa os dados de um filme específico.
		 * @param posicao Posição do filme no sistema.
		 * @return Dados do filme. Null se não há filme na posição.
		 */
		public Filme getFilme(int posicao) {
		    if (!posicaoValida(posicao)) return null;
		    return filmes[posicao - 1];
		}

		/**
		 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
		 * @param posicao Posição do filme.
		 * @param nome Nome do filme.
		 * @param ano Ano de lançamento do filme.
		 * @param local Local onde o filme pode ser assitido.
		 */
		public String cadastraFilme(int posicao, String nome, int ano, String local) {
			if (!posicaoValida(posicao)) {
				return "POSIÇÃO INVÁLIDA";
			}
			
			if (campoInvalido(nome) || campoInvalido(local)) {
				return "FILME INVALIDO";
			}
			
			if (filmeRepetido(nome, ano)) {
				return "FILME JA ADICIONADO";
			}
			this.filmes[posicao - 1] = new Filme(nome, ano, local);
			return "FILME ADICIONADO";
			
		}
		
		private boolean posicaoValida(int posicao) {
			return posicao > 0 && posicao <= 100;
		}
		
		private boolean filmeRepetido(String nome, int ano) {
			for (Filme f : filmes) {
				if (f != null && f.equals(new Filme(nome, ano, ""))) {
					return true;
				}
			}
			return false;
		}
		
		private boolean campoInvalido(String campo) {
			return campo == null || campo.trim().isEmpty();
			}
		
		public String detalharFilme(int posicao) {
			if (posicao < 1 || posicao > 100 || filmes[posicao - 1] == null) {
				return "POSIÇÃO INVÁLIDA!";
				
			}
			
			Filme filme = filmes[posicao - 1];
			boolean isHot = false;
			
			for (Filme f : hotList) {
				if (f != null && f.equals(filme)) {
					isHot = true;
					break;
				}
			}
			return (isHot ? "🔥 " : "") + filme.stringDetalha();
		}
		
		public String adicionarHot(int posicaoFilme, int posicaoHot) {
			if (!posicaoValida(posicaoFilme) || posicaoHot < 1 || posicaoHot > 10) {
				return "POSIÇÃO INVÁLIDA";
			}
			
			Filme filme = filmes[posicaoFilme - 1];
			if (filme == null) return "FILME INEXISTENTE!";
			
			for (Filme f : hotList) {
				if (f != null && f.equals(filme)) {
					return "FILME JÁ ESTÁ NA HOTLIST!";
				}
			}
			
			hotList[posicaoHot - 1] = filme;
			return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!";
		}
		
		public String removeHot(int posicaoHot) {
			if (posicaoHot < 1 || posicaoHot > 10) {
				return "POSIÇÃO INVÁLIDA";
			}

			
			hotList[posicaoHot - 1] = null;
			return "REMOVIDO DA HOTLIST!";
		}
		
		public String exibirHotList() {
			String resultado = "";
			for (int i = 0; i < hotList.length; i++) {
				if (hotList[i] != null) {
					 resultado += (i + 1) + " - " + hotList[i].getNome() + ", " + hotList[i].getAno() + "\n";
				}
			}
		    return resultado.isEmpty() ? "HOTLIST VAZIA!" : resultado;
		    
		}
	}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}

package filmNow;

import static org.junit.jupiter.api.Assertions.*;
