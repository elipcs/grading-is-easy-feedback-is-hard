package filmnow;

import java.util.Objects;

/**
 * Representa um filme com nome, ano, posição e plataforma.
 */
public class Filme {
    private String nome;
    private Integer ano;
    private int posicao;
    private String plataforma;

    /**
     * Cria um filme com nome, ano, posição e plataforma.
     */
    public Filme(String nome, Integer ano, int posicao, String plataforma) {
        this.nome = nome;
        this.ano = ano;
        this.posicao = posicao;
        this.plataforma = plataforma;
    }

    /**
     * Retorna o nome do filme.
     */
    public String getNome() {
        return nome;
    }

    /**
     * Retorna o ano do filme.
     */
    public Integer getAno() {
        return ano;
    }

    /**
     * Retorna a plataforma do filme.
     */
    public String getPlataforma() {
        return plataforma;
    }

    /**
     * Retorna o nome do filme.
     */
    @Override
    public String toString() {
        return nome;
    }

    /**
     * Retorna nome e ano.
     */
    public String detalharFilme() {
        return nome + ", " + ano;
    }

    /**
     * Retorna nome, ano e plataforma.
     */
    public String detalhes() {
        return nome + ", " + ano + "\n" + plataforma;
    }

    /**
     * Gera o hash do filme.
     */
    @Override
    public int hashCode() {
        return Objects.hash(ano, nome);
    }

    /**
     * Compara dois filmes por nome e ano.
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (obj == null || getClass() != obj.getClass())
            return false;
        Filme other = (Filme) obj;
        return Objects.equals(nome, other.nome) && Objects.equals(ano, other.ano);
    }
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	
	private static final int TAMANHO_HOTLIST = 10;
	
	private Filme[] filmes;
	private Filme[] filmesHot;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.filmesHot = new Filme[TAMANHO_HOTLIST];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
	    if (posicao >= 0 && posicao < filmes.length && 
	    	filmes[posicao] != null) {
	        return filmes[posicao];
	    }
	    return null;
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, Integer ano, String local) {

	    // Valida posição
	    if (posicao < 1 || posicao > 100) {
	        return "POSIÇÃO INVÁLIDA";
	    }

	    // Valida nome
	    if (nome == null || nome.trim().isEmpty()) {
	        return "FILME INVÁLIDO";
	    }

	    // Valida local
	    if (local == null || local.trim().isEmpty()) {
	        return "FILME INVÁLIDO";
	    }

	    // Verifica se filme já existe
	    Filme novoFilme = new Filme(nome, ano, posicao, local);

	    for (Filme f : filmes) {
	        if (f != null && f.equals(novoFilme)) {
	            return "FILME JÁ ADICIONADO";
	        }
	    }

	    // Cria e armazena o filme
	    Filme filme = new Filme(nome, ano, posicao, local);
	    filmes[posicao - 1] = filme; 
	    return "FILME ADICIONADO";
	}

	/**
	 * Define um filme na hotlist.
	 * @param posicao Posição na hotlist.
	 * @param filme Filme a ser inserido.
	 */
	public void setFilmeHot(int posicao, Filme filme) {
	    if (posicao >= 0 && posicao < filmesHot.length) {
	        filmesHot[posicao] = filme;
	    }
	}

	/**
	 * Retorna os filmes da hotlist.
	 * @return Array de filmes da hotlist.
	 */
	public Filme[] getFilmesHot() {
		return filmesHot.clone();
	}

	/**
	 * Verifica se um filme está na hotlist.
	 * @param filme Filme a ser verificado.
	 * @return True se estiver, false se não.
	 */
	public boolean filmeEstaNaHotList(Filme filme) {
	    for (Filme f : filmesHot) {
	        if (f != null && f.equals(filme)) {
	            return true;
	        }
	    }
	    return false;
	}

	/**
	 * Verifica se há filme em certa posição da hotlist.
	 * @param posicaoHot Posição a verificar.
	 * @return True se houver filme.
	 */
	public boolean temPosicaoNaHotList(int posicaoHot) {
	    return posicaoHot >= 0 && 
	    	   posicaoHot < filmesHot.length && 
	    	   filmesHot[posicaoHot] != null;
	}
		
	/**
	 * Exibe detalhes do filme da posição indicada.
	 * @param posicao Posição do filme.
	 * @return Detalhes do filme.
	 */
	public String exibirDetalhes(int posicao) {
	    Filme filme = getFilme(posicao);
	    
	    if (filme == null) {
	        return "POSIÇÃO INVÁLIDA!";
	    }

	    String detalhes = filme.detalhes();

	    if (filmeEstaNaHotList(filme)) {
	        detalhes = "🔥 " + detalhes;
	    }

	    return detalhes;
	}

	/**
	 * Remove um filme da hotlist.
	 * @param posicao Posição a ser removida.
	 */
	public void removeFilmeHot(int posicao) {
	    if (posicao >= 0 && posicao < filmesHot.length) {
	        filmesHot[posicao] = null;
	    }
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}

		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}
	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.

		System.out.println(
				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" + 
						"(E)Exibir HotList\n" +
						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)Sair\n" + 
						"\n" + 
						"Opção> ");
		return scanner.nextLine().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * @param opcao Opção digitada.
	 * @param fn Sistema FilmNow.
	 * @param scanner Scanner para entrada.
	 */
	private static void comando(String opcao, FilmNow fn, Scanner scanner) {
		switch (opcao) {
		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibirHotList(fn);
			break;
		case "H":
			atribuirHot(fn, scanner);
	        break;
		case "R":
			removerHot(fn, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("Opção inválida!");
		}
	}

	/**
	 * Imprime lista de filmes.
	 * @param fn Sistema FilmNow.
	 */
	private static void mostrarFilmes(FilmNow fn) {
		Filme[] filmes = fn.getFilmes();
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				System.out.println(formataFilme(i, filmes[i]));
			}
		}
	}

		Filme[] filmes = fn.getFilmesHot();
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				System.out.println(formataFilmeHot(i, filmes[i]));
			} 
		}
	}

	 * Exibe detalhes de um filme.

	 * @param filme Filme a ser exibido.
	 * @return String formatada.

	 * @param fn Sistema FilmNow.

	        System.out.println("POSIÇÃO INVÁLIDA");
	        return;

	    int posicao = Integer.parseInt(entradaPosicao);

	        entradaAno = "0";

	    String local = scanner.nextLine();

	    String resultado = fn.cadastraFilme(posicao, nome, ano, local);
	    System.out.println(resultado);
