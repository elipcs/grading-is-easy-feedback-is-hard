                this.ano = Ano;
                this.local = Local;

                return this.nome;
        }

        public String getAno() {
                return this.ano;
        }
       
		@Override
		public int hashCode() {
			return Objects.hash(ano, nome);
		}
		@Override
		public boolean equals(Object obj) {
			if (this == obj)
				return true;
			if (obj == null)
				return false;
			if (getClass() != obj.getClass())
				return false;
			Filme other = (Filme) obj;
			return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);
		}

                return this.local;

        		return nome + "\n" + local;
        	}
                return nome + ", " + ano + "\n" + local;
        }

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Suporta até 100 filmes e uma
 * hotlist com 10 filmes destacados.
 * 
 * Cada filme é representado por um objeto da classe {@link Filme}.
 * 
 * A classe permite cadastrar filmes, exibir todos os filmes, exibir uma hotlist
 * destacada, detalhar filmes específicos e gerenciar a hotlist.
 * 
 * @author [REDACTED]
 */
public class FilmNow {

	private static final int TAMANHO = 100;

	/**
	 * Lista de filmes. 
	 */
	private Filme[] filmes;
	/**
	 * Lista de filmes favoritados (hot).
	 */
	private Filme[] hotList; // Lista de filmes destacados (hot)
	/**
	 * Variavel que armazena o indice de um filme procurado pelo metodo verificaFilmeHot.
	 */
	private int hotPosicao;

	/**
	 * Constrói um novo sistema FilmNow com capacidade para 100 filmes e 10 filmes
	 * na hotlist.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[10];
		this.hotPosicao = 0;
	}

	/**
	 * Retorna uma cópia da lista de filmes cadastrados.
	 *
	 * @return Um array de filmes (possivelmente com posições nulas).
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Retorna o filme em uma determinada posição.
	 *
	 * @param posicao A posição (índice) do filme no array interno.
	 * @return O filme na posição especificada, ou {@code null} se estiver vazia.
	 */
	public Filme getFilme(int posicao) {
		return filmes[posicao];
	}

	/**
	 * Retorna uma cópia da lista de filmes na hotlist.
	 *
	 * @return Um array com os filmes destacados (hot).
	 */
	public Filme[] getFilmesHot() {
		return this.hotList.clone();
	}

	/**
	 * Cadastra um novo filme em uma posição da lista principal. Se já houver filme
	 * na posição, ele será sobrescrito.
	 *
	 * @param posicao A posição desejada (1 a 100).
	 * @param nome    O nome do filme.
	 * @param ano     O ano de lançamento.
	 * @param local   Onde pode ser assistido.
	 * @return Uma mensagem indicando o resultado da operação.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if (posicao < 1 || posicao > TAMANHO) {
			return "POSIÇÃO INVÁLIDA";

		} else if (nome == null || local == null || nome.trim().isEmpty() || local.trim().isEmpty()) {
			return "FILME INVALIDO";

		} else {
			Filme filme = new Filme(nome, ano, local);
			for (int i = 0; i < filmes.length; i++) {
				if (filmes[i] != null && filmes[i].equals(filme)) {
					return "FILME JÁ ADICIONADO";
				}
			}

			if (filmes[posicao - 1] != null) {
				if (verificaFilmeHot(filmes[posicao - 1])) {
					removerFilmeHot(hotPosicao + 1);
				}
			}

			this.filmes[posicao - 1] = filme;
			return "FILME ADICIONADO";
		}
	}

	/**
	 * Exibe a lista de filmes cadastrados, com sua posição e nome.
	 *
	 * @return Uma String formatada com os filmes. Retorna uma string vazia se não
	 *         houver filmes.
	 */
	public String mostraFilmes() {
		Filme[] filmes = getFilmes();
		StringBuilder sb = new StringBuilder();

		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				sb.append(formataFilme(i, filmes[i])).append("\n");
			}
		}
		return sb.toString().trim();
	}

	/**
	 * Exibe a lista de filmes na hotlist (filmes destacados).
	 *
	 * @return Uma String com os filmes hot formatados. Retorna uma string vazia se
	 *         não houver hotlist.
	 */
	public String mostraFilmesHot() {
		Filme[] filmes = getFilmesHot();
		StringBuilder sb = new StringBuilder();

		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				sb.append(formataFilme(i, filmes[i])).append("\n");
			}
		}
		return sb.toString().trim();
	}

	/**
	 * Formata um filme para exibição com sua posição.
	 *
	 * @param posicao A posição (índice + 1).
	 * @param filme   O filme a ser formatado.
	 * @return Uma String no formato "posicao - nome".
	 */
	private static String formataFilme(int posicao, Filme filme) {
		return (posicao + 1) + " - " + filme.getNome();
	}

	/**
	 * Exibe os detalhes de um filme em uma posição. Se for um filme da hotlist,
	 * inclui 🔥 na frente.
	 *
	 * @param posicao A posição (1 a 100) do filme.
	 * @return Uma String com os detalhes, ou "POSIÇÃO INVÁLIDA"/vazio se inválido.
	 */
	public String detalharFilme(int posicao) {
		if (posicao < 1 || posicao > TAMANHO) {
			return "POSIÇÃO INVÁLIDA";
		}
		Filme filme = getFilme(posicao - 1);
		if (filme == null) {
			return "";
		}
		if (verificaFilmeHot(filme)) {
			return "🔥 " + filme.toString();
		}
		return filme.toString();
	}

	/**
	 * Verifica se um filme está na hotlist.
	 *
	 * @param filme O filme a ser verificado.
	 * @return {@code true} se estiver na hotlist, senão {@code false}.
	 */
	public boolean verificaFilmeHot(Filme filme) {
		for (int i = 0; i < hotList.length; i++) {
			if (hotList[i] != null && hotList[i].equals(filme)) {
				hotPosicao = i;
				return true;
			}
		}
		return false;
	}

	/**
	 * Adiciona um filme da lista principal na hotlist em uma posição específica.
	 *
	 * @param posicaoAtual A posição (1 a 100) do filme na lista principal.
	 * @param posicaoHot   A posição (1 a 10) desejada na hotlist.
	 * @return Mensagem indicando sucesso ou erro.
	 */
	public String addFilmeHot(int posicaoAtual, int posicaoHot) {
		if (posicaoAtual < 1 || posicaoAtual > 100 || posicaoHot < 1 || posicaoHot > 10) {
			return "POSIÇÃO INVALIDA";
		}
		posicaoAtual = posicaoAtual - 1;
		posicaoHot = posicaoHot - 1;

		Filme filme = getFilme(posicaoAtual);
		if (verificaFilmeHot(filme)) {
			return "ESSE FILME JÁ ESTÁ ADICIONADO";
		}
		if (hotList[posicaoHot] != null) {
			this.removerFilmeHot(posicaoHot + 1);
		}

		hotList[posicaoHot] = filmes[posicaoAtual];
		return "ADICIONADO À HOTLIST NA POSIÇÃO " + (posicaoHot + 1) + "!";
	}

	/**
	 * Remove um filme da hotlist de acordo com a posição.
	 *
	 * @param posicaoHot A posição (1 a 10) da hotlist.
	 * @return Uma string vazia se removido, ou mensagem de erro se posição
	 *         inválida.
	 */
	public String removerFilmeHot(int posicaoHot) {
		if (posicaoHot < 1 || posicaoHot > 10) {
			return "POSIÇÃO INVALIDA";
		}
		posicaoHot = posicaoHot - 1;
		hotList[posicaoHot] = null;
		return "";
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 *
 */
public class LeitorFilmNow { 

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */ 
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

		}

		Scanner scanner = new Scanner(System.in);
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 *
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */

		System.out.print("\n---\nMENU\n" + "(A)Adicionar filme\n" + "(M)Mostrar todos\n" + "(D)Detalhar filme\n"
				+ "(E)Exibir HotList\n" + "(H)Atribuir Hot\n" + "(R)Remover Hot\n" + "(S)air\n" + "\n" + "Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 *
	 * @param opcao   Opção digitada.
	 * @param fn      O sistema FilmNow que estamos manipulando.

		case "D":

			mostrarFilmesHot(fn);
			break;
		case "H":
			adicionaFilmeHot(fn, scanner);
			break;
		case "R":
			removeFilmeHot(fn, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("OPÇÃO INVÁLIDA!");
		}
	}
