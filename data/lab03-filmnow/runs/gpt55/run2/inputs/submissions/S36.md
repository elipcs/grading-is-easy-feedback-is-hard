package filmnow;

/**
 * Apresenta a classe dos filmes com nome, ano de lançamento e a plataforma que foi lançado
 * se o filme tiver o mesmo nome e mesma data de lançamento são considerados iguais
 * @author [REDACTED]
 */
public class Filme {
    private String nome;
    private String ano;
    private String plataforma;

    /**
     * Construtor para um novo filme
     *
     * @param nome nome do filme
     * @param ano ano que o filme foi lançado
     * @param plataforma o formato que o filme foi lançado(netflix, cinema, etc.)
     *@author [REDACTED]
     */
    public Filme(String nome, String ano, String plataforma) {
        this.nome = nome.trim();
        this.ano = ano.trim();
        this.plataforma = plataforma.trim();
    }

    /**
     * @return o nome do filme.
     */
    public String getNome() {
        return nome;
    }

    /**
     * @return O ano de lançamento como uma string (pode ser vazia).
     */
    public String getAno() {
        return ano;
    }

    /**
     * @return a plataforma de lançamento
     */
    public String getPlataforma() {
        return plataforma;
    }

    /**
     * retorna em formato de texto a apresentação de como será exibida as informações
     * @return String com a formatação desejada
     *@author [REDACTED]
     */
    @Override
    public String toString() {
        if (ano.length() == 0) {
            return nome + "\n" + plataforma;
        }
        return nome + ", " + ano + "\n" + plataforma;
    }

    /**
     * analisa se ja esiste filme iguail
     * @author [REDACTED]
     *
     * @param obj o objeto que vai ser comparado com o filme
     * @return true se os filmes forem iguais; false caso não sejam
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (obj == null || getClass() != obj.getClass()) {
            return false;
        }
        Filme outro = (Filme) obj;
        return this.nome.equals(outro.nome) && this.ano.equals(outro.ano);
    }

    /**
     * Retorna o código hash do filme
     * @author [REDACTED]
     *
     * @return Um inteiro que representa o hash
     */
    @Override
    public int hashCode() {
        return nome.hashCode() + ano.hashCode();
    }
}

package filmnow;

/**
 * Classe que gerencia os filmes cadastrados com um array contendo os filmes e um hotlists
 * podendo cadastrar, detalhar, exibir e destacar filmes.
 * 
 * @author [REDACTED]
 */
public class FilmNow {
    private static final int TAMANHO = 100;
    private static final int HOT_TAMANHO = 10;

    private Filme[] filmes;
    private Filme[] hotlist;

    /**
     * Construtor
     * @author [REDACTED]
     */
    public FilmNow() {
        this.filmes = new Filme[TAMANHO];
        this.hotlist = new Filme[HOT_TAMANHO];
    }

    /**
     * Retorna o filme em determinada posição
     * 
     * @param posicao a posição desejadad
     * @return null se a posição for inválida
     * @author [REDACTED]
     */
    public Filme getFilme(int posicao) {
        if (posicao < 0 || posicao >= TAMANHO) return null;
        return filmes[posicao];
    }

    /**
     * Retorna uma cópia dos filmes cadastrados.
     * 
     * @return um array de filmes com até 100 posições.
     *  @author [REDACTED]
     */
    public Filme[] getFilmes() {
        return this.filmes.clone();
    }

    /**
     * Cadastra um novo filme em uma posição específica.
     * 
     * @param posicao a posição (0 a 99) onde o filme será armazenado
     * @param nome o nome do filme.
     * @param ano o ano de lançamento do filme.
     * @param plataforma a plataforma de lançamento
     * @return mensagem de sucesso ou erro da operação.
     * @author [REDACTED]
     */
    public String cadastraFilme(int posicao, String nome, int ano, String plataforma) {
        if (posicao < 0 || posicao >= TAMANHO) {
            return "POSIÇÃO INVÁLIDA";
        }

        if (nome == null || nome.trim().equals("") || plataforma == null || plataforma.trim().equals("")) {
            return "FILME INVALIDO";
        }

        Filme novo = new Filme(nome, String.valueOf(ano), plataforma);

        for (Filme f : filmes) {
            if (f != null && f.equals(novo)) {
                return "FILME JA ADICIONADO";
            }
        }

        filmes[posicao] = novo;
        return "FILME ADICIONADO";
    }

    /**
     * Retorna uma lista com todos os filmes cadastrados com posição e nome
     * 
     * @return uma string com a lista dos filmes cadastrados
     * @author [REDACTED]
     */
    public String mostrarTodos() {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < TAMANHO; i++) {
            if (filmes[i] != null) {
                sb.append((i + 1) + " - " + filmes[i].getNome() + "\n");
            }
        }
        return sb.toString().trim();
    }

    /**
     * Detalha um filme escolhido conforme a posição
     * Se o filme tiver na hotlist terá um emoji antes do nome do filme
     * 
     * @param posicao a posição do filme a ser detalhado.
     * @return os detalhes do filme ou mensagem de erro.
     * @author [REDACTED]
     */
    public String detalharFilme(int posicao) {
        if (posicao < 1 || posicao > TAMANHO || filmes[posicao - 1] == null) {
            return "POSIÇÃO INVÁLIDA!";
        }

        Filme filme = filmes[posicao - 1];

        boolean estaNaHot = false;
        for (Filme f : hotlist) {
            if (f != null && f.equals(filme)) {
                estaNaHot = true;
                break;
            }
        }

        if (estaNaHot) {
            return "🔥 " + filme.toString();
        } else {
            return filme.toString();
        }
    }

    /**
     * Adiciona um filme à hotlist em uma posição específica.
     * 
     * @return mensagem de sucesso ou erro.
     * @author [REDACTED]
     */
    public String adicionaHot(int posHot, int posFilme) {
        if (posHot < 1 || posHot > HOT_TAMANHO) {
            return "POSIÇÃO INVÁLIDA";
        }

        if (posFilme < 1 || posFilme > TAMANHO || filmes[posFilme - 1] == null) {
            return "POSIÇÃO INVÁLIDA";
        }

        Filme filme = filmes[posFilme - 1];

        for (Filme f : hotlist) {
            if (f != null && f.equals(filme)) {
                return "FILME JÁ NA HOTLIST";
            }
        }

        hotlist[posHot - 1] = filme;
        return "ADICIONADO À HOTLIST NA POSIÇÃO " + posHot + "!";
    }

    /**
     * Remove um filme da hotlist
     * 
     * @return mensagem confirmando a remoção ou erro.
     */
    public String removerHot(int posHot) {
        if (posHot < 1 || posHot > HOT_TAMANHO) {
            return "POSIÇÃO INVÁLIDA";
        }

        hotlist[posHot - 1] = null;
        return "REMOVIDO DA HOTLIST";
    }

    /**
     * Exibe todos os filmes na hotlist, com posições, nomes e anos
     * 
     * @return uma string com os filmes da hotlist.
     * @author [REDACTED]
     */
    public String exibirHotList() {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < HOT_TAMANHO; i++) {
            if (hotlist[i] != null) {
                sb.append((i + 1) + " - " + hotlist[i].getNome() + ", " + hotlist[i].getAno() + "\n");
            }
        }
        return sb.toString().trim();
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}

    /**

     * @param scanner entrada do usuário
     * @return a opção escolhida em letra maiúscula.

    private static String menu(Scanner scanner) {
        System.out.println(
            "\n---\nMENU\n" +
            "(A)Adicionar filme\n" +
            "(M)Mostrar todos\n" +
            "(D)Detalhar filme\n" +
            "(E)Exibir HotList\n" +
            "(H)Atribuir Hot\n" +
            "(R)Remover Hot\n" +
            "(S)Sair"
        );
        System.out.print("\nOpção> ");
        return scanner.nextLine().trim().toUpperCase();
    }

    /**
     * Executa o comando escolhido
     * 
     * @author [REDACTED]
     */
    private static void comando(String opcao, FilmNow filme, Scanner scanner) {

            adicionarFilme(filme, scanner);

            mostrarFilmes(filme);

            removerHot(filme, scanner);

            System.out.println("Opção inválida!");
