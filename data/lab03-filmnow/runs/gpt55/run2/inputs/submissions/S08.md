package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes.
 *
 * @author [REDACTED]
 *
 */
public class FilmNow {

	private static final int TAMANHO = 100;
	private static final int TAMANHOHOTLIST = 10;
	
	private Filme[] filmes; // uma representacao simploria da lista de filmes
	private Filme[] hotList;
	private String status;
	private int hotlistPosicao;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[TAMANHOHOTLIST];
		this.status = "";
		this.hotlistPosicao = 0;
	}

	/**
	 * Acessa a lista de filmes mantida.
	 *
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 *
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		return filmes[posicao];
	}

	/**
	 * Cria uma cópia do objeto.
	 * 
	 * @return Uma cópia do objeto
	 */
	public Filme[] getFilmesHot() {
		return this.hotList.clone();
	}

	/**
	 * Acessa a String atribuida a status.
	 * 
	 * @return Retorna a String atribuida a status.
	 */
	public String getStatus() {
		return status;
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve
	 * o anterior.
	 *
	 * @param posicao Posição do filme.
	 * @param nome    Nome do filme.
	 * @param ano     Ano de lançamento do filme.
	 * @param local   Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if (posicao < 1 || posicao > TAMANHO) {
			return "POSIÇÃO INVÁLIDA";
		}
		else if (nome == null || local == null || nome.trim().isEmpty() || local.trim().isEmpty()) {
			return "FILME INVALIDO";
		} 
		else {
			Filme filme = new Filme(nome, ano, local);
			for (int i = 0; i < filmes.length; i++) {
				if (filmes[i] != null && filmes[i].equals(filme)) {
					return "FILME JÁ ADICIONADO";
				} 
			}
			if (filmes[posicao - 1] != null) {
				if (verificaFilmeHot(filmes[posicao - 1]) == true) {
					removerFilmeHot(hotlistPosicao + 1);
				}
			}	
			this.filmes[posicao - 1] = filme;
			return "FILME ADICIONADO";
		}
	}

	/**
	 * Imprime lista de filmes.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 */
	public String mostraFilmes() {
		
		Filme[] filmes = getFilmes();
		StringBuilder sb = new StringBuilder();

		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				sb.append(formataFilme(i, filmes[i])).append("\n");
			}
		}
		return sb.toString().trim();
	}

	/**
	 * Imprime a lista de filmes hot.
	 * @param fn O sistema FilmNow a ser manipulado.
	 */
	public String mostraFilmesHot() {
		
		Filme[] filmes = getFilmesHot();
		StringBuilder sb = new StringBuilder();

		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				sb.append(formataFilme(i, filmes[i])).append("\n");
			}
		}
		return sb.toString().trim();
	}

	/**
	 * Formata um filme para impressão. 
	 * 
	 * @param posicao A posição do filme (que é exibida)/
	 * @param filme O filme a ser impresso.
	 * @return A String formatada.
	 */
	private static String formataFilme(int posicao, Filme filme) {
		return (posicao + 1) + " - " + filme.getNome();
	}

	/**
	 * Detalha o filme escolhido. 
	 * 
	 * @param posicao Posição na qual o filme a ser detalhado se encontra na lista de filmes.
	 * @return A String formatada referente aos detalhes do filme.
	 */
	public String detalharFilme(int posicao) {
		if (posicao < 1 || posicao > TAMANHO) {
			return "POSIÇÃO INVÁLIDA";
		}
		Filme filme = getFilme(posicao - 1);
		if (filme == null) {
			return "";
		}
		if (verificaFilmeHot(filme)) {
			return "🔥 " + filme.toString();
		}
		return filme.toString();
	}
	
	/**
	 * Verifica se o filme passado como parãmetro ja está na HotList.
	 * 
	 * @param filme Filme a ser verificado.
	 * @return Um boolean, true -> Já está na hotlist; false -> Não está na hotlist.
	 */
	public boolean verificaFilmeHot(Filme filme) {
		for (int i = 0; i < hotList.length; i++) {
			if (hotList[i] != null && hotList[i].equals(filme)) {
				hotlistPosicao = i;
				return true;
			}
		}
		return false;
	}

	/**
	 * Adiciona um filme a HotList.
	 * 
	 * @param posicaoAtual Posição do filme na lista de filmes.
	 * @param posicaoHot Posição a ser adicionado na HotList.
	 */
	public String CadastrarFilmeHot(int posicaoFilme, int posicaoNaHotList) {
		if (posicaoFilme < 1 || posicaoFilme > TAMANHO || posicaoNaHotList < 1 || posicaoNaHotList > TAMANHOHOTLIST) {
			return "POSIÇÃO INVÁLIDA";
		}
		posicaoFilme = posicaoFilme - 1;
		posicaoNaHotList = posicaoNaHotList - 1;

		Filme filme = getFilme(posicaoFilme);
		if (verificaFilmeHot(filme)) {
			return "ESSE FILME JÁ EXISTE NA HOTLIST!";
		}
		if (hotList[posicaoNaHotList] != null) {
			this.removerFilmeHot(posicaoNaHotList + 1);
		}
		hotList[posicaoNaHotList] = filmes[posicaoFilme];
		return "ADICIONADO À HOTLIST NA POSIÇÃO " + (posicaoNaHotList + 1) + "!";
	}

	/**
	 * Remove um filme da HotList.
	 * 
	 * @param posicaoHot Posição que o filme a ser removido se encontra na HotList.
	 */
	public String removerFilmeHot(int posicaoHot) {
		if (posicaoHot < 1 || posicaoHot > TAMANHOHOTLIST) {
			return "POSIÇÃO INVALIDA";
		}
		posicaoHot = posicaoHot - 1;
		hotList[posicaoHot] = null;
		return "";
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
