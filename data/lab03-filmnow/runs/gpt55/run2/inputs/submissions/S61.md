package filmnow;

/**
 * Representa um filme com informações básicas como nome, ano de lançamento e local.
 * 
 * Esta classe fornece métodos para acessar e modificar as propriedades do filme, além de métodos para comparar filmes e exibir suas informações em formato de string.
 * 
 * @author [REDACTED]
 */
public class Filme {
    private String nome;
    private int ano;
    private String local;

    /**
     * Constrói um novo filme com o nome, ano e local especificados.
     * 
     * @param nome  O nome do filme.
     * @param ano   O ano de lançamento do filme.
     * @param local O local onde o filme foi exibido.
     */
    public Filme(String nome, int ano, String local) {
        this.nome = nome;
        this.ano = ano;
        this.local = local;
    }

    /**
     * Retorna o nome do filme.
     * 
     * @return O nome do filme.
     */
    public String getNome() {
        return nome;
    }

    /**
     * Retorna o ano de lançamento do filme.
     * 
     * @return O ano de lançamento do filme.
     */
    public int getAno() {
        return ano;
    }

    /**
     * Retorna o local onde o filme foi exibido.
     * 
     * @return O local do filme.
     */
    public String getLocal() {
        return local;
    }

    /**
     * Define o nome do filme.
     * 
     * @param nome O novo nome do filme.
     */
    public void setNome(String nome) {
        this.nome = nome;
    }

    /**
     * Define o local onde o filme foi exibido.
     * 
     * @param local O novo local do filme.
     */
    public void setLocal(String local) {
        this.local = local;
    }

    /**
     * Define o ano de lançamento do filme.
     * 
     * @param ano O novo ano de lançamento do filme.
     */
    public void setAno(int ano) {
        this.ano = ano;
    }

    /**
     * Retorna uma representação em string do filme no formato "nome, ano\nlocal".
     * 
     * @return A representação em string do filme.
     */
    @Override
    public String toString() {
        return nome + ", " + ano + "\n" + local;
    }

    /**
     * Compara este filme com outro objeto para determinar se são iguais.
     * 
     * @param obj O objeto a ser comparado.
     * @return {@code true} se o objeto for um filme com o mesmo nome, ano e local; {@code false} caso contrário.
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || getClass() != obj.getClass()) return false;

        Filme filme = (Filme) obj;

        if (ano != filme.ano) return false;
        if (!nome.equals(filme.nome)) return false;
        return local.equals(filme.local);
    }
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes e 10 filmes na hotlist.
 * 
 * @author [REDACTED]
 */
public class FilmNow {

    private Filme[] filmes;
    private Filme[] filmesHot;

    /**
     * Construtor padrão para inicializar o sistema FilmNow.
     * Cria arrays para armazenar até 100 filmes e 10 filmes na hotlist.
     */
    public FilmNow() {
        this.filmes = new Filme[100];
        this.filmesHot = new Filme[10];
    }

    /**
     * Obtém uma cópia do array de filmes.
     * 
     * @return Um array com todos os filmes.
     */
    public Filme[] getFilmes() {
        return this.filmes.clone();
    }

    /**
     * Obtém uma cópia do array de filmes na hotlist.
     * 
     * @return Um array com os filmes da hotlist.
     */
    public Filme[] getFilmesHot() {
        return this.filmesHot.clone();
    }

    /**
     * Obtém o filme na posição especificada.
     * 
     * @param posicao A posição do filme no array (1 a 100).
     * @return O filme na posição especificada, ou null se a posição for inválida ou o filme não existir.
     */
    public Filme getFilme(int posicao) {
        if (posicao < 1 || posicao > 100 || filmes[posicao - 1] == null) {
            return null;
        }
        return filmes[posicao - 1];
    }

    /**
     * Verifica se um filme está na hotlist.
     * 
     * @param filme O filme a ser verificado.
     * @return true se o filme estiver na hotlist, false caso contrário.
     */
    public boolean isHot(Filme filme) {
        for (Filme hot : filmesHot) {
            if (hot != null && hot.equals(filme)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Adiciona um filme à hotlist na posição especificada.
     * 
     * @param filme O filme a ser adicionado.
     * @param posicaoHot A posição na hotlist (1 a 10).
     * @return Uma mensagem indicando o resultado da operação.
     */
    public String atribuirHot(Filme filme, int posicaoHot) {
        if (posicaoHot < 1 || posicaoHot > 10) {
            return "POSIÇÃO INVÁLIDA!";
        }
        if (isHot(filme)) {
            return "FILME JÁ ESTÁ NA HOTLIST!";
        }
        filmesHot[posicaoHot - 1] = filme;
        return "FILME ADICIONADO À HOTLIST!";
    }

    /**
     * Remove um filme da hotlist na posição especificada.
     * 
     * @param posicao A posição na hotlist (1 a 10).
     * @return Uma mensagem indicando o resultado da operação.
     */
    public String removerHot(int posicao) {
        if (posicao < 1 || posicao > 10 || filmesHot[posicao - 1] == null) {
            return "POSIÇÃO INVÁLIDA!";
        }
        filmesHot[posicao - 1] = null;
        return "FILME REMOVIDO DA HOTLIST!";
    }

    /**
     * Adiciona um filme ao sistema na posição especificada.
     * 
     * @param posicao A posição no sistema (1 a 100).
     * @param nome O nome do filme.
     * @param ano O ano de lançamento do filme.
     * @param local O local onde o filme foi produzido ou é exibido.
     * @return Uma mensagem indicando o resultado da operação.
     */
    public String cadastraFilme(int posicao, String nome, int ano, String local) {
        if (posicao < 1 || posicao > 100) {
            return "POSIÇÃO INVÁLIDA!";
        }
        Filme novoFilme = new Filme(nome, ano, local);
        if (filmes[posicao - 1] != null) {
            return "POSIÇÃO OCUPADA!";
        }
        for (Filme filme : filmes) {
            if (filme != null && filme.getNome().equals(nome)) {
                return "FILME JÁ ADICIONADO!";
            }
        }
        filmes[posicao - 1] = novoFilme;
        return "FILME ADICIONADO!";
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos CSV os dados necessários e povoar o sistema de filmes.
 * 
 * @author [REDACTED]
 */
public class LeitorFilmNow {

    private static final int COLUNA_POSICAO = 0;
    private static final int COLUNA_NOME = 1;
    private static final int COLUNA_ANO = 2;
    private static final int COLUNA_LOCAL = 3;

    /**
     * Lê filmes de um arquivo CSV e os coloca no sistema.
     * 
     * @param arquivoFilmes Caminho para o arquivo contendo os filmes.
     * @param fn O sistema FilmNow a manipular.
     * @return O número de filmes adicionados ao sistema.
     * @throws IOException Caso não tenhamos permissão para ler o arquivo.
     * @throws FileNotFoundException Caso o arquivo não exista.
     */
    public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
        int carregados = 0;

        try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
            String linha;
            while ((linha = br.readLine()) != null) {
                carregados += 1;
                if (carregados == 1) {
                    // Pulamos a primeira linha, que contém o cabeçalho
                    continue;
                }
                String[] campos = linha.split(",");
                processaLinhaCsvFilmes(campos, fn);
            }
        }
        
        return carregados;
    }

    /**
     * Processa uma linha do arquivo CSV e adiciona os dados ao sistema.
     * 
     * @param campos As informações lidas do CSV.
     * @param fn O sistema FilmNow a manipular.
     */
    public void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
        int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
        String nome = campos[COLUNA_NOME].trim();
        String ano = campos[COLUNA_ANO].trim();
        String local = campos[COLUNA_LOCAL].trim();

        fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
    }
}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

/**
 * Interface com menus texto para manipular o sistema FilmNow.
 * 
 * @author [REDACTED]
 */
public class MainFilmNow {

    public static void main(String[] args) {
        FilmNow fn = new FilmNow();

        System.out.println("Carregando filmes ...");
        try {
            carregaFilmes("filmes_inicial.csv", fn);
        } catch (FileNotFoundException e) {
            System.err.println("Arquivo não encontrado: " + e.getMessage());
        } catch (IOException e) {
            System.err.println("Erro lendo arquivo: " + e.getMessage());
        }

        Scanner scanner = new Scanner(System.in);
        String escolha = "";
        while (true) {
            escolha = menu(scanner);
            comando(escolha, fn, scanner);
        }
    }

    /**
     * Exibe o menu e captura a escolha do/a usuário/a.
     * 
     * @param scanner Para captura da opção do usuário/a.
     * @return O comando escolhido.
     */
    private static String menu(Scanner scanner) {
        System.out.println(
                "\n---\nMENU\n" + 
                "(A)Adicionar filme\n" + 
                "(M)Mostrar todos\n" + 
                "(D)Detalhar filme\n" + 
                "(E)Exibir HotList\n" + 
                "(H)Atribuir Hot\n" + 
                "(R)Remover Hot\n" + 
                "(S)Sair\n" + 
                "\nOpção> ");
        return scanner.next().toUpperCase();
    }

    /**
     * Interpreta a opção escolhida por quem está usando o sistema.
     * 
     * @param opcao   Opção digitada.
     * @param fn  O sistema FilmNow que estamos manipulando.
     * @param scanner Objeto scanner para o caso do comando precisar de mais input.
     */
    private static void comando(String opcao, FilmNow fn, Scanner scanner) {
        switch (opcao) {
            case "A":
                adicionaFilme(fn, scanner);
                break;
            case "M":
                mostrarFilmes(fn);
                break;
            case "D":
                detalharFilme(fn, scanner);
                break;
            case "E":
                exibirHotList(fn);
                break;
            case "H":
                atribuirHot(fn, scanner);
                break;
            case "R":
                removerHot(fn, scanner);
                break;
            case "S":
                sai();
                break;
            default:
                System.out.println("Opção inválida!");
        }
    }

    /**
     * Adiciona um novo filme ao sistema com validação dos dados.
     * 
     * @param fn O sistema FilmNow a ser manipulado.
     * @param scanner Scanner para capturar dados do usuário.
     */

        System.out.print("\nPosição no sistema> ");
        int posicao = scanner.nextInt();
        if (posicao < 1 || posicao > 100) {
            System.out.println("POSIÇÃO INVÁLIDA");
            return;
        }
        System.out.print("\nNome> ");
        scanner.nextLine(); // Consumir a nova linha
        String nome = scanner.nextLine();
        System.out.print("\nAno> ");
        String ano = scanner.nextLine();
        System.out.print("\nLocal> ");
        String local = scanner.nextLine();

        // Validação dos dados do filme
        if (nome.isEmpty() || local.isEmpty()) {
            System.out.println("FILME INVALIDO");
            return;
        }

        int anoInt;
        try {
            anoInt = Integer.parseInt(ano);
        } catch (NumberFormatException e) {
            anoInt = 0; // Ano inválido será tratado como zero
        }

        System.out.println(fn.cadastraFilme(posicao, nome, anoInt, local));
    }

     * Imprime lista de filmes.
     * 
     * @param fn O sistema FilmNow a ser manipulado.

        Filme[] todosOsFilmes = fn.getFilmes();

        Filme[] filmes = fn.getFilmes();
        for (int i = 0; i < todosOsFilmes.length; i++) {
            if (todosOsFilmes[i] != null) {
                boolean estaNaHotList = false;
                for (Filme filmeHot : filmesHotList) {

                } else {
