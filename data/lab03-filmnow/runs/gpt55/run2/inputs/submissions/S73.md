package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes.
 * 
 * @author [REDACTED]
 *
 */

public class FilmNow {

	private static final int TAMANHO_LISTA = 100;
	private static final int TAMANHO_COLUNA = 3;

	private String[][] filmes = new String[TAMANHO_LISTA][TAMANHO_COLUNA]; // Matriz para armazenar nome, ano e local.
	private String[][] hotList = new String[TAMANHO_LISTA][TAMANHO_COLUNA]; // Matriz para armazenar os filmes da

	/**
	 * Acessa a lista de filmes em matriz.
	 * 
	 * @return O array de filmes formatado.
	 */

	public String getFilme(int posicao) {
		if (posicao < 0 || posicao >= 100 || filmes[posicao][0] == null) {
			return null;
		}
		return filmes[posicao][0] + ", " + filmes[posicao][1] + ", " + filmes[posicao][2];
	}

	/**
	 * Acessa a lista de filmes mantida.
	 * 
	 * @return O array de filmes.
	 */

	public String[] getNomes() {
		String[] nomes = new String[100];
		for (int i = 0; i < 100; i++) {
			nomes[i] = filmes[i][0];
		}
		return nomes;
	}

	/**
	 * Verifica a existencia de um filme.
	 * 
	 * @param nome Nome do filme.
	 * @param ano  Ano de lançamento do filme.
	 * @return valor true ou false.
	 */
	public boolean existeFilme(String nome, int ano) {
		for (int i = 0; i < 100; i++) {
			if (filmes[i][0] != null && filmes[i][0].equals(nome) && Integer.parseInt(filmes[i][1]) == ano) {
				return true;
			}
		}
		return false;
	}

	/**
	 * Cadastra um filme.
	 * 
	 * @param posicao Posição do filme.
	 * @param nome    Nome do filme.
	 * @param ano     Ano de lançamento do filme.
	 * @param local   Local onde o filme pode ser assitido.
	 */
	public void cadastraFilme(int posicao, String nome, int ano, String local) {
		if (posicao >= 0 && posicao < 100) {
			filmes[posicao][0] = nome;
			filmes[posicao][1] = String.valueOf(ano);
			filmes[posicao][2] = local;
		}
	}

	/**
	 * Pega os nomes dos filmes dentro de uma hot list.
	 * 
	 * @return nomes dos filmes dentro da hot list.
	 */
	public String[] getNomesHot() {
		String[] nomesHot = new String[100];
		for (int i = 0; i < 100; i++) {
			nomesHot[i] = hotList[i][0];
		}
		return nomesHot;
	}

	/**
	 * Pega os anos dos filmes dentro de uma HotList.
	 * 
	 * @return anos dos filmes dentro da HotList.
	 */
	public int[] getAnoHot() {
		int[] anosHot = new int[100];
		for (int i = 0; i < 100; i++) {
			if (hotList[i][1] != null) {
				anosHot[i] = Integer.parseInt(hotList[i][1]);
			}
		}
		return anosHot;
	}

	/**
	 * Pega a HotList por completo.
	 * 
	 * @return a HotList.
	 */
	public String[][] getHotList() {
		return hotList;
	}

	/**
	 * Cadastra um filme na HotList.
	 * 
	 * @param posiçãofilme 	dentro da lista filme.
	 * @param posicaoHot 	posciçao colocada na lista HotList.
	 */
	public void cadastraHot(int posicaoFilme, int posicaoHot) {
		if (posicaoFilme >= 0 && posicaoFilme < 100 && posicaoHot >= 0 && posicaoHot < 100) {
			hotList[posicaoHot] = filmes[posicaoFilme];
		}
	}

	/**
	 * Remove um filme da HotList.
	 * 
	 * @param posicaoHot	posição para ser removida na HotList
	 */
	public void removeHot(int posicaoHot) {
		if (posicaoHot >= 0 && posicaoHot < 100 && hotList[posicaoHot] != null) {
			if (filmes[posicaoHot][0] == null) {
				filmes[posicaoHot][0] = hotList[posicaoHot][0];
				filmes[posicaoHot][1] = hotList[posicaoHot][1];
				filmes[posicaoHot][2] = hotList[posicaoHot][2];
			}
			hotList[posicaoHot] = null;
		} else {
			System.out.println("FILME INVALIDO");
		}
	}

	/**
	 * Varifica se um filme para ser detalhada é da HotList.
	 * 
	 * @param posição posição para ser detalhada
	 * @return valor formatado da HotList
	 */
	public String detalharFilmeIsHot(int posicao) {
		if (posicao < 0 || posicao >= 100 || filmes[posicao][0] == null) {
			if (posicao <= 0 || posicao > 100) {
				return "POSIÇÃO INVÁLIDA";
			}
			return null;
		}

		String nome = filmes[posicao][0];
		String ano = filmes[posicao][1];
		String local = filmes[posicao][2];
		boolean isHot = false;

		for (String[] hotFilme : hotList) {
			if (hotFilme != null && hotFilme[0] != null && hotFilme[0].equals(nome) && hotFilme[1].equals(ano)) {
				isHot = true;
				break;
			}
		}

		if (isHot) {
			return "🔥 " + nome + ", " + ano + "\n" + local;
		} else {
			return posicao + " - " + nome + ", " + ano + "\n" + local;
		}
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de
 * filmes.
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;

	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * 
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}

		return carregados;
	}

	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial.
	 * 
	 * @param campos As informações lidas do csv.
	 * @param fn O sistema FilmNow a manipular.
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

/**
 * Interface com menus texto para manipular o sistema FilmNow.
 * 
 * @author [REDACTED]
 *
 */
public class MainFilmNow {

	public static void main(String[] args) {
		FilmNow fn = new FilmNow();

		System.out.println("Carregando filmes ...");
		try {
			carregaFilmes("filmes_inicial.csv", fn);
		} catch (FileNotFoundException e) {
			System.err.println("Arquivo não encontrado: " + e.getMessage());
		} catch (IOException e) {
			System.err.println("Erro lendo arquivo: " + e.getMessage());
		}

		Scanner scanner = new Scanner(System.in);
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}
	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */

	private static String menu(Scanner scanner) {
		System.out.print("\n---\nMENU\n" + "(A)Adicionar filme\n" + "(M)Mostrar todos\n" + "(D)Detalhar filme\n"
				+ "(E)Exibir HotList\n" + "(H)Atribuir Hot\n" + "(R)Remover Hot\n" + "(S)Sair\n" + "\nOpção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opção   Opção digitada.
	 * @param fn      O sistema FilmNow que estamos manipulando.
	 * @param scanner Objeto scanner para o caso do comando precisar de mais input.
	 */

	private static void comando(String opcao, FilmNow fn, Scanner scanner) {
		switch (opcao) {
		case "A":
			System.out.println(" ");
			adicionaFilme(fn, scanner);
			break;
		case "M":
			System.out.println(" ");
			mostrarFilmes(fn);
			break;
		case "D":
			System.out.println(" ");
			detalharFilme(fn, scanner);
			break;
		case "E":
			System.out.println(" ");
			exibirHot(fn);
			break;
		case "H":
			System.out.println(" ");
			atribuirHot(fn, scanner);
			break;
		case "R":
			System.out.println(" ");
			removerHot(fn, scanner);
			break;
		case "S":
			System.out.println(" ");
			sai();
			break;
		default:
			System.out.println("OPÇÃO INVÁLIDA!");
		}
	}

	/**
	 * Imprime lista de filmes.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 */

	private static void mostrarFilmes(FilmNow fn) {
		String[] nomes = fn.getNomes();
		for (int i = 0; i < nomes.length; i++) {
			if (nomes[i] != null) {
				System.out.println(formataFilme(i, nomes[i]));
			}
		}
	}

	/**
	 * Imprime os detalhes de um dos filmes.
	 * 
	 * @param fn      O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar qual contato.
	 */

	private static void detalharFilme(FilmNow fn, Scanner scanner) {
		System.out.print("Qual filme> ");
		int posicao = scanner.nextInt();

		String detalhes = fn.detalharFilmeIsHot(posicao);

		if (detalhes == null) {
			System.out.print("POSIÇÃO INVÁLIDA!\n");
		} else {
			System.out.print("\nDados do filme:\n" + detalhes);
		}

	}

	/**
	 * Formata um filme para impressão.
	 * 
	 * @param posicao A posição do filme (que é exibida)/
	 * @param filme   O filme a ser impresso.
	 * @return A String formatada.
	 */

	private static String formataFilme(int posicao, String filme) {
		return posicao + " - " + filme;
	}

	/**
	 * Cadastra um filme no sistema.
	 * 
	 * @param fn      O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para pedir informações do contato.
	 */

	private static void adicionaFilme(FilmNow fn, Scanner scanner) {
		System.out.print("Posição no sistema> ");
		int posicao = scanner.nextInt();
		scanner.nextLine();

		if (posicao >= 100) {
			System.out.println("NÚMERO MAIOR QUE 100");
			return;
		}

		System.out.print("Nome> ");
		String nome = scanner.nextLine().trim();

		System.out.print("Ano> ");
		String ano = scanner.nextLine();

		System.out.print("Local> ");
		String local = scanner.nextLine().trim();

		if (nome.isEmpty() || local.isEmpty()) {
			System.out.println("FILME INVALIDO");
			return;
		}

		if (fn.existeFilme(nome, Integer.parseInt(ano))) {
			System.out.println("FILME JA ADICIONADO");
		} else {
			fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
		}
	}

	/**
	 * Exibe a lista dos filmes dentro da hot list.
	 * 
	 * @param fn      O sistema FilmNow a ser manipulado.
	 */

	private static void exibirHot(FilmNow fn) {
		String[] filmes = fn.getNomesHot();
		int[] anos = fn.getAnoHot();
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				System.out.println(i + " - " + filmes[i] + ", " + anos[i]);
			}
		}
	}

	/**
	 * Atribui um dos filmes da lista filme para dentro da hot list.
	 * 
	 * @param fn      O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar qual contato.
	 */

	private static void atribuirHot(FilmNow fn, Scanner scanner) {
		System.out.print("Filme> ");
		int filme = scanner.nextInt();

		System.out.print("Posição> ");
		int posicao = scanner.nextInt();

		if (posicao >= 100 || filme >= 100) {
			System.out.println("NÚMERO MAIOR QUE 100");
			return;
		}

		if (fn.getFilme(filme) == null) {
			System.out.println("FILME INVALIDO");
			return;
		}

		fn.cadastraHot(filme, posicao);
		System.out.println("ADICIONADO À HOTLIST NA POSIÇÃO " + posicao + "!");
	}

	/**
	 * Remove um filme de dentro da hot list.
	 * 
	 * @param fn      O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar qual contato.
	 */

	private static void removerHot(FilmNow fn, Scanner scanner) {
		System.out.print("Posição> ");
		int posicao = scanner.nextInt();

		if (posicao >= 0 && posicao < 100 && fn.getNomesHot()[posicao] != null) {
			fn.removeHot(posicao);
		} else {
			System.out.println("POSIÇÃO INVÁLIDA!");
		}
	}

	/**
	 * Sai da aplicação.
	 */

	private static void sai() {
		System.out.print("Vlw flw");
		System.exit(0);
	}

	/**
	 * Lê carga de filmes de um arquivo csv.
	 * 
	 * @param arquivoFilmes O caminho para o arquivo.
	 * @param O             sistema FilmNow a ser populado com os dados.
	 * @throws IOException Caso o arquivo não exista ou não possa ser lido.
	 */

	private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		LeitorFilmNow leitor = new LeitorFilmNow();
		int carregados = leitor.carregaContatos(arquivoFilmes, fn);
		System.out.println("Carregamos " + (carregados - 1) + " registros.");
	}
}
