package filmnow;
import java.util.Arrays; 
/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	private static final int TAMANINHO = 10;
	
	private Filme[] filmes;
	private String[] Hotlist;
	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.Hotlist = new String[TAMANINHO];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	public String[] getHotlist() {
		return this.Hotlist.clone();
	} 
	public void removerHot(int posicao) {
		this.Hotlist[posicao-1] = null;
	}
	/**
	 * 
	 * @param nome
	 * @return
	 */
	public boolean EmHotlist(String nome) {
		for(int i=0;i<10; i++){
			if(nome.equals(Hotlist[i])){
				return true;
			}
		}
		return false;
	}
	/**
	 * Este metódo chamado do adiciona Hotlist vai colocar o nome do filme dentro da Hotlist 
	 * @param filme
	 * @param posicao
	 * @return
	 */
	public String colocarHotlist(int filme, int posicao) {
		String jatem = 	filmes[posicao-1].getFilme();
		if(EmHotlist(jatem)==true) {
			return "JÁ ADICIONADO !!!";
		}
		cadastraHotlist(posicao,filme);
		return "ADICIONADO À HOTLIST NA POSIÇÃO "+ posicao+1 + "!";
	}
	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public String getFilme(int posicao) {
			return filmes[posicao].getFilme();
	}
	public String getAno(int posicao) {
		return filmes[posicao].getAno();
	}
	/**
	 * ele retorna o local
	 * @param posicao
	 * @return
	 */
	public String getLocal(int posicao) {
		return filmes[posicao].getLocal();
	}
	/**
	 * este metodo ele vai detalhar os filme que ele for mandado detalhar
	 * @param posicao
	 * @param filme
	 * @return
	 */
	public String formataFilme(int posicao, String filme) {
		return posicao + " - " + filme;
	}
	/**
	 * este metódo vai detalhar o filme 
	 * @param posicao
	 * @return
	 */
	public String toString(int posicao) {
		posicao = posicao-1;
		String hotliste= "";
		if(posicao<=-1 || posicao>=100) {
			return "POSIÇÃO INVÁLIDA!";
		}
		if(this.filmes[posicao]==null) {
			return "";
		}
		if(EmHotlist(filmes[posicao].getFilme())==true){
			hotliste = "🔥 ";
		}	
		if(filmes[posicao].getAno()==null || filmes[posicao].getAno()=="") {
			return hotliste + filmes[posicao].getFilme() + filmes[posicao].getAno() + "\n" + filmes[posicao].getLocal();
		}
		return hotliste + filmes[posicao].getFilme() + ", "+ filmes[posicao].getAno() + "\n" + filmes[posicao].getLocal();
	}
	/**
	 * este metodo fucinona no pode da amizade
	 * mentira, este metodo fuciona para ver se um filme é igual a outro filme
	 * @param nome
	 * @param ano
	 * @return
	 */
	public boolean equals1(String nome, String ano) {
		for (int i=0; i<100; i++) {
			if(filmes[i] != null) {
				if((nome.equals(filmes[i].getFilme())) && (ano.equals(filmes[i].getAno()))){
					return false;
				}
			}
		}
		return true;
	}
	
	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if (posicao>100 || posicao<1) {
			return "POSIÇÃO INVÁLIDA";
		}
		if(nome==null || nome==""){
			return "FILME INVALIDO";
		}
		if(local==null || local==""){
			return "FILME INVALIDO";
		}
		if(equals1(nome,ano)==false) {
			return "FILME JÁ ADICIONADO";
		}
		filmes[posicao-1] = new Filme(nome, ano, local);
		return "FILME ADICIONADO";
	}
	public void cadastraHotlist(int posicao, int filme) {
		this.Hotlist[posicao-1] = filmes[filme-1].getFilme();
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
