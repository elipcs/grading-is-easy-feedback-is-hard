package filmnow;

import java.util.Objects;
/**
 * Classe que representa um filme com nome, ano de lançamento e local de exibição e um atributo para indicar se o filme está na hotlist.
 * 
 * @author [REDACTED]
 */
public class Filme {
	/**
	 * Nome do filme
	 */
	private String nome;
	/**
	 * Ano de lançamento
	 */
	private String ano;
	/**
	 * Local de exibição do filme
	 */
	private String local;
	/**
	 * Filme esta na hotlist ou nao
	 */
	private boolean hot;
	/**
	 * Cria um filme a partir do seu nome, do ano e do local
	 * Inicia o filme com o filme fora da hotlist
	 * @param nome nome do filme
	 * @param ano ano de lançamento
	 * @param local local de exibição
	 */
	public Filme( String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.hot = false;
	}
	
	public boolean isHot() {
		return hot;
	}

	public void setHot(boolean hot) {
		this.hot = hot;
	}
	public String getNome() {
		return nome;
	}

	public void setNome(String nome) {
		this.nome = nome;
	}

	public String getAno() {
		return ano;
	}

	public void setAno(String ano) {
		this.ano = ano;
	}

	public String getLocal() {
		return local;
	}

	public void setLocal(String local) {
		this.local = local;
	}
	/**
	 * Metodo que retorna de forma textual o nome e o ano do filme
	 * caso nao tenha ano, retorna apenas o nome
	 * @return nome / nome +", "+ano
	 */
	public String nomeAno() {
		if(ano == null || ano == "") {
			return nome;
		}else {
			return nome +", "+ano;
		}
	}
	/**
	 * Retorna a representação textual do filme, no formato:
	 * nome, ano local
	 * caso hot seja true retorna com um emoji de fogo
	 * 🔥nome, ano local
	 */
	@Override
	public String toString() {
		if(this.hot == true) {
			return "🔥"+this.nomeAno()+ "\n" + local;
		}
		return this.nomeAno() + "\n" + local;
	}

	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}
	
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes e 10 filme na Hotlist(filmes em alta).
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	/**
	 * Tamanho máximo da lista de filmes.
	 */
	private static final int TAMANHO = 100;
	/**
	 * Tamanho máximo da lista de filmes.
	 */
	private static final int TAMANHOHOTLIST = 10;
	/**
     * Array que armazena a lista de filmes.
     */
	private Filme[] filme;
	/**
     * Array que armazena a lista de filmes.
     */
	private Filme[] hotList;

	/**
     * Cria o sistema FilmNow.
     * <Inicializa a lista de filmes e a lista de filmes em alta (hotlist)
     */
	public FilmNow() {
		this.filme = new Filme[TAMANHO];
		this.hotList = new Filme[TAMANHOHOTLIST];
	}
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filme;
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		
		return filme[posicao];
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public void cadastraFilme(int posicao, String nome, String ano, String local) {
		this.filme[posicao] = new Filme(nome, ano, local);
	}
    /**
     * Realiza a verificação da validade de um filme e se ele ja foi adicionado.
     * 
     * @param posicao Posição onde o filme será adicionado.
     * @param nome Nome do filme.
     * @param ano Ano de lançamento do filme.
     * @param local Local onde o filme pode ser assistido.
     * @return Uma mensagem que representa o resultado da verificação
     *         - "POSIÇÃO INVÁLIDA" se a posição estiver fora dos limites,
     *         - "FILME INVALIDO" se o nome ou local estiver vazio,
     *         - "FILME JA ADICIONADO" se o filme já estiver adicionado na lista,
     *         - "OK" se o filme está valido.
     */
	public String verificaFilme(int posicao, String nome, String ano, String local) {
		if(posicao > 100 || posicao <= 0) {
			return "POSIÇÃO INVÁLIDA";
		}
		if (nome.equals("") || local.equals("")) {
			return "FILME INVALIDO";
		}
		for(int i = 0;i<TAMANHO;i++) {
			Filme filmeAtual = filme[i];
			
			if(filmeAtual != null && nome.equals(filmeAtual.getNome()) && ano.equals(filmeAtual.getAno())) {
				return "FILME JA ADICIONADO";
			}
		}
		return "OK";
	}
    /**
     * Verifica se um filme já está presente na hotlist.
     * 
     * @param filmeAdicionar O filme a ser verificado.
     * @param posicaoHot Posição na hotlist onde o filme será adicionado.
     * @return true, se o filme pode ser adicionado à hotlist (não está já presente), e false, caso contrário.
     */
	public boolean verificaFilmeHot(Filme filmeAdicionar,int posicaoHot) {
		for(Filme filmeAtual:this.hotList) {
			if(filmeAdicionar.equals(filmeAtual)) {
				return false;
			}
		}
		return true;
	}
	
	public Filme[] getHotList() {
		return hotList;
	}
	 /**
     * Obtém o filme na hotlist em uma posição específica.
     * 
     * @param posicao Posição na hotlist.
     * @return O filme na posição passada.
     */
	public Filme getFilmeHotList(int posicao) {
		return hotList[posicao];
	}
	/**
     * Adiciona um filme à hotlist em uma posição específica.
     * Se já houver um filme na posição, o filme anterior é removido da hotlist e o novo filme é adicionado.
     * 
     * @param filme O filme a ser adicionado à hotlist.
     * @param posicaoHot Posição na hotlist onde o filme deve ser adicionado.
     */
	public void cadastraFilmeHotList(Filme filme, int posicaohot) {
		
		if(this.hotList[posicaohot]!= null) {
			this.hotList[posicaohot].setHot(false);
		}
		this.hotList[posicaohot] = filme;
		filme.setHot(true);
		
	}
	  /**
     * Remove um filme da hotlist na posição passada.
     * 
     * @param posicaoHot Posição na hotlist do filme a ser removido.
     */
	public void removeHotList(int posicaohot) {
		Filme filme = this.hotList[posicaohot];
		filme.setHot(false);
		this.hotList[posicaohot] = null;
	}
	/**
     * Retorna uma descrição detalhada do filme em uma determinada posição.
     * 
     * @param posicao Posição do filme na lista.
     * @return Uma string com a descrição do filme, "POSIÇÃO INVÁLIDA" caso seja uma posicao invalida, ou uma string vazia se não houver filme na posição.
     */
	public String detalhaFilme(int posicao) {
		if(posicao > 100 || posicao <= 0) {
			return "POSIÇÃO INVÁLIDA";
		}
		Filme filme = this.getFilme(posicao);
		if(filme == null) {
			return "";
		}
		return filme.toString();
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

		case "R":
			removerHot(fn, scanner);
