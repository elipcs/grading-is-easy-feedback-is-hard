package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
    
    private static final int TAMANHO = 100;
    private static final int HOT_TAMANHO = 10;
    
    private Filme[] filmes; // uma representacao simploria da lista de filmes
    private Filme[] hotList; // Lista para armazenar os filmes marcados como Hot

    /**
     * Cria o FilmNow.
     */
    public FilmNow() {
        this.filmes = new Filme[TAMANHO];
        this.hotList = new Filme[HOT_TAMANHO];
    }
    
    /**
     * Acessa a lista de filmes mantida.
     * @return O array de filmes.
     */
    public Filme[] getFilmes() {
        return this.filmes;
    }

    /**
     * Acessa os dados de um filme específico.
     * @param posicao Posição do filme no sistema.
     * @return Dados do filme. Null se não há filme na posição.
     */
    public Filme getFilme(int posicao) {
        return filmes[posicao];
    }

    /**
     * Acessa a lista de filmes marcados como Hot.
     * @return O array de filmes Hot.
     */
    public Filme[] getHotFilmes() {
        return this.hotList;
    }

    /**
     * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
     * @param posicao Posição do filme.
     * @param nome Nome do filme.
     * @param ano Ano de lançamento do filme.
     * @param local Local onde o filme pode ser assitido.
     */
    public String cadastraFilme(int posicao, String nome, String ano, String local) {
    	posicao--;
        if (posicao < 0 || posicao >= TAMANHO) {
            return "POSIÇÃO INVALIDA!" ;
        }
        if (nome == null || local == null || nome.trim() == "" || local.trim() == "") {
            return "FILME INVALIDO!";
        }
        Filme novoFilme;
        
        if(ano.trim().equals("")) {
        	novoFilme = new Filme(nome, local);
        }
        else {
        	novoFilme = new Filme(nome, ano, local);
        }
     
        if (existeFilme(novoFilme)) {
            return "FILME JA ADICIONADO!";
        }           
        this.filmes[posicao] = novoFilme;
        return "FILME ADICIONADO!";
    }
    /**
     * Verifica se o filme já está na lista de filmes.
     * 
     * @param novoFilme O filme a ser verificado.
     * @return true se o filme já existe na lista, false caso contrário.
     */        
    private boolean existeFilme(Filme novoFilme) {
        for(Filme filme : filmes) {
            if(novoFilme.equals(filme)) {
                return true;
            }
        }
        return false;
    }
    /**
     * Retorna os detalhes do filme na posição especificada.
     * 
     * @param posicao A posição do filme no array.
     * @return Uma string com os detalhes do filme, ou uma mensagem de erro se a posição for inválida.
     */
    public String detalhaFilme(int posicao){
    	posicao--;
    	if (posicao < 0 || posicao >= TAMANHO) {
            return "POSIÇÃO INVALIDA!";
        }
  
        Filme filme = getFilme(posicao);
        
        if (filme == null) {
            return "POSIÇÃO INVALIDA!";
        }

        String detalhe = filme.toString();
        if (isHot(filme)) {
            detalhe = "🔥 " + detalhe;
        }
        
        return detalhe;
    }
    /**
     * Verifica se o filme está na lista "Hot".
     * 
     * @param filme O filme a ser verificado.
     * @return true se o filme estiver na lista "Hot", false caso contrário.
     */
    private boolean isHot(Filme filme) {
        for (Filme hotFilme : hotList) {
            if (filme.equals(hotFilme)) {
                return true;
            }
        }
        return false;
    }
    
    /**
     * Atribui um filme à lista de "Hot" se ele já estiver cadastrado.
     * 
     * @param posicao Posição do filme na lista de filmes.
     * @return Mensagem de sucesso ou erro.
     */
    public String atribuirHot(int posicao) {
        Filme filme = getFilme(posicao);
        if (filme == null) {
            return "FILME NÃO ENCONTRADO!";
        }
        for (int i = 0; i < HOT_TAMANHO; i++) {
            if (hotList[i] == null) {
                hotList[i] = filme;
                return "ADICIONADO À HOTLIST NA POSIÇÃO " + (i + 1) + "!";
            }
        }
        return "HOTLIST CHEIA!";
    }

    /**
     * Remove um filme da lista Hot.
     * @param posicaoHot Posição na lista Hot.
     * @return True se a remoção foi bem-sucedida, false caso contrário.
     */
    public String removerHot(int posicao) {
        if (posicao <= 0 || posicao >= HOT_TAMANHO || hotList[posicao] == null) {
            return "FILME NÃO ENCONTRADO NA HOTLIST!";
        }
        hotList[posicao] = null;
        return "FILME REMOVIDO DA HOTLIST!";
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
