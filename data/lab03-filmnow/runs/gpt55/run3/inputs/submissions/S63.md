package filmnow;

import java.util.Objects;

public class Filme {

	private String nome;
	private String ano;
	private String local;
	private boolean ehHot;

	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.ehHot = false;
	}

	public String getNome() {
		return nome;
	}

	public String getAno() {
		return ano;
	}

	public String getLocal() {
		return local;
	}
	
	public boolean isEhHot() {
		return ehHot;
	}

	public boolean setEhHot(boolean ehHot) {
		return this.ehHot = ehHot;
	}

	@Override
	public String toString() {
	    if (ano.trim().isEmpty()) {
	        return String.format("%s\n%s", nome, local); // Formata corretamente quando o ano está vazio
	    }
	    return String.format("%s, %s\n%s", nome, ano, local); // Formata corretamente quando o ano está presente
	}

	@Override
	public int hashCode() {
		return Objects.hash(ano, local, nome);
	}

	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
	    return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);	
	    }

}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes.
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {

	static final int TAMANHO = 100;

	private Filme[] filmes; // uma representacao simploria da lista de filmes
	private Filme[] hotList;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[10];
	}

	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * 
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		 if (posicao < 1 || posicao > TAMANHO) {
	            return null; 
	        }
	     return filmes[posicao - 1];

	}
	
	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve
	 * o anterior.
	 * 
	 * @param posicao Posição do filme.
	 * @param nome    Nome do filme.
	 * @param ano     Ano de lançamento do filme.
	 * @param local   Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		String saidaEsperada = "";
		boolean adicionado = false;
		boolean posicaoValida = true;
		Filme filme = new Filme(nome, ano, local);

		if (posicao < 1 || posicao > FilmNow.TAMANHO) {
			saidaEsperada = "POSIÇÃO INVÁLIDA";
			posicaoValida = false;
		} else {
			saidaEsperada = (nome.trim().isEmpty() || local.trim().isEmpty()) ? "FILME INVÁLIDO" : "";

			for (int i = 0; i < filmes.length; i++) {
				if (filmes[i] != null && filmes[i].equals(filme)) {
					saidaEsperada = "FILME JÁ ADICIONADO";
					adicionado = true;
				}
			}

		}

		if (!(nome.trim().isEmpty() || local.trim().isEmpty()) && !(adicionado) && posicaoValida) {
			filmes[posicao - 1] = new Filme(nome, ano, local);
			saidaEsperada = "FILME ADICIONADO";
		}

		return saidaEsperada;

	}

	public Filme[] getHot() {
		return this.hotList.clone();
	}

	/**
	 * Adiciona um filme em uma lista de favoritos. Se já existir filme na posição, sobrescreve
	 * o anterior.
	 * 
	 * @param posicao vai ser a posição que ela deve ocupar na lista de favoritos
	 * @param posicaoFilme indica qual filme vai ser adicionado 
	
	 */
	public String cadastraHot(int posicao, int posicaoFilme) {
		 String saida = "";
		    if (posicao < 0 || posicao >= hotList.length) {
		        return "POSIÇÃO INVÁLIDA";
		    }

		    Filme filme = getFilme(posicaoFilme);
		    if (filme == null) {
		        return "FILME NÃO ENCONTRADO";
		    }

		    if (!verificarFilmeHotList(filme)) {
		        hotList[posicao] = filme;
		        saida = "Hot adicionada na posição " + posicao;
		    } else {
		        saida = "Filme já adicionado na hot";
		    }

		    return saida;

	}
	
	/**
	 * Verificar se algum filme esta na lista de favoritos(hotList)
	 * @param filme que é o filme a ser verificado
	 * @return true se esse filme ta na lista e false se não 
	
	 */

	private boolean verificarFilmeHotList(Filme filme) {
		for (Filme hot : hotList) {
			if (filme.equals(hot)) {
				return true;
			}
		}
		return false;

	}

	/**
	 * Vai mostrar se o filme ta como favorito(hotList) ou é só um filme normal
	 * @param posicao que é a posição do filme
	 * @return a representação em String de um filme marcado como favorito(hotList) ou não
	
	 */
	public String verHotOuFilme(int posicao) {
		if (posicao < 0 || posicao >= filmes.length) {
	        return "POSIÇÃO INVÁLIDA";
	    }
	    
	    Filme filme = getFilme(posicao);
	    if (filme == null) {
	        return "FILME NÃO ENCONTRADO";
	    }
	    
		boolean ehHot = filme.isEhHot();
		ehHot = filme.setEhHot(verificarFilmeHotList(filme));
		return ehHot ? "🔥 " + filme : filme.toString();

	}

	/**
	 * Remove um filme da lista de favoritos(hotList)
	 * @param posicao que é a posição do filme
	
	 */
	public void removendoHot(int posicao) {
		if (hotList[posicao] != null ) {
			hotList[posicao] = null;
		}else {
			System.out.println("FILME NÃO ENCONTRADO");
		}
	}

}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
