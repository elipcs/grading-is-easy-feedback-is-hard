import java.util.Objects;

public class Filme {

		this.anoDeLancamento = ano;
		this.nome = nome;
		this.local = local;
		this.hot = "";
	}
	/**
	 * instancializa um filme vazio
	 */

		this.anoDeLancamento = "";
		this.nome = "";
		this.local = "";
		this.hot = "";
	}

	 * Retorna o status "hot" do filme.
	 *
	 * @return Uma string indicando se o filme está na HotList.
	 */
	public String getHot() {
		return this.hot;
	}
	/**
	 * Define o status "hot" do filme.
	 *
	 * @param hot A string que indica se o filme está na HotList.
	 */
	public void setHot(String hot) {
		this.hot = hot;
	}
	
	/**
	 * Retorna o ano de lançamento do filme.
	 *
	 * @return O ano de lançamento do filme.
	 */
	public String getAnoDeLancamento() {
		return this.anoDeLancamento;
	}
	
	/**
	 * Retorna o nome do filme.
	 *
	 * @return O nome do filme.
	 */
	public String getNome() {
		return this.nome;
	}
	
	/**
	 * Retorna o local de origem do filme.
	 *
	 * @return O local de origem do filme.
	 */
	public String getLocal() {
		return this.local;
	}
	
	/**
	 * Retorna uma representação textual do filme.
	 *
	 * @return Uma string contendo o status "hot", o nome, o ano de lançamento e o local do filme.
	 */
	@Override
	public String toString(){
		return  this.hot + this.nome +", " +this.anoDeLancamento+ "\n"+ this.local;
	}
	
	/**
	 * Gera um código hash para o filme com base no ano de lançamento e no nome.
	 *
	 * @return O código hash gerado.
	 */
	@Override
	public int hashCode() {
		return Objects.hash(anoDeLancamento, nome);
	}
	
	/**
	 * Verifica se este filme é igual a outro objeto.
	 *
	 * @param obj O objeto a ser comparado com este filme.
	 * @return true se os objetos forem iguais; false caso contrário.
	 */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return anoDeLancamento == other.anoDeLancamento && Objects.equals(nome, other.nome);
	}

package filmnow;


/**
 * Classe que representa um FilmNow
 */
public class FilmNow {
	private int tamanhoFilmes;
	private int tamanhoHotList;
	private Filme[] filmes;
	private Filme[] hotList; 

	/**
	 * Construtor que instancializa um FilmNow no sistema
	 */
	public FilmNow() {
		this.tamanhoFilmes = 100;
		this.tamanhoHotList = 10;
		this.filmes = new Filme[tamanhoFilmes];
		this.hotList= new Filme[tamanhoHotList];
	}
	/**
	 * Cadastra um novo filme no sistema.
	 *
	 * @param posicao A posição onde o filme será cadastrado.
	 * @param nome O nome do filme.
	 * @param ano O ano de lançamento do filme.
	 * @param local O local de origem do filme.
	 * @return Mensagem indicando se o cadastro foi bem-sucedido ou não.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		Filme filmeASerCadastrado = new Filme(nome,ano,local);
		
		if(posicaoInvalida(posicao)){
			return "posicao invalida, escolha entre 1 e 100";
		}
		if(nome == "") {
			return "FILME INVALIDO";
		}
		if(local == "") {
			return "FILME INVALIDO";
		}
		
		
		for (int i=0;i<this.filmes.length;i++) {
			if (this.filmes[i] != null) {
				if(filmeASerCadastrado.equals(this.filmes[i])) {
					return "Existe ja";
				}
			}
		}

		this.filmes[posicao -1] = filmeASerCadastrado;

		return "FILME ADICIONADO";

	}
	
	/**
	 * Retorna uma lista de filmes em formato de string.
	 *
	 * @return Um array de strings contendo os nomes dos filmes cadastrados.
	 */
	public String[] getFilmesEmString() {
		String[] arrayString = new String[this.tamanhoFilmes];
		for (int i = 0; i < this.filmes.length; i++) {
			if (this.filmes[i] != null) {
				arrayString[i] = (i + 1)  +" - "+ this.filmes[i].getNome() ;
			}
			
		}
		return arrayString;
	}
	
	/**
	 * Retorna os detalhes de um filme específico.
	 *
	 * @param posicao A posição do filme no sistema.
	 * @return Uma string com os detalhes do filme ou uma mensagem de erro.
	 */
	public String getFilmeDetalhado(int posicao) {
		String filmeDetalhado;
		if(posicaoInvalida(posicao)) {
			return "POSIÇÃO INVÁLIDA";
		}
		if(this.filmes[posicao-1]!=null) {
			filmeDetalhado = this.filmes[posicao-1].toString();
		}else {
			filmeDetalhado = "";
		}
		return filmeDetalhado;
	}
	
	

	private boolean posicaoInvalida(int posicaoFilme) {
		if((posicaoFilme < 1) || (posicaoFilme > 100) ) {
			return true;
		}else {
			return false;
		}
	}
	
	/**
	 * Atribui um filme à HotList.
	 *
	 * @param posicaoFilme A posição do filme no sistema.
	 * @param posicaoHotList A posição onde o filme será colocado na HotList.
	 * @return Mensagem indicando se a atribuição foi bem-sucedida ou não.
	 */
	public String setFilmeHotlist(int posicaoFilme,int posicaoHotList ){

		posicaoHotList= posicaoHotList -1;

		posicaoFilme = posicaoFilme-1;
		
		if(this.filmes[posicaoFilme]==null) {
			return "POSIÇÃO INVÁLIDA";
		}
		
		if(this.hotList[posicaoHotList]!=null) {
			this.hotList[posicaoHotList].setHot("");
		}
		
		
		for(int i = 0; i< this.hotList.length;i++) {
			
			if(this.hotList[i]!=null) {
				
				if(this.hotList[i].equals(this.filmes[posicaoFilme])){
					return "ja esta na hotlist";
				};
			}
			
		}
		
		this.hotList[posicaoHotList]= this.filmes[posicaoFilme];
		this.hotList[posicaoHotList].setHot("🔥");
		return "ADICIONADO Á HOTLIST NA POSIÇÃO " + (posicaoHotList +1);
		
		
	}
	
	/**
	 * Remove um filme da HotList.
	 *
	 * @param posicao A posição do filme na HotList.
	 */
	public void removeFilmeHotList(int posicao) {
		this.hotList[posicao -1].setHot("");
		this.hotList[posicao -1] = null;
	}
	
	/**
	 * Retorna a HotList em formato de string.
	 *
	 * @return Um array de strings contendo os nomes dos filmes na HotList.
	 */
	public String[] toStringHotList() {

		String[] hotListString = new String[this.tamanhoHotList];
		for (int i = 0; i < this.tamanhoHotList; i++) {
			if (this.hotList[i] != null) {
				
				hotListString[i] = (i +1)  +" - "+ this.hotList[i].getNome();
			}
		}
		return hotListString;
	}
	
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
