package filmnow;

import java.util.Objects;

/**
 * Classe que representa o tipo Filme, possuindo nome, ano de lançamento e local de distribuição, 
 * além de um status informando se o filme em questão faz parte da lista Hot. 
 * 
 * @author [REDACTED]
 */
public class Filme {
	
	private String nome;
	private String ano;
	private String local;
	private boolean status;
	
	/**
	 * Construtor da minha classe, responsável por criar o filme a ser listado.
	 * @param nome Nome do filme
	 * @param ano Ano de lançamento
	 * @param local Local de reprodução
	 */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.status = false;
	}
	
	public String getNome() {
		return this.nome;
	}
	
	public String getAno() {
		return this.ano;
	}
	
	public String getLocal() {
		return this.local;
	}

	public boolean getStatus() {
		return this.status;
	}
	
	public void setStatus() {
		this.status = true;
	}
	
	public void removeStatus() {
		this.status = false;
	}
	
	@Override
	public int hashCode() {
		return Objects.hash(ano, local, nome);
	}

	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return Objects.equals(nome, other.nome) && Objects.equals(ano, other.ano);
	}

	/**
	 * Planeja e executa a formatação da representação gráfica do objeto Filme, através de seu status
	 * de filme normal ou do tipo Hot, assim como a presença ou não de um ano de lançamento cadastrado.
	 * @return filme formatado, com ou sem seu status e seu ano de lançamento
	 */
	public String formataFilme() {
		if (status == true) {
			if (ano == "") {
				return "🔥 " + this.nome + "\n" + this.local;
			} else {
				return "🔥 " + this.nome + ", " + this.ano + "\n" + this.local;
			}
		} else {
			if (ano == "") {
				return this.nome + "\n" + this.local;
			} else {
				return this.nome + ", " + this.ano + "\n" + this.local;
			}
		}
	}
	
	@Override
	public String toString() {
		return this.nome + ", " + this.ano;
	}
	
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes e séries na lista
 * normal e até 10 filmes na exclusiva lista de filmes Hot, uma lista para os filmes e séries 
 * que o usuário está assistindo ou precisa assistir logo. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100; // tamanho da lista tradicional de filmes.
	
	private Filme[] filmes; // uma representação simploria da lista de filmes

	private static final int TAM_HOT = 10; // tamanho da lista HotFilmes.
	
	private Filme[] hotList; // uma representação da exclusiva lista de filmes Hot.
	
	/**
	 * Cria o FilmNow, inicializando as listas original e do tipo Hot.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[TAM_HOT];
	}
	
	/**
	 * Verifica se um filme de mesmo nome e ano de lançamento já existe na lista de filmes. 
	 * @param filme Filme em análise
	 * @return true para equivalência, false para divergência
	 */
	public boolean temFilme(Filme filme) {
		for (Filme film : filmes) {
			if (film != null && film.getNome().equals(filme.getNome()) && film.getAno().equals(filme.getAno())) {
				return true;
			} 
		}
		return false;
	}
	
	/**
	 * Retorna uma lista com os filmes não nulos formatados.
	 * @return O array de filmes formatados.
	 */
	public String[] getFilmes() {
		String[] filmesFormat = new String[TAMANHO];
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				filmesFormat[i] = (i + 1) + " - " + filmes[i].getNome();
			}
		}
		return filmesFormat;
	}
	
	/**
	 * Acessa a lista de filmes original, sem formatação.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmesOriginal() {
		return filmes;
	}
	
	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public String getFilme(int posicao) {
		if (posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		} else if (filmes[posicao - 1] == null) {
			return "";
		} else { 
			return filmes[posicao - 1].formataFilme();
		}
	}
	
	/**
	 * Verifica se um filme de mesmo nome e ano de lançamento já existe na lista de filmes Hot. 
	 * @param filme Filme em análise
	 * @return true para equivalência, false para divergência
	 */
	public boolean temFilmeHot(Filme filme) {
		for (Filme hot : hotList) {
			if (hot != null && hot.getNome().equals(filme.getNome()) && hot.getAno().equals(filme.getAno())) {
				return true;
			}
		}
		return false;
	}
	
	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		Filme movie = new Filme(nome, ano, local);
		
		if (posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";	
		} else if (nome.equals("") || local.equals("")) {
				return "FILME INVALIDO";
		} else if (temFilme(movie)) {
			return "FILME JÁ ADICIONADO";
		} else {
			this.filmes[posicao - 1] = movie;
			return "FILME ADICIONADO";
		}
	}
	
	/**
	 * Acessa um filme do tipo Hot através de sua posição.
	 * @param posicao Posição do filme.
	 * @return Filme Hot acessado
	 */
	public String getFilmeHot(int posicao) {
		return hotList[posicao - 1].formataFilme();
	}
	
	/**
	 * Retorna o array da lista Hot.
	 * @return Array de HotList.
	 */
	public Filme[] getHotList() {
		return this.hotList;
	}
	
	/**
	 * Adiciona um filme à categoria Hot, bem como altera seu status pessoal.
	 * @param posicaoNormal Posição do filme na lista de filmes tradicional.
	 * @param posicaoListaHot Posição que o filme vai assumir na lista Hot.
	 * @return mensagem de confirmação
	 */
	public String atribuiHot(int posicaoNormal, int posicaoListaHot) {
	    if (posicaoNormal < 1 || posicaoNormal > 100) {
	    	return "POSIÇÃO INVÁLIDA";
	    } 
	    if (posicaoListaHot < 1 || posicaoListaHot > 100) {
	    	return "POSIÇÃO INVÁLIDA";
	    } 
	    if (temFilmeHot(filmes[posicaoListaHot - 1])) {
	    	return "FILME JÁ ADICIONADO";
	    }
	    if (hotList[posicaoListaHot - 1] != null) {
	    	hotList[posicaoListaHot - 1].removeStatus();
	    }
	    hotList[posicaoListaHot - 1] = filmes[posicaoNormal - 1];
	    filmes[posicaoNormal - 1].setStatus();
		return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoListaHot;
	}
	
	/**
	 * Remove um filme de certa posição na lista Hot. 
	 * @param posicaoListaHot Posição que ele ocupa na lista Hot.
	 * @return mensagem de confirmação
	 */
	public String retiraHot(int posicaoListaHot) {
		Filme removido = hotList[posicaoListaHot - 1];
	    if (removido != null && posicaoListaHot >= 1 && posicaoListaHot <= 10) {
	        removido.removeStatus();
	        hotList[posicaoListaHot - 1] = null;
	        return "REMOVIDO COM SUCESSO";
	    } else {
	        return "POSIÇÃO INVÁLIDA";
	    }
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

public class MainFilmNow {

		FilmNow fn = new FilmNow();

		Scanner scanner = new Scanner(System.in);
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.

		System.out.println(
				"\n---\nMENU\n" +

						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" +
						"(E)Exibir HotList\n" + 
						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)air\n" + 
						"\n" + 
						"Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opcao   Opção digitada.
	 * @param fn  O sistema FilmNow que estamos manipulando.
	 * @param scanner Objeto scanner para o caso do comando precisar de mais input.

			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibeHot(fn);
			break;
		case "H":
			adicionaHot(fn, scanner);
			break;
		case "R":
			removeHot(fn, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("OPÇÃO INVÁLIDA!");
		}
	}
