package filmnow;

/**
 * Representa um filme no sistema FilmNow.
 * 
 * @author [REDACTED]
 */
public class Filme {
    
    /** Nome do filme */
    private String nomeFilme;
    
    /** Local onde o filme pode ser assistido */
    private String local;
    
    /** Ano de lançamento do filme */
    private String anoLancamento;
    
    /** Posição do filme no sistema */
    private int posicao;

     * Cria uma instância de Filme.
     *

        this.local = local;
        this.nomeFilme = nomeFilme;
        this.posicao = posicao;
    }

     * Obtém o nome do filme.
     * 
     * @return Nome do filme.
     */
    public String getNome() {
        return this.nomeFilme;    
    }

     * Obtém o ano de lançamento do filme.
     * 
     * @return Ano de lançamento do filme.
     */
    public String getAno() {
        return this.anoLancamento;
    }

     * Obtém a posição do filme no sistema.
     * 
     * @return Posição do filme.
     */
    public int getPosicao() {
        return this.posicao;
    }

     * Obtém o local onde o filme pode ser assistido.
     * 
     * @return Local do filme.
     */
    public String getLocal() {
        return this.local;
    }
    
    /**
     * Compara este filme com outro objeto para determinar se são iguais.
     * 
     * <p>Dois filmes são considerados iguais se tiverem o mesmo nome e ano de lançamento.</p>
     * 
     * @param obj O objeto a ser comparado.
     * @return True se o objeto for um Filme e tiver o mesmo nome e ano de lançamento, false caso contrário.
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (obj == null || getClass() != obj.getClass()) {
            return false;
        }
        Filme filme = (Filme) obj;
        return nomeFilme.equals(filme.nomeFilme) && anoLancamento.equals(filme.anoLancamento);

     * Retorna uma representação em string do filme.
     * 
     * <p>A representação inclui o nome do filme, o ano de lançamento (se disponível) e o local onde pode ser assistido.</p>
     * 
     * @return Representação em string do filme.
     */
    @Override

        StringBuilder sb = new StringBuilder();
        sb.append(nomeFilme);
        
        if (anoLancamento != null && !anoLancamento.isEmpty()) {
            sb.append(" ").append(anoLancamento);
        }
        
        if (local != null && !local.isEmpty()) {
            if (anoLancamento == null || anoLancamento.isEmpty()) {
                sb.append(" ");
            } else {
                sb.append(" ");
            }
            sb.append(local);
        }
        
        return sb.toString().trim();
    }

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir até 100 filmes.
 * 
 * @author [REDACTED]
 */
public class FilmNow {

    /** Número máximo de filmes que o sistema pode manter */
    private static final int FILMES = 100;
    
    /** Número máximo de filmes na HotList */
    private static final int HOTLIST = 10;
    
    /** Array para armazenar os filmes cadastrados */
    private Filme[] filmes;
    
    /** Array para armazenar os filmes na HotList */
    private Filme[] hotList;

    /**
     * Cria uma instância do FilmNow.
     * Inicializa os arrays de filmes e HotList.
     */
    public FilmNow() {
        this.filmes = new Filme[FILMES];
        this.hotList = new Filme[HOTLIST];
    }

    /**
     * Acessa a lista de filmes mantida.
     * 
     * @return O array de filmes.
     */
    public Filme[] getFilmes() {
        return this.filmes.clone();
    }

    /**
     * Define a lista de filmes.
     * 
     * @param filmes Array de filmes a ser definido.
     */
    public void setFilmes(Filme[] filmes) {
        this.filmes = filmes;
    }

    /**
     * Acessa os dados de um filme específico.
     * 
     * @param posicao Posição do filme no sistema.
     * @return Dados do filme. Retorna null se não há filme na posição.
     */
    public Filme getFilme(int posicao) {
        if (posicao < 1 || posicao > FILMES) {
            System.out.println("Posição inválida!");
            return null;
        }
        return filmes[posicao - 1];
    }

    /**
     * Adiciona um filme em uma posição específica. Se já existir um filme na posição, sobrescreve o anterior.
     * 
     * @param posicao Posição do filme.
     * @param nomeFilme Nome do filme.
     * @param anoLancamento Ano de lançamento do filme.
     * @param local Local onde o filme pode ser assistido.
     * @return Mensagem indicando o sucesso ou falha da operação.
     */
    public String cadastraFilme(int posicao, String nomeFilme, String anoLancamento, String local) {
        if (posicao < 1 || posicao > FILMES) {
            return "POSIÇÃO INVÁLIDA";
        }

        if (nomeFilme == null || nomeFilme.trim().isEmpty() || local == null || local.trim().isEmpty()) {
            return "FILME INVALIDO";
        }

        if (!verificaFilme(nomeFilme, anoLancamento)) {
            return "FILME JÁ ADICIONADO";
        }

        this.filmes[posicao - 1] = new Filme(posicao, nomeFilme, anoLancamento, local);
        return "FILME ADICIONADO";
    }

    /**
     * Verifica se um filme já está cadastrado no sistema.
     * 
     * @param nomeFilme Nome do filme.
     * @param anoLancamento Ano de lançamento do filme.
     * @return True se o filme não estiver cadastrado, false caso contrário.
     */
    private boolean verificaFilme(String nomeFilme, String anoLancamento) {
        for (int i = 0; i < FILMES; i++) {
            Filme filmeAtual = filmes[i];
            if (filmeAtual != null && nomeFilme.equals(filmeAtual.getNome()) && anoLancamento.equals(filmeAtual.getAno())) {
                return false;
            }
        }
        return true;
    }

    /**
     * Mostra todos os filmes cadastrados.
     */
    public void mostrarFilmes() {
        for (int i = 0; i < FILMES; i++) {
            if (filmes[i] != null) {
                System.out.println((i + 1) + " - " + filmes[i].getNome());
            }
        }
    }

    /**
     * Detalha um filme específico, incluindo seu status na HotList.
     * 
     * @param posicao Posição do filme.
     * @return Dados detalhados do filme, incluindo um símbolo se o filme estiver na HotList.
     */
    public String detalharFilme(int posicao) {
        if (posicao < 1 || posicao > FILMES) {
            return "POSIÇÃO INVÁLIDA";
        }

        Filme filme = filmes[posicao - 1];
        if (filme == null) {
            return ""; // Retorna vazio se não houver filme na posição
        }

        boolean isHot = false;
        for (Filme f : hotList) {
            if (f != null && f.equals(filme)) {
                isHot = true;
                break;
            }
        }
        String hotSymbol = isHot ? "🔥 " : "";
        if (filme.getAno() == null || filme.getAno().isEmpty()) {
            return hotSymbol + filme.getNome() + "\n" + filme.getLocal();
        } else {
            return hotSymbol + filme.getNome() + ", " + filme.getAno() + "\n" + filme.getLocal();
        }
    }

    /**
     * Adiciona um filme à HotList em uma posição específica.
     * 
     * @param indiceFilme Índice do filme na lista de filmes.
     * @param posicaoHotList Posição na HotList onde o filme será adicionado.
     */
    public void adicionarHot(int indiceFilme, int posicaoHotList) {
        if (posicaoHotList < 1 || posicaoHotList > HOTLIST) {
            System.out.println("POSIÇÃO INVÁLIDA");
            return;
        }
        posicaoHotList--;
        for (int i = 0; i < hotList.length; i++) {
            if (hotList[i] != null && hotList[i].equals(filmes[indiceFilme - 1])) {
                System.out.println("O FILME JÁ ESTÁ NA HOTLIST");
                return;
            }
        }
        hotList[posicaoHotList] = filmes[indiceFilme - 1];
        System.out.println("ADICIONADO À HOTLIST NA POSIÇÃO " + (posicaoHotList + 1) + "!");
    }

    /**
     * Remove um filme da HotList de uma posição específica.
     * 
     * @param posicaoHotList Posição na HotList do filme a ser removido.
     */
    public void removerHot(int posicaoHotList) {
        if (posicaoHotList < 1 || posicaoHotList > HOTLIST || hotList[posicaoHotList - 1] == null) {
            System.out.println("Posição inválida ou não há filme na posição especificada!");
            return;
        }
        hotList[posicaoHotList - 1] = null;
        System.out.println("REMOVIDO DA HOTLIST NA POSIÇÃO " + posicaoHotList + "!");
    }

    /**
     * Exibe todos os filmes presentes na HotList.
     */
    public void exibirHotList() {
        System.out.println("HotList:");
        for (int i = 0; i < hotList.length; i++) {
            if (hotList[i] != null) {
                System.out.println((i + 1) + " - " + hotList[i].getNome());
            }
        }
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

/**
 * Interface com menus texto para manipular o sistema FilmNow.
 *

 *
 */
public class MainFilmNow {

	public static void main(String[] args) {
		FilmNow fn = new FilmNow();

		System.out.println("Carregando filmes ...");
		try {
			/*
			 * Essa é a maneira de lidar com possíveis erros por falta do arquivo. 
			 */
			carregaFilmes("filmes_inicial.csv", fn);
		} catch (FileNotFoundException e) {

		} catch (IOException e) {
			System.err.println("Erro lendo arquivo: " + e.getMessage());
		}

		Scanner scanner = new Scanner(System.in);
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */
	private static String menu(Scanner scanner) {
		System.out.println(
				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" + 
						"(E)Exibir HotList\n" +
						"(H)Adicionar Hot\n" +
						"(R)Remover Hot\n" +
						"(S)air\n" + 
						"\n" + 
						"Opção> ");
		return scanner.next().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opcao   Opção digitada.
	 * @param fn  O sistema FilmNow que estamos manipulando.
	 * @param scanner Objeto scanner para o caso do comando precisar de mais input.
	 */
	private static void comando(String opcao, FilmNow fn, Scanner scanner) {
		switch (opcao) {
		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			fn.exibirHotList();
			break;
		case "H":
			System.out.print("Filme> ");
            int filme = scanner.nextInt();
            System.out.print("Posicao> ");
            int posicaoHot = scanner.nextInt();
            fn.adicionarHot(filme, posicaoHot);
            break;
		case "R":
			System.out.print("Posicao> ");
            int posicaoRemover = scanner.nextInt();
            fn.removerHot(posicaoRemover);
            break;
		case "S":
			sai();
			break;
		default:
			System.out.println("Opção inválida!");
		}
	}

	/**
	 * Imprime lista de filmes.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.

				System.out.println(formataFilme(i, filmes[i]));

		}
	}

	/**

	 * @param fn O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar qual contato.
	 */
	private static void detalharFilme(FilmNow fn, Scanner scanner) {
		System.out.print("\nQual filme> ");
		int posicao = scanner.nextInt();
		Filme filme = fn.getFilme(posicao - 1);
		if (filme != null) {
            System.out.println("Dados do filme:\n" + filme);
        } else {
            System.out.println("POSIÇÃO INVÁLIDA");
        }
	}

	/**
	 * Formata um filme para impressão.
