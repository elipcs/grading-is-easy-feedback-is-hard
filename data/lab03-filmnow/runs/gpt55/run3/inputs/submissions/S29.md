package filmnow;

import java.util.Objects;

/**
 * Representa um filme com suas informações básicas.
 * A classe armazena o nome, ano de lançamento e o local ou status
 * de exibição de um filme, além de indicar se ele faz parte de uma "hot list".
 * 
 * @author [REDACTED]
 */
public class Filme {

    /**
     * O nome ou título do filme.
     */
    private String nome;

    /**
     * O ano de lançamento do filme.
     */
    private String ano;

    /**
     * O local onde o filme está sendo exibido ou sua disponibilidade.
     */
    private String local;

    /**
     * Sinalizador para indicar se o filme está na hot list.
     */
    private boolean isHot;

    /**
     * Construtor para criar uma nova instância de Filme.
     *
     * @param nomeFilme O nome do filme. Não deve ser nulo.
     * @param ano O ano de lançamento do filme.
     * @param local O local de exibição ou disponibilidade do filme.
     */
    public Filme(String nomeFilme, String ano, String local) {
        this.nome = nomeFilme;
        this.ano = ano;
        this.local = local;
        this.isHot = false; // Por padrão, um filme não está na hot list
    }

    /**
     * Retorna o nome do filme.
     *
     * @return O nome atual do filme.
     */
    public String getNome() {
        return nome;
    }

    /**
     * Define ou atualiza o nome do filme.
     *
     * @param nomeFilme O novo nome para o filme.
     */
    public void setNome(String nomeFilme) {
        this.nome = nomeFilme;
    }

    /**
     * Retorna o ano de lançamento do filme.
     *
     * @return O ano de lançamento.
     */
    public String getAno() {
        return ano;
    }

    /**
     * Define ou atualiza o ano de lançamento do filme.
     *
     * @param ano O novo ano de lançamento.
     */
    public void setAno(String ano) {
        this.ano = ano;
    }

    /**
     * Retorna o local de exibição ou disponibilidade do filme.
     *
     * @return O local/disponibilidade atual.
     */
    public String getLocal() {
        return local;
    }

    /**
     * Define ou atualiza o local de exibição do filme.
     *
     * @param local O novo local/disponibilidade.
     */
    public void setLocal(String local) {
        this.local = local;
    }

    /**
     * Verifica se o filme está na hot list.
     *
     * @return {@code true} se o filme está na hot list, {@code false} caso contrário.
     */
    public boolean isHot() {
        return isHot;
    }

     * Define o status do filme na hot list.
     *
     * @param isHot O novo status da hot list para o filme.
     */
    public void setHot(boolean isHot) {
        this.isHot = isHot;
    }
    
    /**
     * Retorna uma representação detalhada do filme em string.
     * O formato é "Nome, Ano" na primeira linha e "Local" na segunda.
     *
     * @return Uma string formatada com os detalhes do filme.
     */
    public String detalhesFilme() {
        return nome + ", " + ano + "\n" + local;
    }

    /**
     * Gera um código de hash para o objeto Filme.
     * O código é gerado com base nos atributos `nome` e `ano`.
     * Essencial para o funcionamento correto de coleções baseadas em hash.
     *
     * @return O valor do código de hash para este objeto.
     */
    @Override

        return Objects.hash(ano, nome);

     * Compara este objeto Filme com outro para verificar se são iguais.

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. O sistema permite o cadastro de até 100 filmes
 * e a criação de uma "hot list" com 10 filmes em destaque.
 *
 * @author [REDACTED]
 */
public class FilmNow {

    /**
     * Define a capacidade máxima de filmes que o sistema pode armazenar.
     */
    private static final int TAMANHO = 100;

    /**
     * Define a capacidade máxima da hot list.
     */
    private static final int TAMANHOHOT = 10;

    /**
     * Array para armazenar a lista de filmes.
     */
    private Filme[] filmes;

    /**
     * Array para armazenar os filmes da hot list.
     */
    private Filme[] hotList;

    /**
     * Construtor que inicializa o sistema FilmNow, preparando os arrays de filmes e da hot list
     * com seus tamanhos definidos.
     */
    public FilmNow() {
        this.filmes = new Filme[TAMANHO];
        this.hotList = new Filme[TAMANHOHOT];
    }

    /**
     * Retorna uma cópia da lista completa de filmes.
     *
     * @return um array contendo todos os filmes cadastrados.
     */
    public Filme[] getFilmes() {
        return this.filmes.clone();
    }

    /**
     * Acessa os dados de um filme específico a partir de sua posição na lista.
     *
     * @param posicao A posição do filme no sistema (1 a 100).
     * @return O objeto Filme correspondente. Retorna null se a posição estiver vazia.
     */
    public Filme getFilme(int posicao) {
        return filmes[posicao - 1];
    }

    /**
     * Cadastra um novo filme em uma posição específica do sistema. Se já houver um filme
     * na posição indicada, ele será sobrescrito.
     *
     * @param posicao A posição onde o filme será armazenado (1 a 100).
     * @param nome O nome ou título do filme.
     * @param ano O ano de lançamento do filme.
     * @param local O local ou plataforma onde o filme pode ser assitido.
     * @return Uma mensagem indicando o status da operação: "FILME ADICIONADO", "POSIÇÃO INVÁLIDA",
     * "FILME INVÁLIDO" ou "FILME JÁ ADICIONADO".
     */
    public String cadastraFilme(int posicao, String nome, String ano, String local) {
        if (posicao < 1 || posicao > TAMANHO) {
            return "POSIÇÃO INVÁLIDA";
        }
        if (nome == null || local == null || nome.trim().isEmpty() || local.trim().isEmpty()) {
            return "FILME INVÁLIDO";
        }

        for (int i = 0; i < filmes.length; i++) {
            if (filmes[i] != null) {
                if (filmes[i].getNome().equalsIgnoreCase(nome) && filmes[i].getAno().equals(ano)) {
                    return "FILME JÁ ADICIONADO";
                }
            }
        }
        this.filmes[posicao - 1] = new Filme(nome, ano, local);
        return "FILME ADICIONADO";
    }

    /**
     * Retorna a hot list de filmes.
     *
     * @return um array contendo os filmes da hot list.
     */
    public Filme[] getHotList() {
        return this.hotList;
    }

    /**
     * Define a hot list de filmes.
     *
     * @param hotList O novo array de filmes para a hot list.
     */
    public void setHotList(Filme[] hotList) {
        this.hotList = hotList;
    }

    /**
     * Adiciona um filme da lista principal à hot list em uma posição específica.
     *
     * @param posicaoFilme A posição do filme na lista principal (1 a 100).
     * @param posicaoHotList A posição na hot list onde o filme será inserido (1 a 10).
     * @return Uma mensagem indicando o status da operação: "ADICIONADO À HOTLIST...",
     * "POSIÇÃO INVÁLIDA" ou "FILME JÁ PRESENTE NA LISTA".
     */
    public String cadastraHotList(int posicaoFilme, int posicaoHotList) {
        if (posicaoFilme < 1 || posicaoFilme > TAMANHO || filmes[posicaoFilme - 1] == null) {
            return "POSIÇÃO INVÁLIDA";
        }

        if (posicaoHotList < 1 || posicaoHotList > TAMANHOHOT) {
            return "POSIÇÃO INVÁLIDA";
        }

        for (int i = 0; i < hotList.length; i++) {
            if (hotList[i] != null && hotList[i].equals(filmes[posicaoFilme - 1])) {
                return "FILME JÁ PRESENTE NA LISTA";
            }
        }

        this.hotList[posicaoHotList - 1] = this.filmes[posicaoFilme - 1];
        this.hotList[posicaoHotList - 1].setHot(true);
        return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHotList + "!";
    }

    /**
     * Remove um filme da hot list com base na sua posição.
     *
     * @param posicaoHotList A posição do filme na hot list a ser removido (1 a 10).
     * @return Uma mensagem indicando o status da operação: "FILME REMOVIDO" ou "POSIÇÃO INVÁLIDA".
     */
    public String removeHot(int posicaoHotList) {
        if (posicaoHotList < 1 || posicaoHotList > TAMANHOHOT || hotList[posicaoHotList - 1] == null) {
            return "POSIÇÃO INVÁLIDA";
        }
        this.hotList[posicaoHotList - 1].setHot(false);
        this.hotList[posicaoHotList - 1] = null;
        return "FILME REMOVIDO";
    }

    /**
     * Fornece uma string com os detalhes de um filme específico. Adiciona um emoji de fogo se o filme
     * pertencer à hot list.
     *
     * @param posicao A posição do filme na lista principal (1 a 100).
     * @return Uma string formatada com os detalhes do filme ou "POSIÇÃO INVÁLIDA".
     */
    public String detalharFilme(int posicao) {
        if (posicao < 1 || posicao > TAMANHO || filmes[posicao - 1] == null) {
            return "POSIÇÃO INVÁLIDA";
        }
        if (filmes[posicao - 1].isHot()) {
            return ("\uD83D\uDD25 " + filmes[posicao - 1].detalhesFilme());
        }

        return (filmes[posicao - 1].detalhesFilme());
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
