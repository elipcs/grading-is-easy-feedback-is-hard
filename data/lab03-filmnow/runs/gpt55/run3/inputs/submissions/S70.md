package filmnow;

import java.util.Objects;

/**
 * Representação de um Filme que possui nome, ano de lançamento, local de lançamento e um possível status(HOT).
 * 
 * @author [REDACTED]
 */
public class Filme {
	
	private String nome;
	private String ano;
	private String local;
	private String hot;
	
	/**
	 * Constroí um Filme a partir do seu nome, ano de lançamento e do local onde foi lançado. 
	 * Inicialmente o filme não possui status pois a decisão dele ser HOT vai ser do usúario. 
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento.
	 * @param local Local de lançamento.
	 */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.hot = "";
	}
	
	public String getNome() {
		return nome;
	}

	public void setNome(String nome) {
		this.nome = nome;
	}

	public String getAno() {
		return ano;
	}

	public void setAno(String ano) {
		this.ano = ano;
	}

	public String getLocal() {
		return local;
	}

	public void setLocal(String local) {
		this.local = local;
	}
	
	public String getHot() {
		return hot;
	}

	public void setHot(String hot) {
		this.hot = hot;
	}
	
	/**
	 * Gera o HasCode do objeto atráves do seu ano e nome. 
	 * @return o número que representa o hashcode.
	 */
	@Override
	public int hashCode() {
		return Objects.hash(ano, nome);
	}

	/**
	 * Metódo que vai comparar se um filme é igual a outro objeto. Dois filmes são considerados iguais se tiverem o mesmo nome 
	 * e ano de lançamento.
	 * @param obj O objeto a ser comparado com este filme
	 * @return true se o objeto comparado for um filme com o mesmo nome e ano de lançamento, se não retorna false.
	 * */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return Objects.equals(ano, other.ano) && Objects.equals(nome, other.nome);
	}

	/**
	 * Retorna a string que representa o Filme. A representação segue o formato 
	 * “StatusNome, ano
	 * Local de lançamento”.
	 */
	@Override
	public String toString() {
		return this.hot + this.nome +", "+this.ano +"\n"+this.local;
	}
	

}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	private static final int TAMANHOHOT = 10;

	private Filme[] filmes; //uma representacao simploria da lista de filmes
	private Filme[] filmesHots;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.filmesHots = new Filme[TAMANHOHOT];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return 
	 * 		- Dados do filme, se o filme for diferente de null.
	 * 		- "POSIÇÃO INVÁLIDA", se a posição não estiver dentro do limite(1-100).
	 * 		- "", se o filme na posicao informada for igual a null.
	 */
	public String detalharFilme(int posicao) {
		String saida="";
		if(posicao > 100 || posicao < 1)
			saida = "POSIÇÃO INVÁLIDA";
		else if(filmes[posicao-1] != null)
			saida = filmes[posicao-1].toString();
		return saida;
	}

	/**
	 * Adiciona um filme em uma posição se ele não estiver na lista. Se já existir filme na posição, 
	 * sobrescreve o anterior.
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 * @return
	 * 		- "POSIÇÃO INVÁLIDA", se a posição não estiver dentro do limite(1-100).
	 * 		- "FILME INVALIDO", se o nome ou o local for vazio.
	 * 		- "FILME ADICIONADO", se o filme não estiver na lista de filmes.
	 * 		- "FILME JÁ ADICIONADO", se o filme já estiver na lista de filmes.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		Filme filme = new Filme(nome,ano,local);
		String saida;
		if (posicao > 100 || posicao < 1)
			saida = "POSIÇÃO INVÁLIDA";
		else {
			if (nome.isBlank())
				saida = "FILME INVALIDO";
			else {
				if(local.isBlank())
					saida = "FILME INVALIDO";
				else {
					if (verificaExistenciaFilme(filme) == false) {
						this.filmes[posicao-1] = filme;
						saida = "FILME ADICIONADO";
					} else 
						saida = "FILME JÁ ADICIONADO";
				}
			}
		}
		return saida;
	}
	
	private boolean verificaExistenciaFilme(Filme filme) {
		for(Filme f: this.filmes) {
			if (f != null) {
				if(f.equals(filme))
					return true;
			}
		}
		return false;
	}
	
	private boolean existeNaHotList(Filme filme) {
		for(Filme f: this.filmesHots) {
			if (f != null) {
				if(f.equals(filme))
					return true;
			}
		}
		return false;
	}
	
	/**
	 * Adiciona o filme, atráves do seu número, na HotList pela posição informada. Se já existir filme na posição informada
	 * o antigo filme será substituido e deixará de ser hot.
	 * @param posicao A posição da HotList que o filme vai ser inserido.
	 * @param numFilme O número do filme na lista filmes.
	 * @return Uma mensagem indicando o resultado da operação:
     *      - "FILME INVÁLIDO" se o número do filme estiver fora do limite permitido (1-100).
     *      - "POSIÇÃO INVÁLIDA" se a posição da HotList estiver fora do limite permitido (1-10).
     *      - "O FILME JÁ ESTÁ NA HOTLIST!" se o filme já estiver na HotList.
     *      - "ADICIONADO À HOTLIST NA POSIÇÃO posicao!" se o filme for adicionado com sucesso.
	 */
	public String adicionaHot(int posicao, int numFilme) {
		String saida;
		if(numFilme > 100 || numFilme < 1 || filmes[numFilme-1] == null) 
			saida = "FILME INVALIDO";
		else {
			if(posicao > 10 || posicao < 1)
				saida = "POSIÇÃO INVÁLIDA";
			else {
				if(!existeNaHotList(filmes[numFilme-1])) {
					if(filmesHots[posicao-1] != null)
						filmesHots[posicao-1].setHot("");		
					Filme filme = filmes[numFilme-1];
					filme.setHot("🔥");
					filmesHots[posicao-1] = filme;
					saida = "ADICIONADO À HOTLIST NA POSIÇÃO "+posicao+"!";
				} else
					saida = "O FILME JÁ ESTÁ NA HOTLIST!";
			}
		}
		return saida;
	}

	/**
	 * Pega todos os filmes que estão na HotList.
	 * @return uma cópia do array de filmes hots.
	 */
	public Filme[] getHotList() {
		return filmesHots.clone();
	}
	
	/**
	 * Remove um filme da HotList através da sua posição. Assim que o filme é removido ele deixa de possuir o status HOT(🔥).
	 * @param posicao A posição do filme que será removido.
	 * @return Uma mensagem indicando o resultado da operação:
	 *      - "POSIÇÃO INVÁLIDA" se a posicao não estiver dentro do limite (1-10) ou não possuir um filme.
	 *      - "HOT REMOVIDO!" se o filme for removido com sucesso.
	 */
	public String removeHot(int posicao) {
		String saida;
		if (posicao > 10 || posicao < 1 || filmesHots[posicao-1] == null)
			saida = "POSIÇÃO INVÁLIDA";
		else {
			Filme filme = filmesHots[posicao-1];
			for(Filme f: this.filmes) {
				if (f != null) {
					if(f.equals(filme))
						f.setHot("");
				}
			}
			filmesHots[posicao-1] = null;
			saida = "HOT REMOVIDO!";
		}
		return saida;
	}
	
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
