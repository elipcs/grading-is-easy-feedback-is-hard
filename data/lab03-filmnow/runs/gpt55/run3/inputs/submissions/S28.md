package filmnow;

import java.util.Objects;

/**
 * Representa um filme com seu nome, ano de lançamento e local onde pode ser assistido.
 * A unicidade de um filme é definida pela combinação do seu nome e ano.
 *
 * @author [REDACTED]
 */
public class Film {
	private String nome;
	private int ano;
	private String local;
	
	/**
	 * Constrói um objeto Filme a partir do nome, ano e local.
	 *
	 * @param nome O nome do filme.
	 * @param ano O ano de lançamento do filme.
	 * @param local O local ou plataforma onde o filme está disponível.
	 */
	public Film(String nome, int ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
	}
	
	public String getNome() {
		return nome;
	}
	
	public int getAno() {
		return ano;
	}
	
	/**
	 * Retorna uma representação em String do filme no formato "Nome, Ano\nLocal".
	 *
	 * @return A String formatada com os detalhes do filme.
	 */
	public String toString() {
		return nome + ", " + ano + '\n' + local;  
	}
	
	/**
	 * Compara este filme com outro objeto para verificar a igualdade.
	 * Dois filmes são considerados iguais se tiverem o mesmo nome e o mesmo ano.
	 *
	 * @param o O objeto a ser comparado.
	 * @return true se os filmes forem iguais (mesmo nome e ano), false caso contrário.
	 */
	public boolean equals(Object o) {
		if(this == o) return true;
		if(o == null) return false;
		if(getClass() != o.getClass()) return false;
		
		Film outroFilme = (Film) o;
		
		return Objects.equals(this.nome, outroFilme.nome) && Objects.equals(this.ano, outroFilme.ano);
	}
	
	/**
	 * Gera um código hash para o filme baseado no seu nome e ano.
	 *
	 * @return O código hash do objeto.
	 */
	public int hashCode() {
		return Objects.hash(nome, ano);
	}
	
}

package filmnow;
import java.util.ArrayList;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes.
 * Gerencia o cadastro, a consulta e a classificação de filmes em uma "HotList".
 *
 * @author [REDACTED]
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	
	private Film[] filmes; //uma representacao simploria da lista de filmes
	private ArrayList<Integer> hotList;

	/**
	 * Cria o FilmNow, inicializando a coleção de filmes com um tamanho fixo
	 * e a lista de filmes "hot".
	 */
	public FilmNow() {
		this.filmes = new Film[TAMANHO];
		this.hotList = new ArrayList<>();
	}
	
	/**
	 * Acessa a lista de títulos dos filmes mantidos.
	 * @return Um array com os nomes de todos os filmes cadastrados.
	 */
	public String[] getTitulosFilmes() {
		String[] titulos = new String[TAMANHO];
		for(int i = 0; i < TAMANHO; i++) {
			if(filmes[i] != null) {
				titulos[i] = filmes[i].getNome();
			}
		}
		return titulos;
	}

	/**
	 * Acessa os dados de um filme específico a partir de sua posição.
	 * @param posicao A posição do filme no sistema.
	 * @return O objeto Film. Retorna null se não há filme na posição.
	 */
	public Film getFilme(int posicao) {
		return filmes[posicao];
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior.
	 * @param posicao Posição do filme (deve ser entre 0 e 99).
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assistido.
	 */
	public void cadastraFilme(int posicao, String nome, int ano, String local) {
		this.filmes[posicao] = new Film(nome, ano, local);
	}
	
	/**
	 * Verifica se um filme, identificado pelo nome e ano, já existe no sistema.
	 * @param nome O nome do filme a ser verificado.
	 * @param ano O ano de lançamento do filme a ser verificado.
	 * @return true se o filme já foi cadastrado, false caso contrário.
	 */
	public boolean existeFilme(String nome, int ano) {
		for(int i = 0; i < TAMANHO; i++) {
			if(filmes[i] != null) {
				if(this.filmes[i].getNome().equals(nome) && this.filmes[i].getAno() == ano) {
						return true;
				}
			}
		}
		return false;
	}
	
	/**
	 * Retorna uma String com os detalhes de um filme em uma determinada posição.
	 * Se o filme estiver na "HotList", um emoji de fogo é adicionado ao início.
	 * @param posicao A posição do filme no sistema.
	 * @return A String formatada com os detalhes do filme.
	 */
	public String detalhesFilme(int posicao) {
		String detalhes = "";
		
		if(ehHot(posicao)) detalhes += "🔥 ";
		detalhes += filmes[posicao].toString();
		return detalhes;
	}
	
	/**
	 * Retorna um array com as posições de todos os filmes que estão na "HotList".
	 * @return Um array de inteiros contendo as posições dos filmes "hot".
	 */
	public int[] getHotList() {
		int qtdeHot = this.hotList.size();
		int[] hList = new int[qtdeHot];
		for(int i = 0; i < qtdeHot; i++) {
			hList[i] = this.hotList.get(i);
		}
		return hList;
	}
	
	private boolean ehHot(int posicao) {
		for(int i = 0; i < hotList.size(); i++) {
			if(hotList.get(i) == posicao) {
				return true;
			}
		}
		return false;
	}
	
	/**
	 * Adiciona um filme à "HotList" com base em sua posição.
	 * A "HotList" tem um limite de 10 filmes e não permite posições duplicadas.
	 * @param posicao A posição do filme a ser adicionado.
	 */
	public void atribuirHot(int posicao) {
		if(hotList.size() < 10 && !ehHot(posicao)) {
			hotList.add(posicao);
		}
	}
	
	/**
	 * Remove um filme da "HotList" com base em sua posição.
	 * @param posicao A posição do filme a ser removido.
	 */
	public void removerHot(int posicao) {
		hotList.remove(Integer.valueOf(posicao));
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

/**
 * Interface com menus em modo texto para manipular o sistema FilmNow.
 * Permite ao usuário interagir com o sistema para adicionar, listar,
 * detalhar e gerenciar filmes.
 *
 * @author [REDACTED]
 */
public class MainFilmNow {

	/**
	 * Ponto de entrada da aplicação. Carrega os filmes iniciais e inicia o loop
	 * de interação com o usuário através do menu.
	 *
	 * @param args Argumentos da linha de comando (não utilizados).
	 */
	public static void main(String[] args) {
		FilmNow fn = new FilmNow();

		System.out.println("Carregando filmes ...");
		try {
			/*
			 * Essa é a maneira de lidar com possíveis erros por falta do arquivo. 
			 */
			carregaFilmes("filmes_inicial.csv", fn);
		} catch (FileNotFoundException e) {
			System.err.println("Arquivo não encontrado: " + e.getMessage());
		} catch (IOException e) {
			System.err.println("Erro lendo arquivo: " + e.getMessage());
		}

		Scanner scanner = new Scanner(System.in);
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}

	}

	private static String menu(Scanner scanner) {
		System.out.println(
				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" +
						"(E)Exibir HotList\n" +
						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)Sair\n" +
						"\n" + 
						"Opção> ");
		return scanner.next().toUpperCase();
	}

	private static void comando(String opcao, FilmNow fn, Scanner scanner) {
		switch (opcao) {
		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibirHotList(fn);
			break;
		case "H":
			atribuirHot(fn, scanner);
			break;
		case "R":
			removerHot(fn, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("Opção inválida!");
		}
	}

	private static void mostrarFilmes(FilmNow fn) {

		String[] titulos = fn.getTitulosFilmes();
		for (int i = 0; i < titulos.length; i++) {
			if (titulos[i] != null) {
				System.out.println(formataFilme(i, titulos[i]));
			}
		}
	}

	private static void detalharFilme(FilmNow fn, Scanner scanner) {
		System.out.print("\nQual filme> ");
		try {
			int posicao = scanner.nextInt();
			String filmeDetalhado = fn.detalhesFilme(posicao);
			System.out.println("Dados do filme:\n" + filmeDetalhado);
		} catch(NullPointerException e) {
			System.out.println("FILME NÃO REGISTRADO");
		}
	}

	private static String formataFilme(int posicao, String filme) {
		return posicao + " - " + filme;
	}

	private static void adicionaFilme(FilmNow fn, Scanner scanner) {
		System.out.print("\nPosição no sistema> ");
		int posicao = scanner.nextInt();
		scanner.nextLine();
		
		if(posicao < 1 || posicao > 99) {
			System.out.println("POSIÇÃO INVÁLIDA");
			return;
		}
		
		System.out.print("\nNome> ");
		String nome = scanner.nextLine();
		
		if(nome.trim().equals("")) {
			System.out.println("FILME INVÁLIDO");
			return;
		}
		
		System.out.print("\nAno> ");
		String anoStr = scanner.nextLine();
		
		int ano;
		try {
			ano = Integer.parseInt(anoStr.trim());
		} catch (NumberFormatException e) {
			System.out.println("ANO INVÁLIDO");
			return;
		}
		
		if(fn.existeFilme(nome, ano)) {
			System.out.println("FILME JÁ ADICIONADO");
			return;
		}
		
		System.out.print("\nLocal> ");
		String local = scanner.nextLine();
		
		if(local.trim().equals("")) {
			System.out.println("FILME INVÁLIDO");
			return;
		}
		
		fn.cadastraFilme(posicao, nome, ano, local);
	}

	private static void sai() {
		System.out.println("\nVlw flw o/");
		System.exit(0);
	}
	
	private static void exibirHotList(FilmNow fn) {
		int[] hotList = fn.getHotList();
		Film filme;
		for(int i = 0; i < hotList.length; i++) {
			filme = fn.getFilme(hotList[i]);
			System.out.println(formataFilme(hotList[i], filme.getNome()));
		}
	}
	
	private static void atribuirHot(FilmNow fn, Scanner sc) {
		System.out.print("Filme> ");
		int posicao = sc.nextInt();
		if(fn.getFilme(posicao) != null) {
			fn.atribuirHot(posicao);
		}
	}
	
	private static void removerHot(FilmNow fn, Scanner sc) {
		System.out.print("Filme> ");
		int posicao = sc.nextInt();
		fn.removerHot(posicao);
	}

	private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		LeitorFilmNow leitor = new LeitorFilmNow();
		
		int carregados =  leitor.carregaContatos(arquivoFilmes, fn);
		System.out.println("Carregamos " + carregados + " registros.");
	}
}
