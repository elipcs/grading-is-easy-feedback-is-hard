import java.util.Objects;

/**

		this.local = local;

	 * Retorna a posição do filme.

		return this.posicao;
	}

	 * Retorna o nome do filme.
	 * @return Nome do filme.

		return this.nome;
	}
	
	/**
	 * Retorna o ano de lançamento do filme.
	 * @return Ano de lançamento do filme.
	 */
	public int getAno() {
		return this.ano;
	}
	
	/**
	 * Retorna o local onde o filme pode ser assistido.
	 * @return Local onde o filme pode ser assistido.
	 */
	public String getLocal() {
		return this.local;
	}
	
	/**
	 * Retorna indicação de se o filme está ou não na HotList.
	 * @return hotList Indicação se o filme está na HotList.
	 */
	public boolean getHot() {
		return this.hot;
	}
	
	/**
	 * Altera a indicação de Hot do filme.
	 * @param hotList Indicação de Hot.
	 */
	public void setHot(boolean hot) {
		this.hot = hot;
	}

	/**
	 * Retorna o valor do objeto Filme em String.
	 */
	@Override
	public String toString() {
		if (this.hot == true) {
			return "🔥 " + nome + ", " + ano + "\n" + local;
		} else {
			return nome + ", " + ano + "\n" + local;
		}
	}

	/**
	 * Retorna o hashcode do objeto Filme, em inteiro.
	 */
	@Override
	public int hashCode() {
		return Objects.hash(ano, local, nome, posicao);
	}
	
	/**
	 * Compara dois objetos do tipo Filme e retorna verdadeiro,
	 * quando o ano, local e nome são iguais, e falso, quando
	 * não são.
	 */
	@Override
	public boolean equals(Object obj) {
		if (obj == null) {
			return false;
		}
		if (getClass() != obj.getClass()) {
			return false;
		}
		Filme other = (Filme) obj;
		return ano == other.ano && Objects.equals(local, other.local) && Objects.equals(nome, other.nome);
	}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir até 100 filmes. 
 * 
 * @author [REDACTED]
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	/**
	 * Constante inteira que representa o
	 * tamanho máximo da listagem de filmes.
	 */
	private static final int TAMANHO = 100;
	
	/**
	 * Array de objetos do tipo Filme.
	 * Representa a listagem de filmes cadastrados.
	 */
	private Filme[] filmes;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		return filmes[posicao];
	}
	
	/**
	 * Adiciona o estado de Hot para determinado filme.
	 * @param posicao Posição do filme a ser alterado.
	 */
	public void adicionaHot(int posicao) {
		filmes[posicao].setHot(true);
	}
	
	/**
	 * Remove o estado de Hot para determinado filme.
	 * @param posicao Posição do filme a ser alterado.
	 */
	public void removeHot(int posicao) {
		filmes[posicao].setHot(false);
	}

	/**
	 * Verifica se a posição está vazia.
	 * @param posicao Posição que se quer saber se está vazia.
	 * @return boolean Retorna verdadeiro quando a posição está
	 * vazia e falsa quando está preenchida.
	 */
	public boolean verificaSePosicaoEstaVazia(int posicao) {
		if (filmes[posicao] == null) {
			return true;
		}
		if (filmes[posicao].getPosicao() != posicao) {
			return true;
		}
		return false;
	}
	
	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public void cadastraFilme(int posicao, String nome, int ano, String local) {
		this.filmes[posicao] = new Filme(posicao, nome, ano, local);
	}
	
	/**
	 * Verifica se um filme já foi inserido anteriormente, em qualquer posição,
	 * a partir de nome e ano informados.
	 * @param nome Nome do filme que se quer inserir.
	 * @param ano Ano de lançamento do filme que se quer inserir.
	 * @return boolean
	 */
	public boolean verificaSeFilmeJaFoiInserido(String nome, int ano) {
		for (byte i = 0; i < 100; i++) {
			if (this.filmes[i] == null) {
				continue;
			}
			if (this.filmes[i].getNome() == nome || this.filmes[i].getAno() == ano) {
				return true;
			}
		}
		return false;
	}

}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}

package filmnow;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.util.Scanner;

/**
 * Interface com menus texto para manipular o sistema FilmNow.
 * 
 * @author [REDACTED]
 * @author [REDACTED]
 */
public class MainFilmNow {

	public static void main(String[] args) {
		FilmNow fn = new FilmNow();

		System.out.println("Carregando filmes ...");
		try {
			/*
			 * Essa é a maneira de lidar com possíveis erros por falta do arquivo. 
			 */
			carregaFilmes("filmes_inicial.csv", fn);
		} catch (FileNotFoundException e) {
			System.err.println("Arquivo não encontrado: " + e.getMessage());
		} catch (IOException e) {
			System.err.println("Erro lendo arquivo: " + e.getMessage());
		}

		Scanner scanner = new Scanner(System.in);
		String escolha = "";
		while (true) {
			escolha = menu(scanner);
			comando(escolha, fn, scanner);
		}

	}

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.
	 */
	
	private static String menu(Scanner scanner) {
		System.out.print(
					"\n(A)Adicionar filme\n" + 
					"(M)Mostrar todos\n" + 
					"(D)Detalhar filme\n" +
					"(E)Exibir HotList\n" +
					"(H)Atribuir Hot\n" +
					"(R)Remover Hot\n" +
					"(S)Sair\n" + 
					"\n" + 
					"Opção> ");
		return scanner.nextLine().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opcao   Opção digitada.
	 * @param fn  O sistema FilmNow que estamos manipulando.
	 * @param scanner Objeto scanner para o caso do comando precisar de mais input.
	 */
	private static void comando(String opcao, FilmNow fn, Scanner scanner) {
		switch (opcao) {
		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibirHotList(fn);
			break;
		case "H":
			atribuirHot(fn, scanner);
			break;
		case "R":
			removerHot(fn, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("OPÇÃO INVÁLIDA!");
		}
	}

	/**
	 * Remove estado de Hot de um determinado filme, indicado por sua posição.
	 * @param fn O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar qual posição.
	 */
	private static void removerHot(FilmNow fn, Scanner scanner) {
		System.out.print("Posição> ");
		int posicao = Integer.parseInt(scanner.nextLine());
		fn.removeHot(posicao);
	}
	
	/**
	 * Atribui estado de Hot a um determinado filme, indicado por sua posição.
	 * @param fn O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar qual posição.
	 */
	private static void atribuirHot(FilmNow fn, Scanner scanner) {
		System.out.print("Posição> ");
		int posicao = Integer.parseInt(scanner.nextLine());
		fn.adicionaHot(posicao);
		System.out.println("ADICIONADO À HOTLIST NA POSIÇÃO " + posicao + "!");
	}
	
	/**
	 * Imprime lista de filme Hot.
	 * @param fn O sistema FilmNow a ser manipulado.
	 */
	private static void exibirHotList(FilmNow fn) {
		Filme[] filmes = fn.getFilmes();
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				if (filmes[i].getHot() == true) {
					System.out.println(formataFilme(i, filmes[i]));
				}
			}
		}
	}
	
	/**
	 * Imprime lista de filmes.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 */
	private static void mostrarFilmes(FilmNow fn) {
		Filme[] filmes = fn.getFilmes();
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				System.out.println(formataFilme(i, filmes[i]));
			}
		}
	}

	/**
	 * Imprime os detalhes de um dos filmes. 
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para capturar qual contato.
	 */
	private static void detalharFilme(FilmNow fn, Scanner scanner) {
		System.out.print("Posição> ");
		int posicao = Integer.parseInt(scanner.nextLine());
		Filme filme = fn.getFilme(posicao);
		System.out.println(filme);
	}

	/**
	 * Formata um filme para impressão. 
	 * 
	 * @param posicao A posição do filme (que é exibida)/
	 * @param filme O filme a ser impresso.
	 * @return A String formatada.
	 */
	private static String formataFilme(int posicao, Filme filme) {
		return posicao + " - " + filme.getNome();
	}

	/**
	 * Verifica se a posição é valida.
	 * Para ser válida, deve ser mair ou igual a 1
	 * e menor ou igual a 100.
	 * @param posicao Posição
	 * @return boolean 
	 */
	private static boolean verificaPosicaoEhValida(int posicao) {
		if (posicao >= 1 && posicao <= 100) {
			return true;
		} else {
			return false;
		}
	}
	
	/**
	 * Cadastra um filme no sistema. 
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
	 * @param scanner Scanner para pedir informações do contato.
	 */
	private static void adicionaFilme(FilmNow fn, Scanner scanner) {
		System.out.print("\nPosição> ");
		int posicao = Integer.parseInt(scanner.nextLine());
		if (verificaPosicaoEhValida(posicao) == false) {
			System.out.println("OPÇÃO INVÁLIDA!");
			return;
		}
		System.out.print("Nome> ");
		String nome = scanner.nextLine();
		if (nome.length() == 0) {
			System.out.println("FILME INVALIDO");
			return;
		}
		System.out.print("Ano> ");
		int ano = Integer.parseInt(scanner.nextLine());
		System.out.print("Local> ");
		String local = scanner.nextLine();
		if (local.length() == 0) {
			System.out.println("FILME INVALIDO");
			return;
		}
		if (fn.verificaSeFilmeJaFoiInserido(nome, ano) == true) {
			System.out.println("FILME JA ADICIONADO");
			return;
		}
		fn.cadastraFilme(posicao, nome, ano, local);
		System.out.print("FILME ADICIONADO\n");
	}

	/**
	 * Sai da aplicação.
	 */
	private static void sai() {
		System.exit(0);
	}

	/**
	 * Lê carga de filmes de um arquivo csv. 
	 * 
	 * @param arquivoFilmes O caminho para o arquivo.
	 * @param fn O sistema FilmNow a ser populado com os dados.
	 * @throws IOException Caso o arquivo não exista ou não possa ser lido.

	private static void carregaFilmes(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		LeitorFilmNow leitor = new LeitorFilmNow();
		
		int carregados =  leitor.carregaContatos(arquivoFilmes, fn);
		System.out.println("Carregamos " + carregados + " registros.");
	}
}
