package filmnow;

import java.util.Objects;


public class Filmes{
	/**
	 * Nome do filme.
	 */
	private String filme;
	/**
	 * Ano de lançamento.
	 */
	private String ano;
	/**
	 * Local em que está disponível para assistir
	 */
	private String local;
	/**
	 * Indica se faz parte da HotList ou não.
	 */
	private boolean hot;
	
	/**
	 * Controi um Filme com seus atributou: nome do filme, ano de lançamento, 
	 * local em está diponível para assistir e se faz parte da HotList.
	 */
	public Filmes(String filme, String ano, String local) {
		this.filme = filme;
		this.ano = ano;
		this.local = local;
	}
	/**
	 * Retorna o nome do filme
	 * @return String nome nome do filme
	 */
	public String getNome() {
		return filme;
	}
	/**
	 * Retorna o ano de lançamento do filme.
	 * @return String ano Ano de Lançamento do Filme 
	 */
	public String getAno() {
		return ano;
	}
	
	/**
	 * Retorna o local em que o filme está disponível para assistir.
	 * @return String local Local em que o está disponívrll para assistir.
	 */
	public String getLocal() {
		return local;
	}
	
	/**
	 * Configura o status hot ao filme.
	 */
	public void incluirHot() {
		hot = true;
	}
	
	/**
	 * Reomve o status hot do filme.
	 */
	public void removeHot() {
		hot = false;
	}
	
	/**
	 * Código explusivo que representa o objeto baseano no nome do filme e no seu ano de lançamento.
	 * @return int Código que representa o objeto.
	 */
	@Override
	public int hashCode() {
		return Objects.hash(ano, filme);
	}
	
	/**
	 * Compara se dois objetos Filmes são iguais.
	 * @return boolean Valor lógico que indica se os dois objetos são iguais.
	 */
	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filmes other = (Filmes) obj;
		return Objects.equals(ano, other.ano) && Objects.equals(filme, other.filme);
	}
	
	/**
	 * Retorna a representação de um filme.
	 * @return String "(nome do filme), (ano de lançamento), (local)" ou "(nome do filme), 
	 * (ano de lançamento), (local)", se não houver local atribuído. Ou, se tiver sido adicionado à HotList:
	 * "🔥 (nome do filme), (ano de lançamento), (local)" ou "🔥 (ano de lançamento), (local)" se não houver local
	 * atribuído
	 */
	public String toString() {
		if (hot && ano == "") return "🔥 " + filme + ", " + local;
		if (hot) return "🔥 " + filme + ", " + ano + ", " + local;
		if (ano == "") return filme + "\n" + " " + local;
		return filme + ", " + ano + ", " + local;
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	private Filmes[] filmes;
	private Filmes[] hotList;
	
	
	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		this.filmes = new Filmes[TAMANHO];
		this.hotList = new Filmes[10];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public String[] getFilmes() {
		String[] filmesArray = new String[TAMANHO];
		for (int i = 0; i < 100; i++) {
			if (filmes[i] != null) {
				filmesArray[i] = filmes[i].getNome();
			}
		}
		return filmesArray;
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public String getFilme(int posicao) {
		return filmes[posicao].getNome();
	}
	
	/**
	 * Clona a HotList.
	 * @return Filmes[] Uma array da classe Filmes que representa a HotList.
	 */
	public Filmes[] getHotList() {
		Filmes[] hotListArray = new Filmes[10];
		for (int i = 0; i < hotList.length; i++) {
			if (hotList[i] != null) {
				hotListArray[i] = hotList[i];
			}
		} 
		return hotListArray;
	}
	/**
	 * Detalha as características de um filme.
	 * @return String "Dados do filme:\n filme, ano, local" ou "POSICÃO INVÁLIDA" se 
	 * a posição indicada for menor que 1 ou maior que 100
	 */
	public String detalhar(int posicao){
		if (posicao < 1 || posicao > 100) return "POSIÇÃO INVÁLIDA";
		if (filmes[posicao-1] == null) return null;
		return "Dados do filme:\n" + filmes[posicao-1].toString();
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		Filmes filmeObj = new Filmes(nome, ano, local);
		if (posicao-1 < 0 || 99 < posicao-1) {
			return "POSIÇÃO INVÁLIDA";
		} else if (nome == "" || local == ""){
			return "FILME INVÁLIDO";
		} else if (analisaIguais(filmeObj)) {
			return "FILME JÁ ADICIONADO";
		} else {
			filmes[posicao-1] = new Filmes(nome, ano, local);
		}
		return "FILME ADICIONADO";
	}
	
	/**
	 * Analinasa se o filme já foi adicionado à array de filmes.
	 * @param Filmes obj Um Objto do tipo Filmes. 
	 * @return boolean que indica se o filme já foi ou não adicionado.
	 */
	private boolean analisaIguais(Filmes obj) {
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null && filmes[i].equals(obj)) return true;
		}
		return false;
	}
	/**
	 * Adiciona um filme á Array que representa a HotList.
	 * @param int posicao Posição em que o filme deve ser alocado na array + 1
	 * @param int filme posicao em que o filme ocupa na array de filmes + 1
	 * @return String "ADICIONADO À HOTLIST NA POSIÇÃO" ou "" se o filme já tiver sido adicionado
	 */
	public String adicionaHotList(int posicao, int filme) {
		for (int i = 0; i < hotList.length; i++) {
			if (filmes[filme-1].equals(hotList[i])) {
				return "";
			} 
		}
		for (int i = 0; i < filmes.length; i++) {
			if (hotList[posicao-1] != null && hotList[posicao-1].equals(filmes[i])) {
				filmes[i].removeHot();
				break;
			}
		}
		filmes[filme-1].incluirHot();
		hotList[posicao-1] = filmes[filme-1];
		return "ADICIONADO À HOTLIST NA POSIÇÃO";
	}
	/**
	 * Remove o filme da array de HotList.
	 * @param int posicao Posição em que o filme se escontra na array + 1
	 */
	public void removeHotList(int posicao) {
		hotList[posicao-1] = null;
		filmes[posicao-1].removeHot();
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}

			escolha = menu(scanner);

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.
	 * @return O comando escolhido.

		System.out.println(
				"\n---\nMENU\n" + 
						"(A)Adicionar filme\n" + 
						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" +
						"(E)Exibir HotList\n" +
						"(H)Atribuir Hot\n" +
						"(R)Remover Hot\n" +
						"(S)Sair\n" + 
						"\n" + 
						"Opção> ");
		return scanner.nextLine().toUpperCase();
	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.
	 * 
	 * @param opcao   Opção digitada.
	 * @param fn  O sistema FilmNow que estamos manipulando.
	 * @param scanner Objeto scanner para o caso do comando precisar de mais input.

	private static void comando(String opcao, FilmNow fn, Scanner scanner) {
		switch (opcao) {
		case "A":
			adicionaFilme(fn, scanner);
			break;
		case "M":
			mostrarFilmes(fn);
			break;
		case "D":
			detalharFilme(fn, scanner);
			break;
		case "E":
			exibirHotList(fn);
			break;
		case "H":
			atribuirHot(fn, scanner);
			break;
		case "R":
			removeHot(fn, scanner);
			break;
		case "S":
			sai();
			break;
		default:
			System.out.println("OPÇÃO INVÁLIDA!");
		}
	}

	/**
	 * Imprime lista de filmes.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado.
