package filmnow;

public class Filme {
    private String nome;
    private int ano;
    private String local;
    private boolean hot;

    /**
     * Construtor da classe Filme.
     * Inicializa o nome, ano e local do filme conforme os parâmetros recebidos.
     * O atributo 'hot', que indica se o filme está na HotList, é inicializado como falso por padrão.
     * 
     * @param nome Nome do filme.
     * @param ano Ano de lançamento do filme.
     * @param local Local onde o filme está disponível.
     */
    public Filme(String nome, int ano, String local) {
        this.nome = nome;
        this.ano = ano;
        this.local = local;
        this.hot = false;
    }

    public int getAno() {
        return ano;
    }

    public String getNome() {
        return nome;
    }
    public String getLocal() {
        return local;
    }
    public void setHot(boolean hot) { // 
        this.hot = hot;
    }
    /**
     * Retorna a representação em String dos detalhes do filme.
     * Se o filme estiver marcado como 'hot', adiciona o emoji de fogo (🔥) como prefixo.
     * Caso o ano seja zero, exibe apenas o nome e o local em linhas separadas.
     * Caso contrário, exibe o nome seguido do ano e do local em linhas separadas.
     *
     * @return String com os detalhes formatados do filme, incluindo indicação 'hot' se aplicável.
     */
    public String getDetalhes() { // 
        String prefixo = hot ? "🔥 " : "";
        if (ano == 0) {
            return prefixo + nome + "\n" + local;
        }
        return prefixo + nome + ", " + ano + "\n" + local;
    }
    public String nomeAno() {
        return this.nome + ", " + this.ano;
    }
    /**
     * Compara este objeto Filme com outro para verificar igualdade.
     * Dois filmes são considerados iguais se:
     * - São a mesma instância (referência)
     * - Ou se ambos são da classe Filme, possuem o mesmo ano e nomes iguais (considerando null).
     *
     * @param obj o objeto a ser comparado com este filme.
     * @return true se os filmes são iguais; false caso contrário.
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;               
        if (obj == null || getClass() != obj.getClass()) return false;  
        Filme outro = (Filme) obj;
        if (ano != outro.ano) return false;
        if (nome == null) return outro.nome == null;
        return nome.equals(outro.nome);
    }
    /**
     * Gera o código hash para este objeto Filme, baseado no nome e ano.
     * Essencial para manter a consistência com o método equals.
     *
     * @return valor inteiro que representa o código hash do filme.
     */
    @Override
    public int hashCode() {
        int result = (nome == null) ? 0 : nome.hashCode();
        result = 31 * result + ano;
        return result;
    }
    @Override
    public String toString() {
        return getDetalhes();
    }
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 101;
	
	private Filme[] filmes;
	private Filme[] hotList = new Filme[10];
	
	/**
	 * Adiciona um filme à HotList na posição especificada.
	 * Verifica se a posição é válida, se o filme existe no sistema,
	 * e se o filme ainda não está na HotList.
	 * Caso válido, adiciona ou substitui o filme na HotList.
	 *
	 * @param posicaoHotList posição da HotList para adicionar o filme (0-based)
	 * @param posicaoFilme posição do filme no sistema para adicionar à HotList
	 * @return true se o filme foi adicionado com sucesso; false caso contrário
	 */
	public boolean adicionaHot(int posicaoHotList, int posicaoFilme) {
	    if (posicaoHotList < 0 || posicaoHotList >= hotList.length) return false;
	    Filme filme = getFilme(posicaoFilme);
	    if (filme == null) return false;

	    for (Filme f : hotList) {
	        if (filme.equals(f)) return false;  // já está na HotList
	    }
	    hotList[posicaoHotList] = filme;  // substitui filme se já existir
	    return true;
	}
	
	/**
	 * Remove um filme da HotList na posição especificada.
	 * Verifica se a posição é válida e se existe filme na posição.
	 *
	 * @param posicaoHotList posição da HotList para remover o filme (0-based)
	 * @return true se a remoção foi feita com sucesso; false caso contrário
	 */
	public boolean removeHot(int posicaoHotList) {
	    if (posicaoHotList < 0 || posicaoHotList >= hotList.length) return false;
	    if (hotList[posicaoHotList] == null) return false;
	    hotList[posicaoHotList] = null;
	    return true;
	}

	public Filme[] getHotList() {
	    return hotList;
	}
	/**
	 * Verifica se um filme está presente na HotList.
	 * Retorna false se o filme for null.
	 *
	 * @param filme filme a ser verificado na HotList
	 * @return true se o filme está na HotList; false caso contrário
	 */
	public boolean estaNaHotList(Filme filme) {
	    if (filme == null) return false;
	    for (Filme f : hotList) {
	        if (filme.equals(f)) return true;
	    }
	    return false;
	}

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		filmes = new Filme[TAMANHO];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * @return O array de filmes.
	 */
	public Filme[] getFilmes() {
		return filmes.clone();
	}

	/**
	 * Acessa os dados de um filme específico.
	 * @param posicao Posição do filme no sistema.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	public Filme getFilme(int posicao) {
		if (posicao < 0 || posicao >= TAMANHO) {
			return null;
		}
		// Retorna os dados do filme formatados: nome, ano e local
		return filmes[posicao];
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * @param posicao Posição do filme.
	 * @param nome Nome do filme.
	 * @param ano Ano de lançamento do filme.
	 * @param local Local onde o filme pode ser assitido.
	 */
	public boolean cadastraFilme(int posicao, String nome, int ano, String local) {
		if (posicao < 0 || posicao >= filmes.length) {
			return false; // posição inválida
		}
		if (jaExisteFilme(nome, ano)) {
			return false;
		}
		filmes[posicao] = new Filme(nome, ano, local);
		return true;
	}
	/**
	 * Verifica se já existe um filme com o mesmo nome e ano no sistema.
	 * Percorre o array de filmes e compara cada filme existente com um filme temporário
	 * criado com os dados informados (nome e ano).
	 * 
	 * @param nome o nome do filme a verificar
	 * @param ano o ano do filme a verificar
	 * @return true se um filme com o mesmo nome e ano já estiver cadastrado, false caso contrário
	 */
	 private boolean jaExisteFilme(String nome, int ano) {
		 for (int i = 0; i < filmes.length; i++) {
		        Filme f = filmes[i];
		        if (f != null) {
		            Filme filmeParaComparar = new Filme(nome, ano, "qualquer");
		            if (f.equals(filmeParaComparar)) {
		                return true;
		            }
		        }
		    }
		    return false;
	   }
	 /**
	  * Adiciona um filme na posição especificada do sistema.
	  * Verifica se a posição é válida (1 a 100), se os dados são válidos
	  * e se o filme não foi adicionado anteriormente.
	  * Converte o ano recebido como String para int, considerando vazio como zero.
	  * 
	  * @param posicao a posição onde o filme será adicionado (1 a 100)
	  * @param nome o nome do filme
	  * @param anoStr o ano do filme como String (pode ser vazio)
	  * @param local o local do filme
	  * @return mensagem indicando o resultado da operação
	  */
	public String adicionaFilme(int posicao, String nome, String anoStr, String local) {
		if (posicao < 1 || posicao > 100) {
	        return "POSIÇÃO INVÁLIDA";
	    }
	    if (nome == null || nome.length() == 0 || local == null || local.length() == 0) {
	        return "FILME INVALIDO";
	    }
	    int ano = 0;
	    if (anoStr != null && anoStr.length() > 0) {
	        ano = Integer.parseInt(anoStr);  
	    }
	    if (jaExisteFilme(nome, ano)) {
	        return "FILME JÁ ADICIONADO";
	    }
	    filmes[posicao] = new Filme(nome, ano, local);
	    return "FILME ADICIONADO";
	}
	/**
	 * Retorna os detalhes do filme armazenado na posição especificada.
	 * Verifica se a posição é válida (1 a 100). Se a posição estiver vazia,
	 * retorna null para indicar ausência de filme.
	 * 
	 * @param posicao a posição do filme para detalhar (1 a 100)
	 * @return os detalhes do filme ou null se não houver filme na posição
	 */
	public String detalharFilme(int posicao) {
		   if (posicao < 1 || posicao > 100) {
		        return "POSIÇÃO INVÁLIDA";
		    }
		    Filme filme = getFilme(posicao);
		    if (filme == null) {
		        return null;  
	    }
	    return filme.getDetalhes();
	}
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, Integer.parseInt(ano), local);
	}

}
