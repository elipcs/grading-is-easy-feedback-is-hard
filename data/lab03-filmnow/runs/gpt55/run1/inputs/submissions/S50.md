package filmnow;
import java.util.Objects;

/**
 * Representa um filme com nome, ano de lançamento, local de exibição e status de hotlist.
 * 
 * @author [REDACTED]
 */

public class Filme {
	private String nome;
	private String anoLancamento;
	private String local;
	private boolean statusHot;
	
	/**
     * Construtor da classe Filme. Inicializa um novo filme com o nome, ano de lançamento e local especificados.
     * 
     * @param nome Nome do filme.
     * @param ano Ano de lançamento do filme.
     * @param local Local onde o filme está disponível.
     */
	public Filme(String nome, String ano, String local) {
		this.local = local;
		this.nome = nome;
		this.anoLancamento = ano;
	}
	
	public String getNome() {
		return this.nome;
	}
	
	public String getAnoLancamento() {
		return this.anoLancamento;
	}
	
	public String getLocal() {
		return this.local;
	}
	
	public void setHot(boolean status) {
		this.statusHot = status;
	}
	
	public boolean isHot() {
		return statusHot;
	}
	
	public String nomeAno() {
		if(anoLancamento == null || anoLancamento == "") {
			return nome;
		}
		return nome + ", " + anoLancamento;
	}
	@Override 
	public int hashCode() {
		return Objects.hash(anoLancamento,nome);
	}
	
	/**
     * Verifica se este filme é igual a outro objeto.
     * 
     * Dois filmes são considerados iguais se tiverem o mesmo nome e ano de lançamento.
     * 
     * @param obj O objeto a ser comparado com este filme.
     * @return true se os filmes forem iguais, caso contrário false.
     */
	@Override
	public boolean equals(Object obj) {
		   if (this == obj) 
		    	return true;
		    if (obj == null)
		    	return false;
		    if (getClass() != obj.getClass())
		   	return false;
		    Filme other = (Filme) obj;
		    return Objects.equals(anoLancamento, other.anoLancamento)  && Objects.equals(nome, other.nome);
	}
	
	/**
     * Retorna uma representação em string do filme.
     * 
     * Se o status de hotlist estiver definido como true, o nome do filme será mostrado por um emoji de fogo ("🔥").
     * 
     * @return Uma string representando o filme, incluindo seu nome, ano de lançamento e local de exibição.
     */
	@Override
	public String toString() {
		if (statusHot)
			return "🔥 " + this.nome + ", " + this.anoLancamento + "\n" + this.local ;
		return nomeAno() + "\n" + this.local;
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100;
	
	private Filme[] filmes;
	private Filme[] filmesHot;
	
	/**
     * Construtor da classe FilmNow. 
     * Inicializa os arrays de filmes e hotlist com os tamanhos definidos.
     */
	
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.filmesHot = new Filme[10];
	}
	
	 /**
     * Retorna uma cópia do array de filmes.
     * 
     * @return cópia do array de filmes.
     */
	
	public Filme[] getFilmes() {
		return this.filmes.clone();
	}
	
	
	/**
     * Retorna uma cópia do array de filmes da hotlist.
     * 
     * @return uma cópia do array de filmes da hotlist.
     */
	public Filme[] getFilmesHot() {
	    return this.filmesHot.clone(); 
	}
	
	/**
     * Retorna o filme na posição especificada.
     * 
     * @param posicao a posição do filme no array.
     * @return o filme na posição especificada.
     */
	public Filme getFilme(int posicao) {
		return filmes[posicao-1];
	}
	
	/**
     * Retorna o filme na posição especificada da hotlist.
     * 
     * @param posicaoHot a posição do filme na hotlist.
     * @return o filme na posição especificada da hotlist.
     */
	public Filme getFilmeHot(int posicaoHot) {
		return filmesHot[posicaoHot-1];
	}

	 /**
     * Cadastra um novo filme na posição especificada.
     * 
     * @param posicao a posição onde o filme será adicionado.
     * @param nome o nome do filme.
     * @param ano o ano de lançamento do filme.
     * @param local o local onde o filme pode ser assistido.
     * @return uma mensagem indicando o resultado do cadastro.
     */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
	    if (!verificaPosicao(posicao))
	        	return "POSIÇÃO INVÁLIDA";
	        Filme novoFilme = new Filme(nome,ano,local);
	        String msg = verificaFilmeExistente(novoFilme);
	    if(verificaFilmeExistente(novoFilme).equals("FILME ADICIONADO")) {
	    	this.filmes[posicao-1] = novoFilme;
	    }
	    return msg;
	    }
	
	   /**
     * Verifica se o filme já foi adicionado ao sistema.
     * 
     * @param novoFilme o filme a ser verificado.
     * @return uma mensagem indicando se o filme é inválido, já foi adicionado, ou pode ser adicionado.
     */
	  private String verificaFilmeExistente(Filme novoFilme) {
		  if (novoFilme.getNome().equals("") || novoFilme.getLocal().equals("") || novoFilme.getNome() == null || novoFilme.getLocal() == null) 
			  return "FILME INVÁLIDO";
	  else {
	        for (int i = 0; i < filmes.length; i++) {
	            if (getFilmes()[i]!= null && getFilmes()[i].equals(novoFilme)) {
	                return "FILME JÁ ADICIONADO";
	            }
	        }
	        return "FILME ADICIONADO";
	}
		  }
	  
	  /**
	     * Verifica se a posição especificada é válida para o array de filmes.
	     * 
	     * @param posicao a posição a ser verificada.
	     * @return true se a posição for válida, false caso contrário.
	     */
	  private boolean verificaPosicao(int posicao) {
		  return posicao >= 1 && posicao <= 100;
	  }
	  
	   /**
	     * Verifica se a posição especificada é válida para o array de hotlist.
	     * 
	     * @param posicao a posição a ser verificada.
	     * @return true se a posição for válida, false caso contrário.
	     */
	  private boolean verificaPosicaoHot(int posicao) {
		  return posicao > 0 && posicao < 11;
	  }
	  
	  /**
	     * Detalha as informações do filme na posição especificada.
	     * 
	     * @param posicao a posição do filme no array.
	     * @return uma string com os detalhes do filme ou uma mensagem de posição inválida.
	     */
	  public String detalhaFilme(int posicao) {
		  if (verificaPosicao(posicao)) {
				Filme filme = getFilme(posicao);
				if (filme != null) {
				 return filme.toString();
		  }else {
				return "";
				}
		  }else {
			return "POSIÇÃO INVÁLIDA";
		  }
	  }
	  
	  
	  /**
	     * Verifica se o filme pode ser adicionado à hotlist.
	     * 
	     * @param posicaoFilme a posição do filme no array de filmes.
	     * @param posicaoHot a posição onde o filme será adicionado na hotlist.
	     * @return uma mensagem indicando o resultado da verificação.
	     */
	  private String verificaFilmeHot(int posicaoFilme, int posicaoHot) {
		    if (!verificaPosicaoHot(posicaoHot) || !verificaPosicao(posicaoFilme))
		        return "POSIÇÃO INVÁLIDA";
		    Filme filme = this.filmes[posicaoFilme - 1];
		     if (filme == null || filme.getNome().isEmpty() || filme.getLocal().isEmpty())
		        return "FILME INVÁLIDO";
		     
		     for (Filme hotFilmes : filmesHot) {
		    	 if (hotFilmes != null && hotFilmes.equals(filme))
		    		 return "FILME JÁ ADICIONADO NA HOT";
		     }
		     
		    return "FILME ADICIONADO À HOTLIST NA POSICAO " + posicaoHot;
		}
	  
	  
	  /**
	     * Adiciona um filme à hotlist na posição especificada.
	     * 
	     * @param posicaoFilme a posição do filme no array de filmes.
	     * @param posicaoHot a posição onde o filme será adicionado na hotlist.
	     * @return uma mensagem indicando o resultado da operação.
	     */
		public String adicionaHot(int posicaoFilme, int posicaoHot) {
		    String resultadoVerificacao = verificaFilmeHot(posicaoFilme, posicaoHot);
		    if (resultadoVerificacao.equals("FILME ADICIONADO À HOTLIST NA POSICAO " + posicaoHot)) {
		    	if (filmesHot[posicaoHot - 1] != null) {
		    		filmesHot[posicaoHot - 1].setHot(false);
		    	}
		        filmesHot[posicaoHot - 1] = this.filmes[posicaoFilme - 1];
		        this.filmes[posicaoFilme - 1].setHot(true);
		    }
		    return resultadoVerificacao;
		}
		
		 /**
	     * Remove um filme da hotlist na posição especificada.
	     * 
	     * @param posicao a posição do filme na hotlist.
	     * @return uma mensagem indicando o resultado da remoção.
	     */
	  public String removeHot(int posicao) {
		  if (!verificaPosicaoHot(posicao))
		        return "POSIÇÃO INVÁLIDA";
		    Filme filme = this.filmesHot[posicao - 1];
		     if (filme == null || filme.getNome().isEmpty() || filme.getLocal().isEmpty())
		        return "FILME INVÁLIDO";
		  filme.setHot(false);
		  this.filmesHot[posicao-1] = null;
		  return "FILME REMOVIDO DA HOTLIST";
	  }
	  
	  /**
	     * Exibe todos os filmes cadastrados.
	     * 
	     * @return uma string com a lista dos filmes cadastrados ou uma mensagem indicando que não há filmes.
	     */	  
	  public String exibeFilmes() {
			StringBuilder exibicao = new StringBuilder();
			for (int i = 0; i < filmes.length - 1; i++) {
				if (filmes[i] != null) {
					exibicao.append(i+1 + " - " + filmes[i].getNome() + "\n");
				}
			}
			if (exibicao.length() > 0) {
		        exibicao.setLength(exibicao.length() - 1);
		    }
			return exibicao.toString();
		}
	  
	  /**
	     * Exibe a lista de filmes na hotlist.
	     * 
	     * @return uma string com a lista dos filmes na hotlist ou uma mensagem indicando que não há filmes na hotlist.
	     */
	  public String exibirListaHot() {
	        StringBuilder exibicao = new StringBuilder();

	        for (int i = 0; i < filmesHot.length; i++) {
	            if (filmesHot[i] != null) {
	                exibicao.append((i + 1)).append(" - ").append(filmesHot[i].toString()).append("\n");
	            }
	        }
	        if (exibicao.length() > 0) {
		        exibicao.setLength(exibicao.length() - 1);
		    }
			return exibicao.toString();
		}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome,ano, local);
	}

}

			escolha = menu(scanner);

	/**
	 * Exibe o menu e captura a escolha do/a usuário/a.
	 * 
	 * @param scanner Para captura da opção do usuário/a.

		System.out.println(
				"\n---\nMENU\n" +

						"(M)Mostrar todos\n" + 
						"(D)Detalhar filme\n" + 
						"(E)Exibir HotList\n" +
						"(H)Atribuir Hot\r\n" +
						"(R)Remover Hot\r\n" +
						"(S)Sair\n" + 
						"\n" + 
						"Opção> ");

	}

	/**
	 * Interpreta a opção escolhida por quem está usando o sistema.

	 * @param opcao   Opção digitada.

			mostrarFilmes(fn);

			removeFilmesHot(fn, scanner);
