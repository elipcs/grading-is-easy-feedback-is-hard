package filmnow;

/**
 * Objeto Filme e as características individuas de cada um. 
 * 
 * @author [REDACTED]
 *
 */

public class Filme {
	
	private String nome; // nome do filme.
	private String ano; // ano de lançamento do filme.
	private String local; // local que o fime poderá ser assistido.
	private boolean hot; // se o filme está ou não na hotlist.
	
	/**
	 * Construtor onde o filme é criado.
	 * 
	 * @param nome, ano e local que caracterizam o filme.
	 *
	 */
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local = local;
		this.hot = false;
	}
	
	@Override
	/**
	 * Compara este objeto filme com outro objeto especificado para verificar se são logicamente iguais.
	 * 
	 * Dois objetos são considerados iguais se tiverem o mesmo nome e o mesmo ano.
	 * @param o O objeto a ser comparado com este filme.
	 * @return true se o objeto especificado for também um Filme e tiver o mesmo nome e o mesmo ano, false caso contrário.
	 */
	public boolean equals(Object o) {
		if (this == o) {
			return true;
		} 
		if (o == null || this.getClass() != o.getClass()) {
			return false;
		} 

		Filme other = (Filme) o;
		return this.nome.equals(other.nome) && this.ano.equals(other.ano);
		
	}
	
	/**
	 * Retorna o ano do filme.
	 */
	public String getAno() {
		return this.ano;
	}
	
	/**
	 * Modifica o atributo que caracteriza se o filme esta na hotlist ou não.
	 * @param hot True se o filme estiver na hotlist, false caso contrário.
	 */
	public void setHot(boolean hot) {
		this.hot = hot;
	}
	
	/**
	 * Retorna o nome do filme
	 */
	public String toString() {
		return this.nome;
	}
	
	/**
	 * Retorna uma string formatada com o detalhamento do filme tanto para hotlist quando para os filmes em geral.
	 * verifica também se o ano for null e o retorno que deverá ser usado se ele for.
	 */
	public String detalhes() {
		if (hot) {
			if (ano == null) {
				return "🔥 " + nome + ", " + "\n" + local;
			}
			return "🔥 " + nome + ", " + ano + "\n" + local;
		} else {
			if (ano == null) {
				return nome + ", " + "\n" + local;
			}
			
			return nome + ", " + ano + "\n" + local;
		}
		
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int TAMANHO = 100; // declara a variável que guarda o valor total do array filmes
	private static final int TAMANHO_HOT = 10; // declara a variável que guarda o valor total do array hotlist
	
	private Filme[] filmes; //uma representacao simploria da lista de filmes
	private Filme[] hotList; //uma representacao simploria da hotlist
	
	/**
	 * Cria o FilmNow 
	 * 
	 * @param declara o tamanho máximo dos arrays
	 */
	
	public FilmNow() {
		this.filmes = new Filme[TAMANHO];
		this.hotList = new Filme[TAMANHO_HOT];
	}
	
	/**
	 * Acessa a lista de filmes mantida.
	 * 
	 * @return uma cópia do array de filmes.
	 */
	
	public String[] getFilmes() {
		return formataToString(filmes);
	}
	
	/**
	 * Acessa a hotlist original.
	 * 
	 * @return uma cópia da hotlist.
	 */
	
	public String[] getHotList() {
		return formataToString(hotList);
	}
	
	/**
	 * Formata um array para impressão.
	 * 
	 * @param array Array do tipo filme que será usado como variável.
	 * @return variável saida que retorna uma "cópia" do array selecionado como um array de strings.
	 */
	private String[] formataToString(Filme[] array) {
		String[] saida = new String[array.length];
		for (int i = 0; i < array.length; i++) {
			if (array[i] != null) {
				saida[i] = array[i].toString();
			}
		} return saida;
		
	}
	
	/**
	 * Acessa os dados de um filme específico.
	 * 
	 * @param posicao Posição do filme no sistema.
	 * @return POSICAO INVALIDA caso esteja fora do intervalo permitido do array.
	 * @return a descrição do filme em formato de string.
	 * @return Dados do filme. Null se não há filme na posição.
	 */
	
	public String getFilme(int posicao) {
		if (posicao < 1 || posicao > 100) {
			return "POSIÇÃO INVÁLIDA";
		}
		if (filmes[posicao-1] != null) {
			return filmes[posicao-1].toString();
		}
		return "";
	}

	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * 
	 * @param posicao Posição do filme, nome Nome do filme, ano Ano de lançamento do filme, local Local onde o filme pode ser assistido.
	 * @return FILME INVALIDO caso o nome ou o local seja vazio.
	 * @return FILME ADICIONADO o filme foi adicionado com sucesso.
	 * @return FILME JA ADICIONADO o filme já existe no array.
	 * @return POSIÇÃO INVÁLIDA quando a posição que é passada como parâmetro esta fora do intervalo.
	 */
	public String cadastraFilme(int posicao, String nome, String ano, String local) {
		if (posicao >= 1 && posicao <= TAMANHO) {
			Filme filme = new Filme(nome, ano, local);
			
			if (nome == null || nome.equals("") || local == null || local.equals("")) {
				return "FILME INVALIDO";
			}
			
			if (!existeFilme(filme)) {
				filmes[posicao-1] = filme;
				return "FILME ADICIONADO";
			}
				
			return "FILME JA ADICIONADO";
		
		}
		
		return "POSIÇÃO INVÁLIDA";
		
	}
	
	/**
	 * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior. 
	 * 
	 * @param posicao Posição do filme, posicaoHotList Posição do filme na hotlist.
	 * @return "ADICIONADO À HOTLIST NA POSIÇÃO "..."!" quando o filme é adicionado com sucesso.
	 * @return "FILME JÁ ADICIONADO À HOTLIST" quando o filme já existe na hotlist.
	 * @return "POSIÇÃO INVÁLIDA" quando solicitam uma posição da hotlist ou do array de filmes, fora do limite estabelecido.
	 */
	
	public String adicionaHotList(int posicao, int posicaoHotList) {
		if ((posicaoHotList >= 1 && posicaoHotList <= TAMANHO_HOT) || (posicao >= 1 && posicao >= TAMANHO)) {
			if (filmes[posicao-1] == null) {
				return "POSIÇÃO INVÁLIDA";
			}
			if(hotList[posicaoHotList-1] != null) {
				removeHotList(posicaoHotList);
			}
			
			if (!existeHotList(filmes[posicao-1])) {
				filmes[posicao-1].setHot(true);
				hotList[posicaoHotList-1] = filmes[posicao-1];
				return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHotList + "!";
			}
			return "FILME JÁ ADICIONADO À HOTLIST";
			
		}
		
		return "POSIÇÃO INVÁLIDA";
		
	}
	
	/**
	 * Verifica se já existe o filme desejado no array de filmes.
	 * 
	 * @param filme filme a ser verificado a existência.
	 * @return true quando o filme já existe no array.
	 * @return false quando o filme não existe no array.
	 */
	
	private boolean existeFilme(Filme filme) {
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null && filmes[i].equals(filme)) {
				return true;
			}
		}
		return false;
	}
	
	/**
	 * Verifica se já existe o filme desejado na hotlist.
	 * 
	 * @param filme filme a ser verificado a existência.
	 * @return true quando o filme já existe na hotlist.
	 * @return false quando o filme não existe na hotlist.
	 */
	
	public boolean existeHotList(Filme filme) {
		for (int i = 0; i < hotList.length; i++) {
			if (hotList[i] != null && hotList[i].equals(filme)) {
				return true;
			}
		}
		return false;
	}
	
	
	/**
	 * Detalha o filme da posição desejada.
	 * 
	 * @param posicao Posição do filme.
	 * @return "POSIÇÃO INVÁLIDA" quando solicitam uma posição do array de filmes fora do limite estabelecido.
	 * @return retorna o filme detalhado (nome, ano e local)
	 */
	
	public String detalharFilme(int posicao) {
		if(posicao < 1 || posicao > TAMANHO) {
			return "POSIÇÃO INVÁLIDA";
		}
		if(filmes[posicao-1] == null) {
			return "";
		}
		return filmes[posicao-1].detalhes();
	}
	
	/**
	 * Remove o filme da hotlist na posição desejada.
	 *  
	 * @param posicao Posição desejada na hotlist.
	 */
	
	
	public String removeHotList(int posicao) {
		if (hotList[posicao-1] != null) {
			hotList[posicao-1].setHot(false);
			hotList[posicao-1] = null;
			return "REMOVIDO DA HOTLIST!";
		}
		return "POSIÇÃO INVÁLIDA";
	}
	
	/**
	 * Exibe os filmes do array de filmes.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado, onde tudo é realizado.
	 * @return saida Uma string que retorna a posição e o nome do filme no array de filmes.
	 */
	
	
	public String exibirFilmes(FilmNow fn) {
		String[] filmes = fn.getFilmes();
		String saida = "";
		for (int i = 0; i < filmes.length; i++) {
			if (filmes[i] != null) {
				saida += (i+1) + " - " + filmes[i].toString() + "\n";
			}
		}
		
		return saida;
		
	}
	
	/**
	 * Exibe os filmes hotlist e suas posições.
	 * 
	 * @param fn O sistema FilmNow a ser manipulado, onde tudo é realizado.
	 * @return saida Uma string que retorna a posição e o nome do filme na hotlist.
	 */
	
	public String exibirHot(FilmNow fn) {
		String[] hot = fn.getHotList();
		String saida = "";
		for (int i = 0; i < hot.length; i++) {
			if (hot[i] != null) {
				saida += (i+1) + " - " + hot[i].toString() + "\n";
			}
		}
		
		return saida;
	}

}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
