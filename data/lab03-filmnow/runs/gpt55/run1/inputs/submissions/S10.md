package filmnow;
import java.util.Objects;
/**
 * Representa um unico filme com nome, ano, local e se é favorito.
 */
public class Filme {
    private String nome;
    private String ano;
    private String local;
    private boolean hot;
    /**
     * Construtor que incializa o filme com nome, ano, local e se é favorito.
     * 
     * @param nome referencia ao nome do filme
     * @param ano referencia ao ano do filme
     * @param local referencia ao local do filme
     */
    public Filme(String nome, String ano, String local) {
        this.nome = nome;
        this.ano = ano;
        this.local = local;
        this.hot = false;
    }
    /**
     * Retorna o nome do filme.
     * 
     * @return nome string do filme
     */
    public String getNome() {
        return nome;
    }
    /**
     * Retorna o ano do filme.
     * 
     * @return ano string do filme
     */
    public String getAno() {
        return ano;
    }
    /**
     * Define se o filme é favorito.
     * 
     * @param hot boolean que indica se o filme é favorito
     */
    public void setHot(boolean hot) {
        this.hot = hot;
    }
    /**
     * Retorna um boolean do filme se ele é ou não favorito.
     * 
     * @return hot boolean que indica se o filme é favorito
     */
    public boolean getHot(){
        return hot;
    }
    /**
     * Retorna um numero inteiro que representa o hash do filme.
     * @return hashCode inteiro do filme
     */
    @Override
    public int hashCode() {
        return Objects.hashCode(nome);
    }
    /**
     * Compara dois filmes para verificar se são iguais.
     * @param o objeto a ser comparado
     * @return true se os filmes forem iguais, false caso contrário.
     */
    @Override
    public boolean equals(Object o) {
        if (o == null || !(o instanceof Filme)) {
            return false;
        }
        
        Filme novo = (Filme) o;
        return this.nome.equals(novo.nome) && this.ano.equals(novo.ano);
    }
    /**
     * Retorna uma representação em String do filme.
     * @return String formatada com o nome, ano e local do filme e com sinal se é favorito ou não.
     * @return pode retornar string formatada sem o ano, caso esteja vazio.
     */
    @Override
    public String toString() {
        String favorito = "";
        if (hot){
            favorito = "🔥 ";
        }
        if (ano.isEmpty()) {
            return favorito + nome + "\n" + local;
        }
        return  favorito + nome + ", " + ano +  "\n" + local;
    }
}

package filmnow;
/**
 * Representa o sistema de gerenciamento de filmes.
 * Contém métodos para cadastrar filmes, acessar a lista de filmes, adicionar filmes à hotlist e 
 * eliminar filmes da hotlist.
 * 
 */
public class FilmNow {
    private static final int TAMANHO = 100;
    private static final int RANKING = 10;
    
    private Filme[] filmes; //uma representacao simploria da lista de filmes
    private Filme[] hotList; //uma representacao simploria da lista de filmes favoritos

    /**
     * Cria um Array de filmes e uma hotlist.
     * A lista de filmes é limitada a 100 filmes, e a hotlist a 10 filmes.
     * A hotlist é uma lista de filmes favoritos, que podem ser acessados rapidamente.
     */
    public FilmNow() {
        filmes = new Filme[TAMANHO];
        hotList = new Filme[RANKING];
    }
    
    /**
     * Acessa a lista de filmes mantida pelo sistema.
     * @return O array de filmes formatados. Representadados pela posição e nome.
     */
    public String[] getFilmes() {
        String[] lista = filmesFormatados();
        int cont = 0;
        for (String n : lista) {
        	if (n != null) {
        		cont++;
        	}
        }
        
        String[] listaFormatada = new String[cont];
        for (int i = 0; i < lista.length; i++) {
        	if (lista[i]!= null) {
        		listaFormatada[i] = lista[i];
        	}
        }
        return listaFormatada; 
    }

    /**
     * Acessa os dados de um filme específico.
     * 
     * @param posicao Posição do filme no sistema.
     * @return Dados do filme. "POSIÇÃO INVÁLIDA" se não há filme na posição.
     * @return Se a posição for válida, retorna os dados do filme na posição especificada.
     * @return Se não houver filme na posição, retorna uma string vazia.
     */
    public String getFilme(int posicao) {
        if (!posValidaFilme(posicao)) {
            return "POSIÇÃO INVÁLIDA!";
        }

        if (filmes[posicao-1] == null) {
            return "";
        }
        return filmes[posicao-1].toString();
    }

    /**
     * Adiciona um filme em uma posição. Se já existir filme na posição, sobrescreve o anterior.
     *  
     * @param posicao Posição do filme.
     * @param nome Nome do filme.
     * @param ano Ano de lançamento do filme.
     * @param local Local onde o filme pode ser assitido.
     * 
     * @return "POSIÇÃO INVÁLIDA!" se a posição for inválida
     * @return "FILME INVÁLIDO" se o nome do filme for vazio ou nulo, ou se o local for nulo.
     * @return "FILME JÁ ADICIONADO" se o filme já existir no sistema.
     * @return "FILME ADICIONADO" se o filme for adicionado com sucesso.
     */
    public String cadastraFilme(int posicao, String nome, String ano, String local) {
        if (1 > posicao || posicao > TAMANHO) {
            return "POSIÇÃO INVÁLIDA";
        }
        if (nome == null || nome.trim().isEmpty() || local == null || local.isEmpty()) {
            return "FILME INVÁLIDO";
        }

        Filme filme = new Filme(nome, ano, local);
        if (existeFilme(filme)) {
            return "FILME JÁ ADICIONADO";
        }
        
        filmes[posicao-1] = filme;
        return "FILME ADICIONADO";
    }
    /**
     * Adiciona um filme à hotlist.
     * 
     * @param posicaoHot Posição do filme na hotlist.
     * @param posicaoFilme Posição do filme na lista de filmes.
     * 
     * @return "POSIÇÃO INVÁLIDA" se a posição do filme ou da hotlist for inválida.
     * @return "FILME JÁ ADICIONADO" se o filme já estiver na hotlist.
     * @return "ADICIONADO À HOTLIST NA POSIÇÃO" + posicao na hot list + "!" se o filme for adicionado com sucesso.
     */
    public String adicionaHot(int posicaoHot, int posicaoFilme) {
        if(!posValidaFilme(posicaoFilme) || !posValidaHot(posicaoHot) || filmes[posicaoFilme-1] == null) {
            return "POSIÇÃO INVÁLIDA";
        }
        if(filmes[posicaoFilme-1].getHot()) {
            return "FILME JÁ ADICIONADO";
        }
        
        hotList[posicaoHot-1] = filmes[posicaoFilme-1];
        filmes[posicaoFilme-1].setHot(true);
        return "ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!";
        
    }
    /**
     * Elimina um filme da hotlist.
     * 
     * @param posicaoHot
     */
    public void eliminarHot(int posicaoHot) {
        if (posValidaHot(posicaoHot) && hotList[posicaoHot-1] != null) {
            hotList[posicaoHot-1].setHot(false);
            hotList[posicaoHot-1] = null;
        }
    }
    
    /**
     * Acessa a lista de filmes favoritos.
     * 
     * @return Retorna a lista de filmes favoritos.
     */
    public String[] getHotList() {
        String[] lista = new String[RANKING];
        for (int i = 0; i < RANKING; i++) {
            if (hotList[i] != null){
                String anoFormatado = hotList[i].getAno().trim().isEmpty() ? "" : ", " + hotList[i].getAno();
                lista[i] = (i + 1) + " - " + hotList[i].getNome() + anoFormatado;
            }
        }
        return lista;
    }
    /**
     * Verifica se a posição do filme é válida.
     * @param p posição do filme
     * @return true se a posição do filme for válida, false caso contrário.
     */
    private boolean posValidaFilme(int p) {
        return 1 <= p && p <= TAMANHO;
    }
    /**
     * Verifica se a posição da hotlist é válida.
     * @param p posição da hotlist
     * @return true se a posição na hotList for válida, false caso contrário.
     */
    private boolean posValidaHot(int p) {
        return 1 <= p && p <= RANKING;
    }
    /**
     * Verifica se o filme já existe no sistema.
     * @param filme filme a ser verificado
     * @return true se o filme já existir na lista de filmes, false caso contrário.
     */
    private boolean existeFilme(Filme filme) {
        for (Filme f: filmes) {
            if(f != null && f.equals(filme)) {
                return true;
            }
        }
        return false;
    }
    /**
     * Formata a lista de filmes para exibição.
     * Cada filme é repsentado pela posicao + " - " + nome do filme.
     * @return Um array de strings representando os filmes formatados.
     */
    private String[] filmesFormatados() {
        String[] lista = new String[TAMANHO];
        int cont = 0;
        for (Filme f : filmes) {
        	if (f != null)
        		cont++;
        }
        if (cont == 0 ) {
        	return new String[0];
        }
        for (int i = 0; i < TAMANHO; i++) {
            if (filmes[i] != null) {
                lista[i] = (i + 1) + " - " + filmes[i].getNome();
            }
        }
        return lista;
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
		String nome = campos[COLUNA_NOME].trim();
		String ano = campos[COLUNA_ANO].trim();
		String local = campos[COLUNA_LOCAL].trim();

		fn.cadastraFilme(posicao, nome, ano, local);
	}

}
