package filmnow;

import java.util.Objects;

public class Filme {
	private String nome;
	private String ano;
	private String local;
	
	public Filme(String nome, String ano, String local) {
		this.nome = nome;
		this.ano = ano;
		this.local= local;
	}

	public String getNome() {
		return nome;
	}

	public String getAno() {
		return ano;
	}

	public String getLocal() {
		return local;
	}

	@Override
	public String toString() {
		return "Filme [nome=" + nome + ", ano=" + ano + ", local=" + local + "]";
	}

	@Override
	public int hashCode() {
		return Objects.hash(ano, local, nome);
	}

	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Filme other = (Filme) obj;
		return Objects.equals(ano, other.ano) && Objects.equals(local, other.local) && Objects.equals(nome, other.nome);
	}
}

package filmnow;

/**
 * Sistema que mantém os seus filmes prediletos. Podem existir 100 filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class FilmNow {
	
	private static final int MAX_FILMES = 100;
	private static final int MAX_HOTLIST = 10;
	private Filme[] filmes;
	private Filme[] hotList;

	/**
	 * Cria o FilmNow.
	 */
	public FilmNow() {
		filmes = new Filme[MAX_FILMES];
		hotList = new Filme[MAX_HOTLIST];	
	}
	
	public boolean cadastraFilme(int posicao, Filme novoFilme) {
        if (posicao < 1 || posicao > MAX_FILMES) {
            System.out.println("POSIÇÃO INVÁLIDA");
            return false;
        }

        for (Filme filme : filmes) {
            if (filme != null && filme.equals(novoFilme)) {
                System.out.println("FILME JA ADICIONADO");
                return false;
            }
        }

        if (novoFilme.getNome().isEmpty() || novoFilme.getLocal().isEmpty()) {
            System.out.println("FILME INVALIDO");
            return false;
        }

        filmes[posicao - 1] = novoFilme;
        System.out.println("FILME ADICIONADO");
        return true;
    }
	
	
    public void mostrarTodos() {
        for (int i = 0; i < filmes.length; i++) {
            if (filmes[i] != null) {
                System.out.println((i + 1) + " - " + filmes[i]);
            }
        }
    }

    public void detalharFilme(int posicao) {
        if (posicao < 1 || posicao > MAX_FILMES || filmes[posicao - 1] == null) {
            System.out.println("POSIÇÃO INVÁLIDA!");
            return;
        }

        Filme filme = filmes[posicao - 1];
        boolean isHot = false;
        for (Filme hotFilm : hotList) {
            if (hotFilm != null && hotFilm.equals(filme)) {
                isHot = true;
                break;
            }
        }
        
        if (isHot) {
            System.out.println("🔥 " + filme);
        } else {
            System.out.println(filme);
        }
        
        System.out.println(filme.getLocal());
    }

    public boolean adicionarHot(int posicaoFilme, int posicaoHot) {
        if (posicaoHot < 1 || posicaoHot > MAX_HOTLIST) {
            System.out.println("POSIÇÃO INVALIDA");
            return false;
        }

        Filme filme = filmes[posicaoFilme - 1];
        if (filme == null) {
            System.out.println("FILME NAO EXISTE");
            return false;
        }

        for (Filme hotFilm : hotList) {
            if (hotFilm != null && hotFilm.equals(filme)) {
                System.out.println("FILME JA NA HOTLIST");
                return false;
            }
        }

        hotList[posicaoHot - 1] = filme;
        System.out.println("ADICIONADO À HOTLIST NA POSIÇÃO " + posicaoHot + "!");
        return true;
    }

    public void exibirHotList() {
        for (int i = 0; i < hotList.length; i++) {
            if (hotList[i] != null) {
                System.out.println((i + 1) + " - " + hotList[i]);
            }
        }
    }

    public void removerHot(int posicaoHot) {
        if (posicaoHot < 1 || posicaoHot > MAX_HOTLIST || hotList[posicaoHot - 1] == null) {
            System.out.println("POSIÇÃO INVALIDA");
            return;
        }

        hotList[posicaoHot - 1] = null;
        System.out.println("REMOVIDO DA HOTLIST");
    }
}

package filmnow;

import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;

/**
 * Lógica para ler de arquivos csv os dados necessários e povoar o sistema de filmes. 
 * 
 * @author [REDACTED]
 *
 */
public class LeitorFilmNow {

	private static final int COLUNA_POSICAO = 0;
	private static final int COLUNA_NOME = 1;
	private static final int COLUNA_ANO = 2;
	private static final int COLUNA_LOCAL = 3;


	/**
	 * Lê filmes de um arquivo CSV e os coloca no sistema.
	 * @param arquivoFilmes Caminho para arquivo contendo os filmes.
	 * @param fn O sistema FilmNow a manipular.
	 * @return O número de filmes adicionados ao sistema.
	 * @throws IOException Caso não tenhamos permissão de ler o arquivo.
	 * @throws FileNotFoundException Caso o arquivo não exista.
	 */
	public int carregaContatos(String arquivoFilmes, FilmNow fn) throws FileNotFoundException, IOException {
		int carregados = 0;

		try (BufferedReader br = new BufferedReader(new FileReader(arquivoFilmes))) {
			String linha;
			while ((linha = br.readLine()) != null) {
				carregados += 1;
				if (carregados == 1) {
					// pulamos a primeira linha, o cabeçalho
					continue;
				}
				String[] campos = linha.split(",");
				processaLinhaCsvFilmes(campos, fn);
			}
		}
		
		return carregados;
	}

	
	/**
	 * Coloca no sistema os dados de uma linha do arquivo de filmes inicial. 
	 * 
	 * @param campos As informações lidas do csv. 
	 * @param fn O sistema FilmNow a manipular. 
	 */
	private void processaLinhaCsvFilmes(String[] campos, FilmNow fn) {
		try {
			int posicao = Integer.parseInt(campos[COLUNA_POSICAO]);
			String nome = campos[COLUNA_NOME].trim();
			String ano = campos[COLUNA_ANO].trim();
			String local = campos[COLUNA_LOCAL].trim();
	
			Filme novoFilme = new Filme(nome, ano, local);
	        fn.cadastraFilme(posicao, novoFilme);
		} catch (NumberFormatException e) {
            System.out.println("Erro ao processar os dados: formato inválido.");
        }
	}

}

package filmnow;

import java.util.Scanner;

public class MainFilmNow {
    public static void main(String[] args) {
        FilmNow filmNow = new FilmNow();
        Scanner scanner = new Scanner(System.in);
        String opcao;

        do {
            System.out.println("(A)Adicionar filme");
            System.out.println("(M)Mostrar todos");
            System.out.println("(D)Detalhar filme");
            System.out.println("(E)Exibir HotList");
            System.out.println("(H)Atribuir Hot");
            System.out.println("(R)Remover Hot");
            System.out.println("(S)Sair");
            System.out.print("\nOpção> ");
            opcao = scanner.nextLine().trim().toUpperCase();

            switch (opcao) {
                case "A":
                    adicionarFilme(scanner, filmNow);
                    break;
                case "M":
                    filmNow.mostrarTodos();
                    break;
                case "D":
                    detalharFilme(scanner, filmNow);
                    break;
                case "E":
                    filmNow.exibirHotList();
                    break;
                case "H":
                    atribuirHot(scanner, filmNow);
                    break;
                case "R":
                    removerHot(scanner, filmNow);
                    break;
                case "S":
                    System.out.println("Encerrando o programa.");
                    break;
                default:
                    System.out.println("OPÇÃO INVÁLIDA!");
            }
        } while (!opcao.equals("S"));
        
        scanner.close();
    }

    private static void adicionarFilme(Scanner scanner, FilmNow filmNow) {
        System.out.print("Posição> ");
        int posicao = Integer.parseInt(scanner.nextLine().trim());
        System.out.print("Nome> ");
        String nome = scanner.nextLine().trim();
        System.out.print("Ano> ");
        String ano = scanner.nextLine().trim();
        System.out.print("Local> ");
        String local = scanner.nextLine().trim();

        Filme novoFilme = new Filme(nome, ano, local);
        filmNow.cadastraFilme(posicao, novoFilme);
    }

    private static void detalharFilme(Scanner scanner, FilmNow filmNow) {
        System.out.print("Filme> ");
        int posicao = Integer.parseInt(scanner.nextLine().trim());
        filmNow.detalharFilme(posicao);
    }

    private static void atribuirHot(Scanner scanner, FilmNow filmNow) {
        System.out.print("Filme> ");
        int posicaoFilme = Integer.parseInt(scanner.nextLine().trim());
        System.out.print("Posicao> ");
        int posicaoHot = Integer.parseInt(scanner.nextLine().trim());
        filmNow.adicionarHot(posicaoFilme, posicaoHot);
    }

    private static void removerHot(Scanner scanner, FilmNow filmNow) {
        System.out.print("Posicao> ");
        int posicaoHot = Integer.parseInt(scanner.nextLine().trim());
        filmNow.removerHot(posicaoHot);
    }
}
